\section{اهداف یادگیری}
پس از مطالعهٔ این درس باید بتوانید:
\begin{objectives}
  \item محدودیت‌هایی از \lr{RADIUS} را که \lr{Diameter} برای رفع آن‌ها طراحی شد \textbf{توضیح دهید}.
  \item گره‌ها و عامل‌های \lr{Diameter}، اتصال‌های همتا و مسیریابی بر پایهٔ قلمرو را \textbf{توصیف کنید}.
  \item سرآیند و \acro{AVP} پیام \lr{Diameter}، از جمله پرچم‌ها، شناسهٔ فروشنده و لایه‌گذاری را \textbf{رمزگشایی کنید} و
        طول پیام‌ها را \textbf{محاسبه کنید}.
  \item فرمان‌های پایه و کلاس‌های کد نتیجه را \textbf{فهرست کنید} و کاربردهای \lr{S6a}، \lr{Gx}، \lr{Rx} و \lr{Gy} در
        \acro{LTE} را \textbf{دنبال کنید}.
  \item سهمیهٔ کنترل اعتبار را \textbf{اندازه‌گذاری کنید} و بار تراکنشی یک \acro{HSS} را \textbf{تخمین بزنید}.
  \item توضیح دهید چرا نسل پنجم \lr{Diameter} را با واسط‌های سرویس‌محور \lr{HTTP/2} جایگزین کرد و این دو چگونه با هم
        \textbf{همکاری می‌کنند}.
\end{objectives}

\section{دانش پیش‌نیاز}
به نقاط مرجع \acro{EPC} از درس‌های ۵ و ۹ (\lr{S6a}، \lr{Gx}، \lr{Rx}، \lr{Gy}) و آشنایی کاربردی با پروتکل‌های انتقال
نیاز دارید: جریان بایت قابل اطمینان \acro{TCP} و پیوندهای پیام‌محور و چندخانه‌ای \acro{SCTP} (درس ۶). پروتکل‌های
دودویی معمولاً از عناصر \emph{نوع--طول--مقدار} (\acro{TLV}) ساخته می‌شوند؛ \acro{AVP} در \lr{Diameter} یکی از آن‌هاست.

\paragraph{واژه‌های کلیدی} واژگان این درس همراه با معادل انگلیسی آن‌ها:
\begin{keyterms}
احراز هویت، مجوزدهی، حساب‌داری & \lr{AAA} & چارچوب احراز هویت، مجوزدهی و حساب‌داری. \\
جفت ویژگی--مقدار & \lr{AVP} & عنصر اطلاعاتی \acro{TLV} در \lr{Diameter}. \\
کد فرمان & \lr{Command code} & یک جفت درخواست/پاسخ را مشخص می‌کند (مثلاً ۳۱۶ = \lr{Update-Location}). \\
شناسهٔ کاربرد & \lr{Application-ID} & واسط را مشخص می‌کند (مثلاً ۱۶۷۷۷۲۵۱ = \lr{S6a/S6d}). \\
قلمرو & \lr{Realm} & دامنهٔ مدیریتی که برای مسیریابی به کار می‌رود (مثلاً دامنهٔ یک اپراتور). \\
عامل رله / پراکسی & \lr{Relay / proxy agent} & عامل‌هایی که پیام‌ها را هدایت می‌کنند (پراکسی می‌تواند آن‌ها را تغییر دهد). \\
عامل مسیریابی / لبهٔ \lr{Diameter} & \lr{DRA / DEA} & عامل مسیریابی درون شبکه و عامل لبه در مرز شبکه. \\
تبادل قابلیت، پاسبان، قطع & \lr{CER / DWR / DPR} & فرمان‌های پایهٔ مدیریت همتا. \\
کد نتیجه & \lr{Result-Code} & نتیجهٔ یک درخواست (۲۰۰۱ = موفقیت). \\
کنترل اعتبار & \lr{Credit control} & صورت‌حساب برخط با اعطای سهمیه (\lr{Gy}). \\
\end{keyterms}

\section{مقدمه و انگیزه}
هر بار که یک گوشی \acro{LTE} متصل می‌شود، \acro{MME} از \acro{HSS} بردارهای احراز هویت و نمایهٔ مشترک را درخواست
می‌کند؛ هر بار که یک تماس \acro{VoLTE} آغاز می‌شود، \acro{IMS} از \acro{PCRF} یک حامل تضمین‌شده می‌خواهد؛ و هر چند
مگابایت یک بار، \lr{P-GW} یک مشترک پیش‌پرداخت از سامانهٔ صورت‌حساب اعتبار بیشتری می‌خواهد. همهٔ این گفت‌وگوها -- میان
گره‌های کنترلی هسته و بدون تماس با دادهٔ کاربر -- از یک پروتکل استفاده می‌کنند: \lr{Diameter} \cite{rfc6733}.

\lr{Diameter} از نسل \lr{RADIUS} \cite{rfc2865}، پروتکل \acro{AAA} دوران شماره‌گیری تلفنی، است (نام آن یک بازی با واژه
است: قطر دو برابر شعاع است). \lr{RADIUS} نمی‌توانست نیازهای سطح اپراتور -- قابلیت اطمینان، ویژگی‌های بزرگ، پیام‌های
آغازشده از کارساز و رومینگ میان قلمروها -- را برآورده کند؛ از این رو \acro{IETF} پروتکل \lr{Diameter} را به‌صورت یک
پروتکل پایهٔ توسعه‌پذیر به‌علاوهٔ \emph{کاربردها} طراحی کرد. \acro{3GPP} آن را برای \acro{IMS} (انتشار ۵) پذیرفت و ستون
فقرات سیگنالینگ \acro{EPC} کرد. در نسل پنجم این پروتکل جای خود را به \acro{API}های \lr{HTTP/2} می‌دهند، اما شبکه‌های
\lr{Diameter} تا سال‌ها ترافیک نسل چهارم را حمل خواهند کرد و مفاهیم طراحی آن -- قلمروها، عامل‌ها و تبادل قابلیت -- در
معماری سرویس‌محور نسل پنجم دوباره دیده می‌شوند.

\subsection{گره‌ها و عامل‌ها}
یک \emph{کارخواه} \lr{Diameter} (مثلاً \acro{MME}) درخواست‌ها را به یک \emph{کارساز} (مثلاً \acro{HSS}) می‌فرستد. میان
آن‌ها \emph{عامل‌ها} ممکن است پیام‌ها را هدایت کنند:
\begin{itemize}
  \item \textbf{رله} بر اساس اطلاعات مسیریابی و بدون بررسی کاربرد هدایت می‌کند؛
  \item \textbf{پراکسی} هدایت می‌کند و می‌تواند برای اعمال سیاست پیام‌ها را تغییر دهد؛
  \item عامل \textbf{تغییر مسیر} به‌جای هدایت، با نشانی کارساز درست پاسخ می‌دهد؛
  \item عامل \textbf{ترجمه} به پروتکل دیگری (مثلاً \lr{RADIUS}) تبدیل می‌کند.
\end{itemize}
اپراتورها \textbf{عامل‌های مسیریابی \lr{Diameter}} (\acro{DRA}) را به‌عنوان مسیریاب‌های مرکزی سیگنالینگ به کار می‌گیرند
تا هر کارخواه به‌جای یک شبکهٔ کامل فقط به دو \acro{DRA} اتصال داشته باشد، و \textbf{عامل‌های لبهٔ \lr{Diameter}}
(\acro{DEA}) را در مرز به سوی شرکای رومینگ (از طریق یک مبادلهٔ \acro{IP}، یعنی \acro{IPX}) قرار می‌دهند که توپولوژی
داخلی را پنهان می‌کنند.

\subsection{همتاها و قلمروها}
دو گره‌ای که مستقیماً به هم وصل‌اند \emph{همتا} هستند. پیش از مبادلهٔ پیام‌های کاربردی، یک \emph{تبادل قابلیت}
(\lr{CER/CEA}) انجام می‌دهند و هویت خود (\lr{Origin-Host}، \lr{Origin-Realm}) و کاربردهای پشتیبانی‌شده را اعلام
می‌کنند. هر گره یک \emph{جدول همتا} و یک \emph{جدول مسیریابی قلمرو} نگه می‌دارد. درخواست \lr{Destination-Realm}
(برای درخواست‌های مسیریابی‌شده الزامی) و به‌طور اختیاری \lr{Destination-Host} را حمل می‌کند؛ گره‌ها آن را به همتایی
می‌فرستند که آن قلمرو و کاربرد را خدمت می‌دهد. در شبکه‌های \acro{3GPP} قلمروها از الگوی
\lr{\texttt{epc.mnc<MNC>.mcc<MCC>.3gppnetwork.org}} پیروی می‌کنند \cite{ts23003}، بنابراین یک \acro{MME} میزبان
می‌تواند فقط با استفاده از \acro{IMSI} درخواست \lr{S6a} را به \acro{HSS} خانگی مشترک هدایت کند.

\section{توضیحات فنی تفصیلی}

\subsection{ساختار پیام}
پیام \lr{Diameter} یک سرآیند ۲۰ بایتی است که \acro{AVP}ها پس از آن می‌آیند (\cref{fig:header}) \cite{rfc6733}.
\begin{itemize}
  \item \textbf{نسخه} = ۱؛ \textbf{طول پیام} سرآیند را نیز شامل می‌شود و مضربی از ۴ است.
  \item \textbf{پرچم‌های فرمان}: \lr{R} (درخواست، در پاسخ‌ها صفر)، \lr{P} (قابل پراکسی)، \lr{E} (پاسخ خطا)،
        \lr{T} (احتمالاً پس از جایگزینی دوباره ارسال شده).
  \item \textbf{کد فرمان} (۲۴ بیت) فرمان را مشخص می‌کند؛ درخواست و پاسخ کد مشترک دارند.
  \item \textbf{شناسهٔ کاربرد} (۳۲ بیت) کاربرد را مشخص می‌کند (۰ برای پروتکل پایه).
  \item \textbf{شناسهٔ گام‌به‌گام}: پاسخ‌ها را با درخواست‌ها روی یک گام تطبیق می‌دهد؛ هر عامل آن را جایگزین می‌کند.
  \item \textbf{شناسهٔ سرتاسری}: برای هر درخواست در کل مسیر یکتاست؛ برای تشخیص تکراری‌ها به کار می‌رود.
\end{itemize}
هر \acro{AVP} یک کد ۳۲ بیتی، پرچم‌های \lr{V} (ویژهٔ فروشنده: یک \lr{Vendor-ID} پس از آن می‌آید؛ ۱۰۴۱۵ برای
\acro{3GPP})، \lr{M} (الزامی: گیرنده باید آن را بفهمد یا پیام را رد کند) و \lr{P} (امنیت سرتاسری، منسوخ)، یک طول ۲۴
بیتی که سرآیند و داده را بدون لایه‌گذاری پوشش می‌دهد، و داده‌ای دارد که تا مرز ۴ بایت لایه‌گذاری می‌شود. انواع داده شامل
\lr{Integer32/64}، \lr{Unsigned32/64}، \lr{OctetString}، \lr{UTF8String}، \lr{DiameterIdentity}، \lr{Address}،
\lr{Enumerated} و \emph{\lr{Grouped}} است -- \acro{AVP}ای که داده‌اش خود دنباله‌ای از \acro{AVP}هاست و به
\lr{Diameter} اجازه می‌دهد ساختارهای تودرتو مانند نمایهٔ اشتراک را بیان کند.

\subsection{فرمان‌ها و کدهای نتیجه}
نام فرمان‌ها جفتی از اختصارهاست که به \lr{R} (درخواست) و \lr{A} (پاسخ) ختم می‌شوند. پروتکل پایه مدیریت همتا را تعریف
می‌کند -- \textbf{\lr{CER/CEA}} (\lr{Capabilities-Exchange}، کد ۲۵۷)، \textbf{\lr{DWR/DWA}} (\lr{Device-Watchdog}،
۲۸۰)، \textbf{\lr{DPR/DPA}} (\lr{Disconnect-Peer}، ۲۸۲) -- و نیز فرمان‌های نشست مانند \textbf{\lr{RAR/RAA}}
(\lr{Re-Auth}، ۲۵۸)، \textbf{\lr{STR/STA}} (\lr{Session-Termination}، ۲۷۵) و \textbf{\lr{ASR/ASA}}
(\lr{Abort-Session}، ۲۷۴). پاسخ‌ها یک \acro{AVP} از نوع \lr{\texttt{Result-Code}} دارند که رقم هزارگان آن کلاس را

قواعد \acro{PCC} را در \lr{CCA-Initial} دریافت می‌کند؛ \acro{PCRF} بعدها تغییرات را با \lr{RAR} اعمال می‌کند؛
\lr{CCR-Update} رویدادها (مثلاً آستانه‌های مصرف) را گزارش می‌دهد؛ \lr{CCR-Termination} نشست را پایان می‌دهد.
\paragraph{\lr{Rx}} \cite{ts29214}: \lr{P-CSCF} رسانهٔ یک تماس را در \lr{AAR} توصیف می‌کند؛ \acro{PCRF} کیفیت خدمات را
استخراج و قواعد را از طریق \lr{Gx} نصب می‌کند (\cref{fig:volte})؛ \lr{STR} نشست رسانه را پایان می‌دهد.
\paragraph{کنترل اعتبار \lr{Gy}} \cite{rfc8506}: \acro{PCEF} در \lr{CCR-Initial} برای یک گروه نرخ‌گذاری سهمیه (زمانی
یا حجمی) درخواست می‌کند، با \lr{CCR-Update} مصرف را گزارش و سهمیهٔ بیشتر درخواست می‌کند و با \lr{CCR-Termination}
سهمیهٔ مصرف‌نشده را برمی‌گرداند؛ اگر اعتبار تمام شود، \acro{OCS} با نشانگر واحد نهایی پاسخ می‌دهد.

\subsection{\lr{Diameter} در نسل پنجم}
هستهٔ نسل پنجم \lr{Diameter} را با واسط‌های سرویس‌محور \lr{HTTP/2} جایگزین می‌کند (درس‌های ۱۸ و ۱۹):
\lr{S6a} $\to$ \lr{N8 (Nudm)}، \lr{Gx} $\to$ \lr{N7 (Npcf\_SMPolicyControl)}، \lr{Rx} $\to$
\lr{N5 (Npcf\_PolicyAuthorization)} و \lr{Gy/Gz} $\to$ \lr{N40 (Nchf)}. دلایل: (۱) فناوری و ابزارهای وب (\acro{REST}،
\acro{JSON}، \lr{OpenAPI}) همه‌جا در دسترس‌اند؛ (۲) توابع شبکه به‌جای تکیه بر جدول‌های ایستای همتا، یکدیگر را به‌طور
پویا ثبت و کشف می‌کنند (\acro{NRF})؛ (۳) پیاده‌سازی‌های بی‌حالت و ابرزاد با متعادل‌کننده‌های بار \acro{HTTP} آسان‌تر
مقیاس‌پذیرند. در دوران همزیستی نسل چهارم و پنجم، توابع همکاری میان این دو جهان ترجمه می‌کنند و \acro{HSS} و
\acro{UDM} اغلب یک محصول ترکیبی واحدند.

\section{مدل‌های ریاضی}

\subsection{طول \lr{AVP} و پیام}\label{sec:len}
\paragraph{قاعده} برای یک \acro{AVP} با $\ell$ بایت داده و طول سرآیند $h = 8$ (یا ۱۲ با \lr{Vendor-ID}):
\begin{equation}\label{eq:avplen}
  L_{\mathrm{AVP}} = h + \ell, \qquad
  L_{\mathrm{on\,wire}} = 4\left\lceil \frac{h+\ell}{4}\right\rceil,
\end{equation}
و طول پیام برابر $20 + \sum L_{\mathrm{on\,wire}}$ روی همهٔ \acro{AVP}های سطح بالاست.
\paragraph{مثال} \lr{Origin-Host} با مقدار \lr{\texttt{mme01.epc.mnc010.mcc432.3gppnetwork.org}} ۳۹ نویسه دارد:
$L_{\mathrm{AVP}} = 8+39 = 47$ که روی سیم به ۴۸ بایت لایه‌گذاری می‌شود. \lr{Origin-Realm} با مقدار
\lr{\texttt{epc.mnc010.mcc432.3gppnetwork.org}} (۳۳ نویسه): ۴۱، لایه‌گذاری‌شده به ۴۴. فقط \acro{AVP}های هویت حدود ۱۰۰
بایت در هر پیام هزینه دارند؛ به همین دلیل پیام‌های \lr{Diameter} حتی برای درخواست‌های ساده معمولاً چند صد بایت‌اند.

\subsection{تشخیص خرابی با پاسبان}\label{sec:watchdog}
همتا وقتی به مدت $T_\mathrm{w}$ ترافیکی دریافت نکرده باشد \lr{DWR} می‌فرستد (مقدار توصیه‌شده ۳۰ ثانیه، با لرزش تصادفی
$\pm2$ ثانیه برای جلوگیری از همگام‌شدن). اگر در $T_\mathrm{w}$ دیگری پاسخی نرسد، اتصال مشکوک تلقی می‌شود و ترافیک به
همتای جایگزین منتقل می‌شود. بنابراین بدترین زمان تشخیص روی یک اتصال بیکار تقریباً برابر است با
\begin{equation}\label{eq:tdet}
  T_{\mathrm{det}} \approx 2\,T_\mathrm{w},
\end{equation}
یعنی با مقادیر پیش‌فرض حدود یک دقیقه؛ روی یک اتصال پرکار، درخواست‌های کاربردی بی‌پاسخ خرابی را بسیار زودتر آشکار
می‌کنند. اپراتورها $T_\mathrm{w}$ را روی پیوندهای حیاتی کاهش می‌دهند (مثلاً ۶ ثانیه) و ترافیک پاسبان را با جایگزینی
سریع‌تر معاوضه می‌کنند (\cref{exr:watchdog}).

\subsection{بار تراکنشی \lr{HSS}}\label{sec:hssload}
\paragraph{مدل} $N$ مشترک هر کدام روزانه $a$ بار متصل می‌شوند و $u$ بار \acro{TAU} همراه با تغییر \acro{MME} انجام
می‌دهند؛ هر کدام یک \lr{AIR} و یک \lr{ULR} لازم دارد (\lr{AIR} فقط اگر \acro{MME} بردار نداشته باشد)؛ از رویه‌های دیگر
صرف‌نظر می‌شود. نرخ میانگین تراکنش \lr{S6a} برابر است با
\begin{equation}\label{eq:hss}
  \lambda_{\mathrm{S6a}} \approx \frac{N\,(a+u)\,\bar n}{86\,400}\ \text{transactions/s},
\end{equation}
که در آن $\bar n$ میانگین تعداد تراکنش‌ها در هر رویه است.
\paragraph{مثال} $N = 10^7$، $a+u = 3$ و $\bar n = 2$: به‌طور میانگین $\lambda \approx 694$ تراکنش بر ثانیه. ساعت اوج
چند برابر بیشتر است و راه‌اندازی مجدد یک مخزن \acro{MME} می‌تواند انفجارهایی بسیار بالاتر از میانگین ایجاد کند (درس ۹)؛
بنابراین بخش‌های جلویی \acro{HSS} برای اوج‌ها اندازه‌گذاری می‌شوند، نه برای میانگین‌ها.

\subsection{اندازه‌گذاری سهمیهٔ کنترل اعتبار}\label{sec:quota}
\paragraph{مسئله} سهمیهٔ حجمی \lr{Gy} چقدر باید بزرگ باشد؟
\paragraph{مدل} کاربر داده را با نرخ $r$ (بایت بر ثانیه) مصرف می‌کند؛ هر اعطا $Q$ بایت می‌دهد؛ \acro{PCEF} وقتی سهمیهٔ
فعلی مصرف شد گزارش می‌دهد و سهمیهٔ جدید درخواست می‌کند.
\paragraph{نتایج} تعداد \lr{CCR-Update} به ازای هر کاربر در واحد زمان و بیشینهٔ بیش‌مصرف (مصرفی که اپراتور نمی‌تواند
هزینه‌اش را بگیرد اگر اعتبار کاربر درست پس از اعطا تمام شود) عبارت‌اند از
\begin{equation}\label{eq:quota}
  \lambda_{\mathrm{CCR}} = \frac{r}{Q}, \qquad O_{\max} \approx Q .
\end{equation}
\paragraph{مثال و تفسیر} $r = \qty{2}{\mega\bit\per\second} = 250$ کیلوبایت بر ثانیه و $Q = 10$ مگابایت: هر ۴۰ ثانیه یک
به‌روزرسانی (۹۰ در ساعت) و تا ۱۰ مگابایت درآمد در معرض خطر برای هر کاربر. دو برابر کردن $Q$ سیگنالینگ را نصف می‌کند
اما خطر را دو برابر -- بنابراین اپراتورها به کاربران کم‌موجودی سهمیه‌های کوچک و به کاربران پرموجودی سهمیه‌های بزرگ
می‌دهند. یک زمان اعتبار محدود می‌کند که سهمیه تا چه مدت نگه داشته شود.

\begin{procedurebox}{\lr{S6a} هنگام اتصال}\label{proc:s6a}
\begin{steps}
  \item \acro{MME} برای یک \acro{IMSI} ناشناخته به بردار نیاز دارد. پیام \lr{AIR} (کد ۳۱۸، شناسهٔ کاربرد ۱۶۷۷۷۲۵۱،
        پرچم \lr{R} روشن) را با \lr{Session-Id}، \lr{Auth-Session-State}، \lr{Origin-Host/Realm}،
        \lr{Destination-Realm} استخراج‌شده از \acro{IMSI}، \lr{User-Name} = \acro{IMSI}، \lr{Visited-PLMN-Id} و
        \lr{Requested-EUTRAN-Authentication-Info} می‌سازد.
  \item \acro{DRA} بر اساس \lr{Destination-Realm} و شناسهٔ کاربرد آن را به یک بخش جلویی \acro{HSS} هدایت می‌کند؛
        \acro{HSS} پاسخ \lr{AIA} را با \lr{Result-Code} ۲۰۰۱ و \lr{Authentication-Info} (\acro{AVP}های گروهی
        \lr{E-UTRAN-Vector}) برمی‌گرداند.
  \item پس از احراز هویت، \acro{MME} پیام \lr{ULR} (۳۱۶) را با \lr{RAT-Type} و \lr{ULR-Flags} می‌فرستد؛ \acro{HSS}
        هویت \acro{MME} را ذخیره می‌کند، اگر \acro{MME} پیشینی وجود داشته باشد به آن \lr{CLR} می‌فرستد و \lr{ULA} را با
        \lr{Subscription-Data} برمی‌گرداند.
  \item خطاها به‌صورت \lr{Experimental-Result} ظاهر می‌شوند، مثلاً ۵۰۰۱ (کاربر ناشناخته): \acro{MME} اتصال را با علت
        \acro{NAS} مناسب رد می‌کند.
\end{steps}
\end{procedurebox}

\section{مثال‌های حل‌شده}

\begin{examplebox}{رمزگشایی یک سرآیند}\label{ex:decode}
\textbf{مسئله.} ۲۰ بایت نخست یک پیام (به‌صورت هگز) عبارت‌اند از
\lr{\texttt{01 00 01 2C 80 00 01 3C 01 00 00 23 \ldots}}. نسخه، طول، پرچم‌ها، فرمان و کاربرد را رمزگشایی کنید.

\textbf{حل.} نسخه ۱؛ طول $\mathtt{0x00012C} = 300$ بایت؛ پرچم‌ها $\mathtt{0x80}$ = فقط \lr{R} (یک درخواست، غیرقابل
پراکسی)؛ فرمان $\mathtt{0x00013C} = 316$ = \lr{Update-Location}؛ شناسهٔ کاربرد $\mathtt{0x01000023} = 16777251$ =
\lr{S6a}. پس: یک \lr{ULR} از یک \acro{MME}. (درخواست‌های واقعی \lr{S6a} معمولاً \lr{P} را نیز روشن می‌کنند.)
\end{examplebox}

\begin{examplebox}{اندازهٔ پیام}\label{ex:size}
\textbf{مسئله.} یک \lr{AIR} شامل این‌هاست: \lr{Session-Id} (۶۰ بایت داده)، \lr{Auth-Session-State}
(\lr{Unsigned32})، \lr{Origin-Host} و \lr{Origin-Realm} مانند \cref{sec:len}، \lr{Destination-Realm} (۳۳ بایت)،
\lr{User-Name} (\acro{IMSI} ۱۵ رقمی)، \lr{Visited-PLMN-Id} (\lr{OctetString} سه‌بایتی، ویژهٔ فروشنده) و
\lr{Requested-EUTRAN-Authentication-Info} (گروهی، ویژهٔ فروشنده، شامل یک \acro{AVP} از نوع \lr{Unsigned32} به نام
\lr{Number-Of-Requested-Vectors}، ویژهٔ فروشنده). طول آن را محاسبه کنید.

\textbf{حل.} \lr{Session-Id}: $4\lceil68/4\rceil = 68$؛ \lr{Auth-Session-State}: $8+4 = 12$؛ \lr{Origin-Host}: ۴۸؛
\lr{Origin-Realm}: ۴۴؛ \lr{Destination-Realm}: ۴۴؛ \lr{User-Name}: $4\lceil23/4\rceil = 24$؛ \lr{Visited-PLMN-Id}:
$4\lceil15/4\rceil = 16$؛ \acro{AVP} درونی $12+4 = 16$، پس \acro{AVP} گروهی $12+16 = 28$ است. مجموع
$20 + 68+12+48+44+44+24+16+28 = 304$ بایت.
\end{examplebox}

\begin{examplebox}{سیاست سهمیه}\label{ex:quota}
\textbf{مسئله.} یک \acro{OCS} باید حداکثر ۲۰۰۰ \lr{CCR-Update} در ثانیه را برای $10^6$ کاربر دادهٔ فعال با نرخ
میانگین ۵۰ کیلوبایت بر ثانیه پردازش کند. کوچک‌ترین سهمیهٔ یکنواخت چیست؟

\textbf{حل.} $\lambda_{\mathrm{total}} = N r/Q \le 2000 \Rightarrow Q \ge 10^6\times5\times10^4/2000 = 25$ مگابایت.
اپراتور ۲۵ مگابایت (یا بیشتر) را به کار می‌برد و خطر را با زمان‌سنج‌های اعتبار و سهمیه‌های کوچک‌تر برای کاربران
کم‌موجودی محدود می‌کند.
\end{examplebox}

\section{نکته‌های کلیدی}
\begin{takeawaybox}
\begin{itemize}
  \item \lr{Diameter} = \acro{AAA} قابل اطمینان (\lr{TCP/SCTP}) و توسعه‌پذیر: پروتکل پایه + کاربردهایی که با شناسهٔ
        کاربرد مشخص می‌شوند.
  \item سرآیند ۲۰ بایتی (نسخه، طول، پرچم‌های \lr{R/P/E/T}، کد فرمان، شناسهٔ کاربرد، شناسه‌های گام‌به‌گام و سرتاسری)
        + \acro{AVP}هایی که تا ۳۲ بیت لایه‌گذاری می‌شوند.
  \item همتاها: \lr{CER/CEA}، \lr{DWR/DWA}، \lr{DPR/DPA}؛ مسیریابی بر اساس \lr{Destination-Realm} از طریق رله‌ها،
        پراکسی‌ها و \lr{DRA/DEA}.
  \item کاربردهای \acro{LTE}: \lr{S6a} (\lr{AIR, ULR, CLR, IDR})، \lr{Gx} (\lr{CCR/CCA}، \lr{RAR})، \lr{Rx}
        (\lr{AAR})، \lr{Gy} (کنترل اعتبار).
  \item کلاس‌های \lr{Result-Code} از \lr{1xxx} تا \lr{5xxx}؛ خطاهای ویژهٔ \acro{3GPP} در \lr{Experimental-Result}.
  \item اندازه‌گذاری سهمیه سیگنالینگ $r/Q$ را با خطر درآمدی $\approx Q$ معاوضه می‌کند.
  \item نسل پنجم \lr{Diameter} را با \lr{HTTP/2 SBI} جایگزین می‌کند؛ در دوران همزیستی نسل ۴ و ۵ همکاری لازم است.
\end{itemize}
\end{takeawaybox}

\section{تمرین‌ها}

\begin{exercisebox}{1}{پیام‌های آغازشده از کارساز}\label{exr:srv}
دو وضعیت در \acro{EPC} را نام ببرید که در آن کارساز باید یک درخواست \lr{Diameter} را آغاز کند و توضیح دهید چرا
\lr{RADIUS} به‌طور بومی نمی‌توانست از آن‌ها پشتیبانی کند.
\end{exercisebox}

\begin{exercisebox}{1}{دو شناسه}\label{exr:ids}
توضیح دهید چرا \lr{Diameter} هم به شناسهٔ گام‌به‌گام و هم به شناسهٔ سرتاسری نیاز دارد. وقتی یک رله درخواستی را هدایت
می‌کند، بر سر هر یک چه می‌آید؟
\end{exercisebox}

\begin{exercisebox}{2}{طول \lr{AVP}}\label{exr:avp}
طول روی سیم این موارد را محاسبه کنید: (الف) یک \acro{AVP} از نوع \lr{Result-Code}؛ (ب) یک \acro{AVP} ویژهٔ فروشنده از
نوع \lr{UTF8String} شامل \acro{APN} با مقدار \lr{\texttt{internet}}؛ (ج) یک \acro{AVP} گروهی شامل (الف) و (ب).
\end{exercisebox}

\begin{exercisebox}{2}{تنظیم پاسبان}\label{exr:watchdog}
یک \acro{DRA} ۴۰۰ اتصال همتای بیکار دارد. نرخ پیام‌های پاسبان را برای $T_\mathrm{w} = 30$ ثانیه و ۶ ثانیه و بدترین زمان
تشخیص خرابی متناظر را محاسبه کنید.
\end{exercisebox}

\begin{exercisebox}{3}{توپولوژی عامل‌ها}\label{exr:topo}
یک اپراتور ۲۰ \acro{MME}، ۱۰ بخش جلویی \acro{HSS}، ۳۰ \lr{P-GW}، ۸ \acro{PCRF} و ۴ گرهٔ \acro{OCS} دارد. تعداد اتصال‌های
همتای \lr{Diameter} را برای یک شبکهٔ کامل واسط‌های \lr{S6a}، \lr{Gx} و \lr{Gy} و برای طرحی با یک جفت افزونهٔ \acro{DRA}
(هر گره به هر دو \acro{DRA} وصل است) مقایسه کنید. پیامدهای قابلیت اطمینان را بررسی کنید.
\end{exercisebox}

\begin{exercisebox}{3}{سهمیه با بودجهٔ خطر}\label{exr:risk}
به کاربری با موجودی $B$ سهمیه‌های $Q = \min(Q_{\max}, \alpha B)$ اعطا می‌شود. نشان دهید که تعداد مورد انتظار
\lr{CCR-Update}ها برای مصرف کل موجودی با نرخ ثابت، وقتی $Q_{\max}$ هرگز نرسد و سهمیه‌ها در $Q_{\min}$ متوقف شوند، حدود
$\log(B/Q_{\min})/\log(1/(1-\alpha))$ است و آن را برای $\alpha = 0.2$، $B = 1$ گیگابایت و $Q_{\min} = 1$ مگابایت
محاسبه کنید.
\end{exercisebox}

\subsection*{پاسخ‌ها}
\addcontentsline{toc}{subsection}{پاسخ‌ها}

\begin{solutionbox}{\ref{exr:srv}}
(۱) \lr{Cancel-Location} (\lr{CLR}) از \acro{HSS} به \acro{MME} وقتی مشترک در \acro{MME} دیگری ثبت‌نام می‌کند یا
اشتراکش لغو می‌شود؛ (۲) \lr{Re-Auth} (\lr{RAR}) از \acro{PCRF} به \lr{P-GW} وقتی یک تماس \acro{VoLTE} به حامل
اختصاصی نیاز دارد یا به حد مصرف می‌رسد؛ همچنین \lr{Insert-Subscriber-Data} از \acro{HSS} وقتی نمایه تغییر می‌کند. در
\lr{RADIUS} کارخواه‌ها درخواست می‌فرستند و کارسازها فقط پاسخ می‌دهند؛ راه استانداردی برای آغاز تراکنش از سوی کارساز
وجود ندارد (فقط افزونه‌های بعدی مانند \lr{Dynamic Authorization})، و \acro{NAT} یا دیواره‌های آتش جلوی کارخواه‌های
\acro{UDP} آن را غیرقابل اطمینان می‌کنند.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:ids}}
شناسهٔ گام‌به‌گام به گره اجازه می‌دهد پاسخ را با درخواستی که روی یک اتصال معین فرستاده تطبیق دهد؛ هر رله آن را با مقدار
خود جایگزین و هنگام بازگرداندن پاسخ مقدار اصلی را بازیابی می‌کند. شناسهٔ سرتاسری را مبدأ تعیین می‌کند و هرگز تغییر
نمی‌کند؛ همراه با \lr{Origin-Host} به کارساز امکان می‌دهد تا مدتی تکراری‌ها را تشخیص دهد (مثلاً درخواستی که پس از
جایگزینی با پرچم \lr{T} دوباره فرستاده شده است).
\end{solutionbox}

\begin{solutionbox}{\ref{exr:avp}}
(الف) $8+4 = 12$ بایت. (ب) $12 + 8 = 20$ بایت (از قبل هم‌تراز). (ج) $8$ (یا ۱۲ اگر ویژهٔ فروشنده باشد) $+12+20 = 40$
بایت (۴۴ با \lr{Vendor-ID}).
\end{solutionbox}

\begin{solutionbox}{\ref{exr:watchdog}}
هر اتصال بیکار در هر $T_\mathrm{w}$ از سوی هر طرفی که آغاز کند یک \lr{DWR} و یک \lr{DWA} تولید می‌کند؛ با شمارش یک
\lr{DWR} و یک \lr{DWA} برای هر اتصال در هر $T_\mathrm{w}$: $400\times2/30 = 26.7$ پیام بر ثانیه برای ۳۰ ثانیه و ۱۳۳ پیام
بر ثانیه برای ۶ ثانیه -- در هر دو حالت ناچیز. زمان تشخیص $\approx 2T_\mathrm{w}$: ۶۰ ثانیه در برابر ۱۲ ثانیه. زمان‌سنج
کوتاه‌تر روی پیوندهای حیاتی به‌روشنی می‌ارزد؛ خطر آن جایگزینی کاذب روی پیوندهای پرازدحام است.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:topo}}
شبکهٔ کامل: \lr{S6a} $20\times10 = 200$، \lr{Gx} $30\times8 = 240$، \lr{Gy} $30\times4 = 120$: ۵۶۰ اتصال که هر یک باید
پیکربندی و پایش شود. با دو \acro{DRA}: $(20+10+30+8+4)\times2 = 144$ اتصال (به‌علاوهٔ یک اتصال میان دو \acro{DRA}).
\acro{DRA}ها مسیریابی، متعادل‌سازی بار و پیوند نشست را متمرکز می‌کنند و افزودن گره را ساده می‌کنند، اما به نقاط
حیاتی تبدیل می‌شوند: باید به‌صورت جفت‌های افزونهٔ جغرافیایی با ظرفیت کافی برای حمل همهٔ ترافیک در صورت خرابی یکی
مستقر شوند.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:risk}}
هر اعطا کسر $\alpha$ از موجودی باقی‌مانده را مصرف می‌کند، پس پس از $k$ اعطا موجودی $B(1-\alpha)^k$ است؛ سهمیه‌ها وقتی
$\alpha B(1-\alpha)^k = Q_{\min}$ شود به زیر $Q_{\min}$ می‌رسند، یعنی
$k \approx \log(\alpha B/Q_{\min})/\log(1/(1-\alpha))$ که اساساً همان عبارت خواسته‌شده است. برای $\alpha=0.2$،
$B = 1000$ مگابایت و $Q_{\min} = 1$ مگابایت: $k \approx \ln(200)/\ln(1.25) = 5.30/0.223 = 23.7$ اعطا، که پس از آن چند
اعطای کمینه موجودی را تمام می‌کنند. سهمیه‌های هندسی هم سیگنالینگ (لگاریتمی در $B$) و هم خطر (حداکثر $\alpha$ از
موجودی) را کوچک نگه می‌دارند.
\end{solutionbox}
