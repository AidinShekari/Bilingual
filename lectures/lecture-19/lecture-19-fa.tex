\section{اهداف یادگیری}
پس از مطالعهٔ این درس باید بتوانید:
\begin{objectives}
  \item معماری سامانهٔ نسل پنجم را به شکل نقاط مرجع \textbf{رسم کنید} و هر \acro{NF} را به پیشینهٔ آن در \acro{EPC}
        \textbf{نگاشت کنید}.
  \item هویت‌های نسل پنجم (\lr{SUPI, SUCI, 5G-GUTI, GPSI, PEI}) و نحوهٔ حفاظت \acro{SUCI} از حریم مشترک را \textbf{توضیح دهید}.
  \item نشست‌های \acro{PDU}، انواع آن‌ها و حالت‌های پیوستگی نشست و خدمت (\acro{SSC}) را \textbf{توصیف کنید}.
  \item چیدمان‌های \acro{UPF} با طبقه‌بند فراسو و لنگرهای محلی برای رایانش لبه را \textbf{طراحی کنید} و دستاورد تأخیر و بک‌هال را
        \textbf{تخمین بزنید}.
  \item حالت‌های \lr{RM/CM}، نواحی ثبت‌نام و همکاری با \acro{EPC} را \textbf{توضیح دهید}.
\end{objectives}

\section{دانش پیش‌نیاز}
به معماری \acro{EPC} (درس‌های ۹، ۱۲، ۱۳)، نمای کلی نسل پنجم (درس ۱۴) و سازوکارهای سرویس‌محور درس ۱۸ (\acro{NRF}، \acro{SBI})
نیاز دارید. این درس توصیف می‌کند هستهٔ نسل پنجم از \emph{چه} تشکیل شده است؛ درس ۲۰ رویه‌هایی را که از آن استفاده می‌کنند، درس ۲۱
برش‌بندی و درس ۲۲ کیفیت خدمات را توصیف می‌کنند.

\paragraph{واژه‌های کلیدی} واژگان این درس همراه با معادل انگلیسی آن‌ها:
\begin{keyterms}
شناسهٔ دائمی / پنهان‌شدهٔ مشترک & \lr{SUPI / SUCI} & هویت دائمی / شکل رمزشدهٔ آن که روی هوا فرستاده می‌شود. \\
شناسهٔ موقت سراسری & \lr{5G-GUTI} & هویت موقتی که \acro{AMF} تخصیص می‌دهد. \\
نشست \acro{PDU} & \lr{PDU session} & پیوند میان \acro{UE} و یک شبکهٔ داده از طریق \lr{5GC}. \\
نام شبکهٔ داده & \lr{DNN} & نام یک شبکهٔ داده (معادل \acro{APN} در نسل پنجم). \\
حالت پیوستگی نشست و خدمت & \lr{SSC mode} & قواعد نگه‌داشتن یا تغییر لنگر نشست در طول تحرک. \\
لنگر نشست \acro{PDU} & \lr{PSA} & \acro{UPF}ای که \lr{N6} را به سوی شبکهٔ داده خاتمه می‌دهد. \\
طبقه‌بند فراسو / نقطهٔ انشعاب & \lr{UL CL / BP} & کارکردهای \acro{UPF} که ترافیک را میان لنگرها تقسیم می‌کنند. \\
مدیریت ثبت / اتصال & \lr{RM / CM} & حالت‌های مدیریت ثبت‌نام و اتصال. \\
واسط \lr{N26} & \lr{N26} & واسط \acro{AMF}--\acro{MME} برای همکاری \acro{EPC} و \lr{5GC}. \\
تابع همکاری غیر \acro{3GPP} & \lr{N3IWF} & دروازه برای دسترسی غیرمطمئن غیر\acro{3GPP} (مثلاً \lr{Wi-Fi}). \\
\end{keyterms}

\section{مقدمه و انگیزه}
هستهٔ نسل پنجم (\lr{5GC}) نخستین هستهٔ موبایل است که از آغاز برای استقرار ابری طراحی شد. مفاهیم آزموده‌شدهٔ \acro{EPC} را نگه
می‌دارد -- یک لنگر تحرک در صفحهٔ کنترل، یک لنگر نشست در صفحهٔ کاربر، یک پایگاه دادهٔ مشترک، کنترل سیاست -- اما آن‌ها را از نو
سازمان می‌دهد: دسترسی و تحرک از مدیریت نشست جدا می‌شود، صفحهٔ کاربر به‌طور کامل از طریق \acro{PFCP} کنترل می‌شود، دادهٔ
اشتراک میان بخش‌های جلویی و یک مخزن تقسیم می‌شود و هر کارکرد صفحهٔ کنترل یک تولیدکنندهٔ خدمت است (درس ۱۸) \cite{ts23501}.
ساختار جدید از قابلیت‌هایی پشتیبانی می‌کند که \acro{EPC} به‌طور بومی نمی‌توانست ارائه دهد: ثبت‌نام مستقل از دسترسی
(\acro{3GPP} و غیر\acro{3GPP})، چند لنگر صفحهٔ کاربر برای هر نشست جهت رایانش لبه، نشست‌های اترنت و بی‌ساختار، برش‌بندی شبکه و
مدل کیفیت خدمات مبتنی بر جریان.

\subsection{هویت‌ها}
\begin{itemize}
  \item \textbf{\acro{SUPI}}: شناسهٔ دائمی اشتراک، معمولاً یک \acro{IMSI} (یا یک شناسهٔ دسترسی شبکه).
  \item \textbf{\acro{SUCI}}: \acro{SUPI} که با کلید عمومی شبکهٔ خانگی پنهان شده است؛ فقط بخش مسیریابی آشکار است
        (\cref{fig:ids}). تابع آشکارسازی \acro{UDM} \acro{SUPI} را بازیابی می‌کند \cite{ts33501}.
  \item \textbf{\lr{5G-GUTI}}: هویت موقت = \acro{GUAMI} (\acro{PLMN}، ناحیهٔ \acro{AMF}، مجموعهٔ \acro{AMF}، اشاره‌گر \acro{AMF}) +
        \lr{5G-TMSI}؛ \lr{5G-S-TMSI} ۴۸ بیتی در فراخوانی و \acro{RRC} به کار می‌رود.
  \item \textbf{\acro{PEI}}: شناسهٔ دائمی تجهیزات (\acro{IMEI})؛ \textbf{\acro{GPSI}}: شناسهٔ عمومی کلی (\acro{MSISDN} یا شناسهٔ
        بیرونی) که به سوی کاربردهای بیرونی به کار می‌رود.
\end{itemize}
برخلاف \acro{LTE}، یک \acro{UE} نسل پنجم هرگز هویت دائمی خود را آشکارا روی واسط رادیویی نمی‌فرستد: حتی نخستین ثبت‌نام از یک
\acro{SUCI} استفاده می‌کند، که ضعف «شکارچی \acro{IMSI}» نسل‌های پیشین را می‌بندد.

\subsection{نشست‌های \lr{PDU}}
یک نشست \acro{PDU} \acro{UE} را به یک شبکهٔ داده که با یک \acro{DNN} و یک برش (\lr{S-NSSAI}) مشخص می‌شود وصل می‌کند. انواع
نشست عبارت‌اند از \acro{IPv4}، \acro{IPv6}، \lr{IPv4v6}، \emph{اترنت} (\acro{UPF} قاب‌های اترنت را سوئیچ می‌کند، مثلاً برای
شبکه‌های محلی صنعتی) و \emph{بی‌ساختار} (دادهٔ مبهم). درون یک نشست، ترافیک در جریان‌های کیفیت خدمات حمل می‌شود (درس ۲۲)؛ یک
\acro{UE} می‌تواند هم‌زمان چند نشست به \acro{DNN}ها یا برش‌های مختلف داشته باشد.

\subsection{پیوستگی نشست و خدمت}
\emph{حالت \acro{SSC}} یک نشست تعیین می‌کند هنگام حرکت \acro{UE} بر سر لنگر آن چه می‌آید \cite{ts23501}:
\begin{itemize}
  \item \textbf{حالت \acro{SSC} ۱}: \acro{PSA} (و نشانی \acro{IP}) در طول عمر نشست حفظ می‌شود -- مانند \lr{P-GW} در \acro{EPC}.
  \item \textbf{حالت \acro{SSC} ۲}: شبکه ممکن است نشست را آزاد کند و از \acro{UE} بخواهد نشست جدیدی با لنگری نزدیک‌تر برقرار
        کند («پیش از ساختن، بشکن»).
  \item \textbf{حالت \acro{SSC} ۳}: شبکه پیش از آزادکردن نشست قدیمی، نشست جدیدی با لنگر جدید برقرار می‌کند («پیش از شکستن، بساز»)؛
        کاربرد می‌تواند به‌آرامی مهاجرت کند.
\end{itemize}

\section{توضیحات فنی تفصیلی}

\subsection{\lr{AMF}}
\acro{AMF} \lr{N1} (\acro{NAS}) و \lr{N2} (\acro{NGAP}) را خاتمه می‌دهد \cite{ts24501,ts38413}. به ثبت‌نام، دسترس‌پذیری و
فراخوانی، مدیریت اتصال، تحرک و احراز هویت دسترسی با \acro{AUSF} رسیدگی می‌کند. برخلاف \acro{MME} سیگنالینگ نشست را پردازش
نمی‌کند: پیام‌های \acro{NAS} نشست به‌طور شفاف به \acro{SMF} رله می‌شوند. یک \acro{AMF} در دسترسی \acro{3GPP} و غیر\acro{3GPP}
به یک \acro{UE} خدمت می‌دهد، پس ثبت‌نام مستقل از دسترسی است. \acro{AMF}ها در \emph{مجموعه‌های \acro{AMF}} سازمان می‌یابند: هر
\acro{AMF} از یک مجموعه می‌تواند زمینهٔ یک \acro{UE} را از ذخیره‌سازی مشترک تحویل بگیرد.

\subsection{\lr{SMF} و \lr{UPF}}
\acro{SMF} نشست‌های \acro{PDU} را مدیریت می‌کند: \acro{UPF}ها را انتخاب، نشانی‌های \acro{IP} را تخصیص و قواعد \acro{PFCP}
(\lr{PDR, FAR, QER, URR, BAR} -- درس ۱۳) را از طریق \lr{N4} نصب می‌کند، و با \acro{PCF} برای سیاست و با \acro{CHF} برای
صورت‌حساب تعامل دارد. \acro{UPF} تنها کارکرد صفحهٔ کاربر است: یک \emph{\acro{UPF} میانی} (\lr{I-UPF}) ممکن است \acro{RAN} را وصل
کند، یک \emph{طبقه‌بند فراسو} (\lr{UL CL}) یا \emph{نقطهٔ انشعاب} (برای چندخانگی \acro{IPv6}) ممکن است بخشی از ترافیک را منحرف
کند، و یک یا چند \emph{لنگر نشست \acro{PDU}} (\acro{PSA}) \lr{N6} را خاتمه می‌دهند (\cref{fig:upf}).

\subsection{رایانش لبه}
برای خدمت به کاربردهای کم‌تأخیر، \acro{SMF} می‌تواند یک \lr{UL CL} و یک \acro{PSA} محلی در یک سایت لبه قرار دهد تا ترافیک
نشانی‌های کاربرد لبه به‌صورت محلی مسیریابی شود و بقیه به لنگر مرکزی برود. توابع کاربرد می‌توانند از طریق \acro{NEF} یا \acro{PCF}
بر این مسیریابی اثر بگذارند و \lr{TS 23.548} کشف کارسازهای کاربرد لبه را می‌افزاید \cite{ts23548}. نتیجه یک نشست \acro{PDU}
واحد با خروج محلی و مرکزی هم‌زمان است -- چیزی که در \acro{EPC} بدون ترتیبات ویژه ناممکن بود.

\subsection{حالت‌ها و نواحی ثبت‌نام}
حالت‌های \acro{RM}: \lr{RM-DEREGISTERED} و \lr{RM-REGISTERED}؛ حالت‌های \acro{CM}: \lr{CM-IDLE} و \lr{CM-CONNECTED}. یک \acro{UE}
در \lr{CM-CONNECTED} ممکن است افزون بر آن در \lr{RRC\_INACTIVE} باشد: اتصال \lr{N2} و زمینه برقرار می‌مانند در حالی که \acro{RAN}
\acro{UE} را درون یک ناحیهٔ اعلان \acro{RAN} فراخوانی می‌کند، که سیگنالینگ ارسال‌های کوچک و پرتکرار را صرفه‌جویی می‌کند.
\acro{AMF} یک \emph{ناحیهٔ ثبت‌نام} (یک فهرست \acro{TAI}) تخصیص می‌دهد و به‌روزرسانی‌های ثبت‌نام تحرکی و دوره‌ای را انجام می‌دهد،
همان‌گونه که \acro{MME} با \acro{TAU}ها می‌کرد.

\subsection{همکاری با \lr{EPC}}
در طول همزیستی طولانی نسل‌های ۴ و ۵، یک \acro{UE} میان \acro{LTE} (\acro{EPC}) و \acro{NR} (\lr{5GC}) جابه‌جا می‌شود. با
\emph{\lr{N26}} \acro{MME} و \acro{AMF} زمینه‌ها را مستقیم مبادله می‌کنند و نشست‌های \acro{IP} از گذار جان سالم به در می‌برند؛ این
به گره‌های ترکیبی نیاز دارد: \lr{SMF+PGW-C}، \lr{UPF+PGW-U}، \lr{HSS+UDM} و \lr{PCF+PCRF} به هر دو سامانه خدمت می‌دهند
(\cref{fig:iw}). بدون \lr{N26} \acro{UE} دوباره ثبت‌نام می‌کند و پیوستگی نشست به لنگر ترکیبی وابسته است. صدا با بازگشت \acro{EPS}
پشتیبانی می‌شود (یک تماس صوتی نسل پنجم برای \acro{VoLTE} به \acro{LTE} تحویل می‌شود) هرجا \lr{VoNR} در دسترس نباشد.

\subsection{دسترسی غیر \lr{3GPP}}
دسترسی غیرمطمئن غیر\acro{3GPP} (مثلاً \lr{Wi-Fi} عمومی) از طریق یک \acro{N3IWF} وصل می‌شود که یک تونل \acro{IPsec} از \acro{UE} را
خاتمه می‌دهد و \lr{N2/N3} را به سوی هسته عرضه می‌کند؛ دسترسی مطمئن غیر\acro{3GPP} از یک \acro{TNGF} استفاده می‌کند. در هر دو
حالت \acro{UE} نزد همان \acro{AMF} ثبت‌نام می‌کند و از همان رویه‌های \acro{NAS} و نشست \acro{PDU} استفاده می‌کند -- طراحی مستقل از
دسترسی \lr{5GC}.

\section{مدل‌های ریاضی}

\subsection{اندازهٔ \lr{SUCI}}\label{sec:suci}
\paragraph{مدل} با \acro{ECIES}، خروجی طرح شامل کلید عمومی موقت ($k$ بایت)، \acro{MSIN} رمزشده (با همان طول $m$ متن آشکار، با
کدگذاری \acro{BCD}) و یک برچسب \acro{MAC} ($t$ بایت) است:
\begin{equation}\label{eq:suci}
  L_{\mathrm{out}} = k + m + t .
\end{equation}
\paragraph{مثال} یک \acro{MSIN} ده‌رقمی ($m = 5$ بایت): نمایهٔ \lr{A} (\lr{X25519}، $k = 32$، $t = 8$) ۴۵ بایت می‌دهد؛ نمایهٔ
\lr{B} (نقطهٔ فشردهٔ \lr{P-256}، $k = 33$) ۴۶ بایت \cite{ts33501}. سربار در مقایسه با حریم به‌دست‌آمده اندک است و محاسبه یک بار
برای هر تلاش ثبت‌نامی که به هویت دائمی نیاز دارد انجام می‌شود.

\subsection{تأخیر و جایابی \lr{UPF}}\label{sec:rtt}
\paragraph{مدل} تأخیر رفت‌وبرگشت میان \acro{UE} و یک کارساز کاربرد پشت یک \acro{PSA} در فاصلهٔ فیبری $D$:
\begin{equation}\label{eq:rtt}
  \mathrm{RTT} \approx \mathrm{RTT}_{\mathrm{radio}} + 2\,\frac{D}{v} + T_{\mathrm{proc}},
\end{equation}
با $v \approx 2\times10^8$ متر بر ثانیه در فیبر (حدود ۵ میکروثانیه به ازای هر کیلومتر).
\paragraph{مثال} $D = 500$ کیلومتر (مرکز دادهٔ مرکزی) ۵ میلی‌ثانیه می‌افزاید؛ $D = 20$ کیلومتر (سایت لبه) ۰٫۲ میلی‌ثانیه. با
\lr{RTT} رادیویی ۴ میلی‌ثانیه و پردازش ۱ میلی‌ثانیه، \lr{RTT} از ۱۰ به ۵٫۲ میلی‌ثانیه کاهش می‌یابد -- لبه آن را نصف می‌کند.
\paragraph{تفسیر} انتشار تنها یک جمله است؛ برای اهداف \acro{URLLC} در حد چند میلی‌ثانیه، همهٔ اجزا -- عددشناسی رادیویی، زمان‌بندی،
انتقال و پردازش -- باید با هم مهندسی شوند.

\subsection{تخلیهٔ بک‌هال}\label{sec:offload}
\paragraph{مدل} اگر کسر $\alpha$ از ترافیک $T$ یک سایت به‌صورت محلی خدمت شود (\lr{UL CL} به یک \acro{PSA} محلی و نهان‌گاه یا
کاربرد لبه)، ترافیک حمل‌شده به سوی هستهٔ مرکزی
\begin{equation}\label{eq:offload}
  T_{\mathrm{bh}} = (1-\alpha)\,T
\end{equation}
است.
\paragraph{مثال} با خدمت ۶۰٪ ویدئو از یک نهان‌گاه لبه، سایتی با ۱۰ گیگابیت بر ثانیه ترافیک ساعت اوج به‌جای ۱۰ به ۴ گیگابیت بر
ثانیه بک‌هال به سوی هستهٔ مرکزی نیاز دارد.

\subsection{وقفه در حالت‌های \lr{SSC}}\label{sec:ssc}
\paragraph{مدل} وقتی لنگر تغییر می‌کند، وقفهٔ خدمت تقریباً عبارت است از
\begin{equation}\label{eq:ssc}
  T_{\mathrm{int}} \approx
  \begin{cases}
    0 & \text{\rl{\lr{SSC} ۱ (بدون تغییر لنگر) و \lr{SSC} ۳ (پیش از شکستن، بساز)}},\\
    T_{\mathrm{rel}} + T_{\mathrm{est}} + T_{\mathrm{app}} & \text{\rl{\lr{SSC} ۲ (پیش از ساختن، بشکن)}},
  \end{cases}
\end{equation}
که در آن $T_{\mathrm{rel}}$ و $T_{\mathrm{est}}$ زمان‌های آزادسازی و برقراری و $T_{\mathrm{app}}$ زمان اتصال دوبارهٔ کاربرد با نشانی جدید
است.
\paragraph{مثال} $T_{\mathrm{rel}} = 50$ میلی‌ثانیه، $T_{\mathrm{est}} = 150$ میلی‌ثانیه و $T_{\mathrm{app}} = 300$ میلی‌ثانیه: نیم ثانیه
بدون خدمت در \acro{SSC} ۲ -- برای یک بارگیری پس‌زمینه پذیرفتنی، اما نه برای یک تماس ویدئویی که باید از \acro{SSC} ۱ یا ۳ استفاده
کند.

\section{معماری شبکه و نمودارها}
یک \lr{5GC} ملی معمولاً چند سایت مرکزی دارد که مجموعه‌های \acro{AMF}، \acro{SMF}ها، \lr{UDM/UDR}، \acro{PCF}، \acro{NRF} و

\acro{SMF} برای هر نشست تعیین می‌کند کدام \acro{UPF}ها زنجیر شوند: برای نشست اینترنت یک \acro{PSA} مرکزی، برای نشست اترنت صنعتی یک
\acro{PSA} محلی در محل کارخانه، برای نشستی با خدمات لبه یک \lr{UL CL} با هر دو (\cref{fig:upf}). برش‌ها (درس ۲۱) ممکن است از
\acro{SMF}ها و \acro{UPF}های اختصاصی استفاده کنند.

\section{رویه‌های گام‌به‌گام}

\begin{procedurebox}{محاسبه و آشکارسازی \lr{SUCI}}\label{proc:suci}
\begin{steps}
  \item \acro{USIM} (یا \acro{ME}) کلید عمومی شبکهٔ خانگی و شناسهٔ آن و طرح حفاظتی مورد استفاده را نگه می‌دارد.
  \item برای هر ثبت‌نامی که به هویت دائمی نیاز دارد، \acro{UE} یک جفت کلید موقت می‌سازد، با کلید عمومی شبکهٔ خانگی یک راز مشترک
        استخراج و \acro{MSIN} را رمزنگاری می‌کند؛ برچسب \acro{MAC} را محاسبه می‌کند.
  \item \acro{UE} \acro{SUCI} (بخش مسیریابی آشکار به‌علاوهٔ خروجی طرح) را در \lr{Registration Request} می‌فرستد.
  \item \acro{AMF} آن را از طریق \acro{AUSF} به \acro{UDM} خانگی هدایت می‌کند که \acro{SIDF} آن با کلید خصوصی \acro{SUPI} را بازیابی
        می‌کند.
\end{steps}
\end{procedurebox}

\begin{procedurebox}{افزودن لنگر محلی برای ترافیک لبه}\label{proc:ulcl}
\begin{steps}
  \item یک کاربرد لبه مستقر می‌شود و بازهٔ نشانی آن در \acro{SMF} پیکربندی می‌شود (یا از طریق \lr{NEF/PCF} گزارش می‌شود).
  \item وقتی \acro{UE} در ناحیهٔ سایت لبه است، \acro{SMF} یک \acro{UPF} آنجا را به‌عنوان \lr{UL CL} و یک \acro{PSA} محلی برمی‌گزیند و
        قواعد \acro{PFCP} را نصب می‌کند: ترافیک منطبق با نشانی‌های لبه به \acro{PSA} محلی و بقیه از طریق \lr{N9} به \acro{PSA} مرکزی
        هدایت می‌شود.
  \item \acro{SMF} تونل \lr{N3} \acro{RAN} را (از طریق \acro{AMF} و \lr{N2}) به سوی \lr{UL CL} به‌روز می‌کند.
  \item وقتی \acro{UE} ناحیه را ترک می‌کند، \acro{SMF} \lr{UL CL} و \acro{PSA} محلی را حذف می‌کند؛ نشست از طریق لنگر مرکزی ادامه
        می‌یابد.
\end{steps}
\end{procedurebox}

\begin{examplebox}{نگاشت \lr{EPC} به \lr{5GC}}\label{ex:map}
\textbf{مسئله.} کدام کارکردهای \lr{5GC} این‌ها را بر عهده می‌گیرند: (الف) مدیریت فهرست ناحیهٔ ردیابی \acro{MME}، (ب) برقراری

\textbf{حل.} (الف) \acro{AMF} (نواحی ثبت‌نام)؛ (ب) \acro{SMF} (مدیریت نشست \acro{PDU}، \lr{N4} به \acro{UPF})؛ (ج) \acro{SMF}

\begin{examplebox}{بودجهٔ تأخیر لبه}\label{ex:edge}
\textbf{مسئله.} یک کاربرد واقعیت افزوده به \lr{RTT} کمتر از ۱۰ میلی‌ثانیه نیاز دارد. \lr{RTT} رادیویی ۴ و پردازش ۱ میلی‌ثانیه است.
کارساز کاربرد تا چه فاصلهٔ فیبری می‌تواند قرار گیرد؟

\textbf{حل.} $4 + 1 + 2D/v \le 10 \Rightarrow D \le 5\ \text{ms}\times v/2 = 500$ کیلومتر مسیر فیبر. مسیرهای فیبر معمولاً ۱٫۵ برابر
فاصلهٔ خط مستقیم‌اند، پس حدود ۳۳۰ کیلومتر خط مستقیم -- یک لبهٔ منطقه‌ای، نه لزوماً در محل سلول.
\end{examplebox}

\begin{examplebox}{انتخاب حالت \lr{SSC}}\label{ex:ssc}
\textbf{مسئله.} حالت‌های \acro{SSC} را برگزینید برای (الف) صدای \acro{IMS}، (ب) خودرویی که به‌روزرسانی نقشه بارگیری می‌کند، (ج) یک
بازی ابری که می‌تواند میان کارسازهای لبه مهاجرت کند.

\textbf{حل.} (الف) \acro{SSC} ۱ -- ثبت‌نام \acro{IMS} به نشانی \acro{IP} وابسته است؛ (ب) \acro{SSC} ۲ -- یک وقفهٔ کوتاه پذیرفتنی است و
لنگر محلی بک‌هال را کاهش می‌دهد؛ (ج) \acro{SSC} ۳ -- کاربرد پیش از آزادشدن لنگر قدیمی اتصالی از طریق لنگر جدید باز می‌کند.
\end{examplebox}

\section{نکته‌های کلیدی}
\begin{takeawaybox}
\begin{itemize}
  \item \lr{5GC} \acro{AMF} (دسترسی، تحرک) را از \acro{SMF} (نشست‌ها) جدا می‌کند و \acro{UPF}ها را فقط از طریق \acro{PFCP} کنترل
        می‌کند.
  \item \acro{SUPI} هرگز آشکار فرستاده نمی‌شود؛ \acro{SUCI} = اطلاعات مسیریابی + خروجی \acro{ECIES} ($k + m + t$ بایت).
  \item نشست‌های \acro{PDU} (\acro{IP}، اترنت، بی‌ساختار) می‌توانند چند لنگر داشته باشند؛ \lr{UL CL} و \acro{BP} خروج محلی را ممکن
        می‌کنند.
  \item حالت‌های \acro{SSC} ۱/۲/۳ پایداری لنگر را با مسیریابی بهینه معاوضه می‌کنند؛ \acro{SSC} ۲ خدمت را قطع می‌کند، \acro{SSC} ۳ نه.
  \item \acro{UPF}های لبه \lr{RTT} را به اندازهٔ $2D/v$ و بک‌هال را با ضریب $(1-\alpha)$ کاهش می‌دهند.
  \item همکاری با \acro{EPC} به \lr{N26} و گره‌های ترکیبی (\lr{SMF+PGW-C, UPF+PGW-U, HSS+UDM, PCF+PCRF}) تکیه دارد.
\end{itemize}
\end{takeawaybox}

\section{تمرین‌ها}

\begin{exercisebox}{1}{چه کسی انجام می‌دهد؟}\label{exr:who}
کدام \acro{NF} این‌ها را انجام می‌دهد: (الف) فراخوانی؛ (ب) تخصیص نشانی \acro{IP}؛ (ج) انتخاب برش هنگام ثبت‌نام؛ (د) آشکارسازی
\acro{SUCI}؛ (هـ) عرضهٔ موقعیت \acro{UE} به یک کاربرد بیرونی مجاز؟
\end{exercisebox}

\begin{exercisebox}{1}{نشست‌های اترنت}\label{exr:eth}
کاربردی برای نشست‌های \acro{PDU} اترنت بیاورید و توضیح دهید چرا نشست‌های \acro{IP} برای آن ناخوشایندند.
\end{exercisebox}

\begin{exercisebox}{2}{اندازهٔ \lr{SUCI}}\label{exr:suci}
طول خروجی طرح را برای یک \acro{MSIN} نُه‌رقمی در نمایه‌های \lr{A} و \lr{B} محاسبه کنید. پنهان‌سازی در مقایسه با طرح تهی که
\acro{MSIN} را آشکار می‌فرستد چند بایت می‌افزاید؟
\end{exercisebox}

\begin{exercisebox}{2}{تخلیهٔ لبه}\label{exr:offload}
یک شهر ۵۰ سایت لبه دارد که هر کدام ۸ گیگابیت بر ثانیه ترافیک ساعت اوج دارند. چه کسری باید در لبه خدمت شود تا کل بک‌هال به هستهٔ
مرکزی زیر ۲۰۰ گیگابیت بر ثانیه بماند؟
\end{exercisebox}

\begin{exercisebox}{3}{طراحی تأخیر}\label{exr:latency}
یک کارخانه میان کنترل‌گر یک ربات و \acro{UE} به \lr{RTT} کمتر از ۵ میلی‌ثانیه نیاز دارد. \lr{RTT} رادیویی با شکاف‌های کوچک ۲ و پردازش
۰٫۵ میلی‌ثانیه است. \acro{PSA} و کنترل‌گر کجا باید باشند و کدام نوع نشست و حالت \acro{SSC} را برمی‌گزینید؟ اگر \lr{RTT} رادیویی ۴
میلی‌ثانیه بود چه؟
\end{exercisebox}

\begin{exercisebox}{3}{همکاری بدون \lr{N26}}\label{exr:n26}
توضیح دهید وقتی \acro{UE} از پوشش \acro{NR} به \acro{LTE} می‌رود بر سر یک نشست \acro{IP} در جریان چه می‌آید: (الف) با \lr{N26} و
لنگرهای ترکیبی، (ب) بدون \lr{N26} اما با لنگرهای ترکیبی، (ج) بدون لنگرهای ترکیبی. کدام حالت نشانی \acro{IP} را حفظ می‌کند؟
\end{exercisebox}

\subsection*{پاسخ‌ها}
\addcontentsline{toc}{subsection}{پاسخ‌ها}

\begin{solutionbox}{\ref{exr:who}}
(الف) \acro{AMF} (و \acro{RAN} برای \lr{RRC\_INACTIVE})؛ (ب) \acro{SMF}؛ (ج) \acro{AMF} همراه با \acro{NSSF}؛ (د) \acro{UDM}
(\acro{SIDF})؛ (هـ) \acro{NEF}، که اطلاعات را از \lr{AMF/UDM} به دست می‌آورد.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:eth}}
شبکه‌های محلی صنعتی (مثلاً شبکه‌سازی حساس به زمان میان کنترل‌گرها و ماشین‌ها) و \acro{VPN}های لایهٔ ۲ سازمانی. دستگاه‌ها از
پروتکل‌های اترنت و نشانی‌دهی \acro{MAC}، گاه غیر\acro{IP}، استفاده می‌کنند؛ حمل آن‌ها روی نشست‌های \acro{IP} به تونل‌زنی یا دروازه
نیاز دارد که تأخیر و پیچیدگی می‌افزاید، در حالی که یک نشست اترنت به \acro{UPF} اجازه می‌دهد قاب‌ها را مستقیم سوئیچ کند.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:suci}}
۹ رقم در \acro{BCD} پنج بایت اشغال می‌کنند (نیم‌بایت آخر لایه‌گذاری می‌شود). نمایهٔ \lr{A}: $32 + 5 + 8 = 45$ بایت؛ نمایهٔ \lr{B}:
$33 + 5 + 8 = 46$ بایت. طرح تهی \acro{MSIN} پنج‌بایتی را حمل می‌کند، پس پنهان‌سازی ۴۰ یا ۴۱ بایت می‌افزاید.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:offload}}
کل ترافیک $50\times8 = 400$ گیگابیت بر ثانیه؛ $(1-\alpha)\times400 \le 200 \Rightarrow \alpha \ge 0.5$: نیمی از ترافیک باید در لبه
خدمت شود.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:latency}}
$2 + 0.5 + 2D/v \le 5 \Rightarrow D \le 250$ کیلومتر: در اصل یک سایت منطقه‌ای، اما گام‌های انتقال، صف و لرزش بودجه را مصرف می‌کنند،
پس \acro{PSA} و کنترل‌گر در محل کارخانه قرار می‌گیرند (\acro{UPF} محلی). یک نشست اترنت (اگر پروتکل کنترل لایهٔ ۲ است) با
\acro{SSC} ۱ (لنگر محلی ثابت). با \lr{RTT} رادیویی ۴ میلی‌ثانیه فقط ۱ میلی‌ثانیه باقی می‌ماند: $D \le 100$ کیلومتر، و در عمل همان
طراحی درون‌محلی با حاشیهٔ اندک.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:n26}}
(الف) \acro{AMF} زمینه را از طریق \lr{N26} به \acro{MME} می‌دهد؛ نشست از طریق همان \lr{SMF+PGW-C/UPF+PGW-U} به \acro{EPC} منتقل
