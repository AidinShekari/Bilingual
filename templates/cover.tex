% =====================================================================
%  Lecture cover page.  Uses \ACN... metadata and \ACNstr... strings.
%  The cover is drawn in a left-to-right TikZ overlay; text direction of
%  each node is chosen by \ACNnodetext so that Persian titles stay RTL.
% =====================================================================
\makeatletter
\newcommand{\ACNgenmark}[1]{%
  \ifnum#1=\ACNgen\relax acnteal\else white\fi}
\newcommand{\ACNcover}{%
  \thispagestyle{empty}%
  \begin{tikzpicture}[remember picture,overlay]
    % ---- top band with hexagonal cell motif
    \fill[acnnavy] (current page.north west) rectangle
      ([yshift=-10.2cm]current page.north east);
    \begin{scope}
      \clip (current page.north west) rectangle ([yshift=-10.2cm]current page.north east);
      \foreach \col in {0,...,14}{\foreach \row in {0,...,7}{%
        \pgfmathsetmacro\hx{\col*1.56+mod(\row,2)*0.78}%
        \pgfmathsetmacro\hy{\row*1.35}%
        \node[regular polygon,regular polygon sides=6,minimum size=1.8cm,
          draw=acnblue!55!acnnavy,line width=0.5pt,rotate=30]
          at ([xshift=\hx cm,yshift=-\hy cm]current page.north west) {};}}
      \fill[acnnavy,opacity=0.72] (current page.north west) rectangle
        ([yshift=-10.2cm,xshift=-6.5cm]current page.north east);
    \end{scope}
    \fill[acnteal] ([yshift=-10.2cm]current page.north west) rectangle
      ([yshift=-10.45cm]current page.north east);
    % ---- course line
    \node[anchor=north west,text=white,font=\ACNheadfont\small\bfseries,
      text width=15cm,align=\ACNalign]
      at ([xshift=2.3cm,yshift=-1.6cm]current page.north west)
      {\ACNnodetext{\ACNstrCourse\quad\textcolor{acnteal!70!white}{|}\quad\ACNstrNotes}};
    % ---- big lecture number
    \node[anchor=north west,text=acnteal!80!white,font=\ACNdispfont\bfseries\fontsize{86}{86}\selectfont]
      at ([xshift=2.1cm,yshift=-2.5cm]current page.north west) {\ACNlatin{\ACNnumber}};
    \node[anchor=north west,text=white!85!acnsky,font=\ACNheadfont\large\bfseries]
      at ([xshift=2.3cm,yshift=-5.85cm]current page.north west)
      {\ACNnodetext{\ACNstrLecture~\ACNnumber}};
    % ---- titles
    \node[anchor=north west,text=white,font=\ACNdispfont\bfseries\fontsize{25}{30}\selectfont,
      text width=16.2cm,align=\ACNalign]
      at ([xshift=2.3cm,yshift=-6.75cm]current page.north west)
      {\ACNnodetext{\ACNtitle}};
    \node[anchor=north west,text=acnsky,font=\ACNotherfont\normalsize,
      text width=16.2cm,align=\ACNotheralign]
      at ([xshift=2.3cm,yshift=-8.85cm]current page.north west)
      {\ACNothertext{\ACNtitleother}};
    % ---- generation track
    \begin{scope}[shift={([xshift=2.6cm,yshift=-11.55cm]current page.north west)}]
      \draw[acnline,line width=1.4pt] (0,0) -- (12,0);
      \foreach \g [count=\k from 0] in {2,3,4,5,6}{%
        \ifnum\g=\ACNgen
          \node[circle,fill=acnteal,draw=acnteal,line width=1pt,minimum size=0.95cm,
            text=white,font=\ACNheadfont\small\bfseries] at (\k*3,0) {\g G};
        \else
          \node[circle,fill=white,draw=acnline,line width=1pt,minimum size=0.95cm,
            text=acngray,font=\ACNheadfont\small\bfseries] at (\k*3,0) {\g G};
        \fi}
    \end{scope}
  \end{tikzpicture}%
  \vspace*{10.6cm}%
  \par\noindent
  \begin{tcolorbox}[enhanced,colback=acnmist,boxrule=0pt,arc=4pt,
      left=10pt,right=10pt,top=8pt,bottom=8pt,sidebyside,lefthand ratio=0.5,
      sidebyside gap=14pt,segmentation hidden]
    \ACNinfo{\ACNstrPart}{\ACNpart}\par\smallskip
    \ACNinfo{\ACNstrTime}{\ACNtime}
    \tcblower
    \ACNinfo{\ACNstrLevel}{\ACNlevel}\par\smallskip
    \ACNinfo{\ACNstrPrereq}{\ACNprereq}
  \end{tcolorbox}
  \vspace{2pt}
  {\ACNheadfont\bfseries\color{acnnavy}\ACNstrContents}\par
  {\color{acnline}\rule{\linewidth}{0.6pt}}\par\vspace{-0.6em}
  \begingroup\small\setlength{\parskip}{0pt}%
  \renewcommand*{\l@section}[2]{\ACNtocline{##1}{##2}}%
  \setcounter{tocdepth}{1}%
  \setlength{\columnsep}{1.2cm}%
  \begin{multicols}{2}\@starttoc{toc}\end{multicols}%
  \endgroup
  \vfill
  {\footnotesize\color{acngray}\ACNstrCoverNote\par}
  \clearpage
}
% one compact contents line: number, title, dotted leader, page
\newcommand{\ACNtocline}[2]{%
  \par\noindent\begingroup\hypersetup{linkcolor=acnnavy}\@tempdima=2.0em\relax
  \ACNheadfont\footnotesize #1\nobreak\leaders\hbox{\,\textcolor{acnline}{.}\,}\hfill
  \nobreak\textcolor{acnteal}{\bfseries #2}\par\endgroup\vspace{2.5pt}}
\newcommand{\ACNinfo}[2]{%
  {\ACNheadfont\scriptsize\bfseries\color{acnteal}\MakeUppercase{#1}}\par
  {\small\color{acnnavy}#2}}
\makeatother
