  sl/.style={draw=acnnavy,fill=acnsky,line width=0.4pt},

    \pgfmathtruncatemacro\scs{15*\n}
    \node[caplabel,anchor=east] at (-0.01,\y+0.3) {$\mu=\m$: \scs\,kHz};
    \node[caplabel,anchor=west] at (1.01,\y+0.3) {\n~\bil{slot(s)}{شکاف}};
  }
