\subsection{مقدمه}

نمونه‌گیر \lr{Metropolis} هفته‌ی قبل یک پیچ تنظیم داشت، یعنی انحراف معیار
پیشنهادی، و یک سؤال آزاردهنده: \emph{از کجا بدانیم زنجیره همگرا شده
است؟} این هفته به هر دو نگرانی جواب می‌دهد.

\textbf{نمونه‌گیری \lr{Gibbs}} مسئله‌ی تنظیم را برای دسته‌ی بزرگ و مهمی از
مدل‌ها حذف می‌کند. به‌جای پیشنهاد یک پرش برای همه‌ی پارامترها و گاهی رد آن،
هر بار \textbf{یک پارامتر} را به‌روز می‌کند و آن را \emph{دقیقاً} از توزیعش
به شرط مقادیر فعلی همه‌ی پارامترهای دیگر می‌کشد. هر نمونه‌ای پذیرفته می‌شود.
بسیاری از برنامه‌های کلاسیک بیزی \lr{(BUGS}، \lr{JAGS}، که در سراسر
کتاب‌های \lr{Lee} \lr{\&} \lr{Wagenmakers} و \lr{Kruschke} استفاده می‌شوند) بر
همین ایده ساخته شده‌اند.

\textbf{تشخیص همگرایی} بخش صادقانه‌ی \lr{MCMC} است. راه‌حل استاندارد اجرای
\textbf{چند زنجیره از نقاط شروع مختلف} و بررسی توافق آن‌هاست. \(\hat R\)
این را کمّی می‌کند که «آیا زنجیره‌ها توافق دارند؟»، و \lr{ESS} این را که
«نمونه‌های همبسته چقدر اطلاعات مستقل دارند؟». هیچ نتیجه‌ی \lr{MCMC} نباید
بدون این‌ها گزارش شود.

\paragraph{این موضوع کجا به کار می‌رود؟}

\begin{itemize}
\tightlist
\item
  \textbf{هر تحلیل \lr{MCMC} منتشرشده} باید شاخص‌های همگرایی را گزارش
  کند؛ داوران در روان‌شناسی و علوم اعصاب امروز به‌طور معمول \(\hat R\) و
  \lr{ESS} را می‌خواهند.
\item
  \textbf{نمونه‌گیری \lr{Gibbs}} پایه‌ی برنامه‌های بیزی کلاسیک \lr{BUGS} و
  \lr{JAGS} است که در بسیاری از کتاب‌های درسی مدل‌سازی شناختی \lr{(Lee} \lr{\&}
  \lr{Wagenmakers}، \lr{Kruschke)} و در پژوهش‌های بوم‌شناسی، همه‌گیرشناسی و
  آموزش استفاده می‌شوند.
\item
  \textbf{پردازش تصویر و مخابرات:} نمونه‌گیرهای \lr{Gibbs} برای بازسازی
  تصویر و برای آشکارسازی در گیرنده‌های چندآنتنی \lr{(MIMO)} به کار
  رفته‌اند.
\item
  \textbf{مدل‌های موضوعی \lr{(topic models)}} برای پیدا کردن موضوع‌ها در
  مجموعه‌های بزرگ متن، به‌طور کلاسیک با نمونه‌گیری \lr{Gibbs} برازش می‌شدند.
\end{itemize}

\subsection{جزوه‌ی درس}

\paragraph{به زبان ساده}

\lr{Metropolis} به یک اندازه‌ی گام خوب‌انتخاب‌شده نیاز دارد.
\textbf{نمونه‌گیری \lr{Gibbs}} برای بسیاری از مدل‌ها از این مشکل دوری
می‌کند، چون \textbf{هر بار یک پارامتر} را به‌روز می‌کند: همه‌ی پارامترهای
دیگر را روی مقدار فعلی‌شان ثابت نگه دار و پارامتر انتخاب‌شده را مستقیم از
توزیع دقیقش در همان شرایط بکش (\textbf{\lr{full conditional}} آن). بعد
سراغ پارامتر بعدی برو و همین‌طور ادامه بده. هیچ‌وقت چیزی رد نمی‌شود. نیمه‌ی
دوم هفته به این سؤال ناخوشایند جواب می‌دهد: «آیا نمونه‌گیر من واقعاً کار
کرده؟» \textbf{چند زنجیره} را از نقطه‌های شروع خیلی متفاوت اجرا می‌کنیم.
اگر همه به توصیف یک توزیع برسند، شاهد خوبی برای همگرایی است؛ \(\hat R\)
\lr{(«R-hat»)} یک عدد است که می‌سنجد زنجیره‌ها با هم توافق دارند یا نه
(باید کمتر از 1.01 باشد). چون نمونه‌های پشت سر هم همبسته‌اند، 4,000 نمونه
ممکن است فقط اطلاعات مثلاً 1,000 نمونه‌ی مستقل را داشته باشند: این عدد
\textbf{اندازه‌ی مؤثر نمونه \lr{(ESS)}} است و می‌گوید میانگین‌های ما چقدر
دقیق‌اند.

\textbf{زمان‌بندی پیشنهادی (60 دقیقه):} نمونه‌گیری \lr{Gibbs} 20 · وقتی
\lr{Gibbs} به زحمت می‌افتد 5 · همگرایی و چند زنجیره 8 · \(\hat R\) 10 ·
\lr{ESS} و \lr{MCSE} 10 · چک‌لیست گزارش‌دهی 7.

\subsubsection{\texorpdfstring{1. نمونه‌گیری
\lr{Gibbs}}{1. نمونه‌گیری Gibbs}}

برای پارامترهای \(\boldsymbol\theta = (\theta_1,\allowbreak \dots,\allowbreak \theta_k)\)،
\textbf{\lr{full conditional}} \(\theta_m\) پسین آن به شرط داده و
\emph{همه‌ی} پارامترهای دیگر است،
\(p(\theta_m \mid \boldsymbol\theta_{-m},\allowbreak \mathbf{y})\). نمونه‌گیری
\lr{Gibbs} بین آن‌ها می‌چرخد:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  از \(\boldsymbol\theta^{(0)}\) شروع کنید.
\item
  برای \(s = 1, \dots, S\):

  \begin{itemize}
  \tightlist
  \item
    بکشید
    \(\theta_1^{(s)} \sim p(\theta_1 \mid \theta_2^{(s-1)},\allowbreak \dots,\allowbreak \theta_k^{(s-1)},\allowbreak \mathbf{y})\)
  \item
    بکشید
    \(\theta_2^{(s)} \sim p(\theta_2 \mid \theta_1^{(s)},\allowbreak \theta_3^{(s-1)},\allowbreak \dots,\allowbreak \mathbf{y})\)
  \item
    \ensuremath{\ldots} (همیشه با جدیدترین مقادیر)
  \item
    بکشید
    \(\theta_k^{(s)} \sim p(\theta_k \mid \theta_1^{(s)},\allowbreak \dots,\allowbreak \theta_{k-1}^{(s)},\allowbreak \mathbf{y})\)
  \end{itemize}
\end{enumerate}

این حالت خاصی از \lr{Metropolis\lr{--}Hastings} است که در آن پیشنهاد خود
\lr{full conditional} است، پس احتمال پذیرش همیشه 1 است.

\textbf{پیدا کردن \lr{full conditional}ها.} چگالی مشترک
\(p(\mathbf{y}\mid\boldsymbol\theta)\,p(\boldsymbol\theta)\) را بنویسید،
فقط ضرایبی را که \(\theta_m\) دارند نگه دارید، و یک توزیع شناخته‌شده را
تشخیص دهید. این وقتی کار می‌کند که پیشین‌ها (به‌طور شرطی)
\textbf{\lr{conjugate}} باشند.

\textbf{مثال: یک مدل نرمال برای لگاریتم زمان واکنش.} فرض کنید \(y_i\)
لگاریتم زمان‌های واکنش باشند،
\(y_i \sim \mathcal{N}(\mu,\allowbreak 1/\sqrt{\tau})\)، که در آن
\(\tau = 1/\sigma^2\) \textbf{دقت} \lr{(precision)} است. پیشین‌ها:
\(\mu \sim \mathcal{N}(m_0, s_0)\) و
\(\tau \sim \mathrm{Gamma}(a_0, \text{\lr{rate }} b_0)\)، مستقل. آن‌وقت:

\[
\mu \mid \tau, \mathbf{y} \sim \mathcal{N}\!\left(\frac{m_0/s_0^2 + \tau \sum_i y_i}{1/s_0^2 + N\tau},\ \ \frac{1}{\sqrt{1/s_0^2 + N\tau}}\right),
\]

\[
\tau \mid \mu, \mathbf{y} \sim \mathrm{Gamma}\!\left(a_0 + \frac{N}{2},\ \ \text{\lr{rate}} = b_0 + \frac12\sum_i (y_i - \mu)^2\right).
\]

\textbf{خواندن فرمول‌ها.} میانگین شرطی \(\mu\) میانگین وزنی (با وزن دقت)
میانگین پیشین و میانگین داده است (مثل \lr{beta-binomial} هفته‌ی 9). توزیع
شرطی \(\tau\) برای هر نقطه‌ی داده نیم مشاهده به پارامتر شکل و نیم مربع
باقی‌مانده به \lr{rate} اضافه می‌کند.

\begin{quote}
قاطی کردن \lr{rate} و \lr{scale} رایج‌ترین باگ در نمونه‌گیرهای \lr{Gibbs}
است.
\end{quote}

\paragraph{خواندن ریاضی}

\begin{itemize}
\tightlist
\item
  \(\boldsymbol\theta_{-m}\): «همه‌ی پارامترها به‌جز \(\theta_m\)».
\item
  \textbf{دقت \lr{(precision)}} \(\tau = 1/\sigma^2\): معکوس واریانس.
  دقت زیاد = نویز کم. کار کردن با دقت‌ها ترکیب اطلاعات را به یک جمع ساده
  تبدیل می‌کند.
\item
  \textbf{به‌روزرسانی \(\mu\):} میانگین شرطی
  \(\dfrac{m_0/s_0^2 + \tau\sum y_i}{1/s_0^2 + N\tau}\) میانگین وزنی
  میانگین \lr{prior} یعنی \(m_0\) (با وزن \(1/s_0^2\)، دقت \lr{prior)} و
  میانگین داده (با وزن کل \(N\tau\)، دقت داده) است. دقت شرطی \emph{جمع}
  این دو دقت است. مثال حل‌شده: \(1 + 4\times10 = 41\)، پس داده غالب است.
\item
  \textbf{به‌روزرسانی \(\tau\):} یک توزیع \lr{Gamma} که \lr{shape} آن
  برای هر مشاهده \(1/2\) و \lr{rate} آن به اندازه‌ی نصف هر مربع باقی‌مانده
  بزرگ می‌شود. میانگین آن، \lr{shape/rate}، تقریباً «1 / (میانگین مربع
  باقی‌مانده)» است، یعنی همان دقتی که داده نشان می‌دهد.
\end{itemize}

\subsubsection{\texorpdfstring{2. وقتی \lr{Gibbs} به زحمت می‌افتد، و
\lr{Metropolis-within-Gibbs}}{2. وقتی Gibbs به زحمت می‌افتد، و Metropolis-within-Gibbs}}

\begin{itemize}
\tightlist
\item
  \textbf{همبستگی‌های قوی پسین.} \lr{Gibbs} موازی با محورهای مختصات حرکت
  می‌کند. اگر پسین یک \lr{ridge} باریک مورب باشد (مثل \lr{trade-off}
  \(A\)\lr{--}\(d\) مدل توانی)، هر گام فقط فاصله‌ی ریزی می‌تواند حرکت کند و
  زنجیره آرام در امتداد \lr{ridge} زیگزاگ می‌زند. راه‌حل‌ها:
  \lr{reparameterization}، به‌روزرسانی هم‌زمان پارامترهای همبسته
  («بلوک‌بندی»)، یا نمونه‌گیرهای گرادیانی (هفته‌ی 13).
\item
  \textbf{\lr{full conditional}های \lr{non-conjugate.}} بیشتر مدل‌های
  شناختی (منحنی‌های فراموشی، \lr{SDT}، \lr{GCM)} توزیع شرطی بسته ندارند.
  \textbf{\lr{Metropolis-within-Gibbs}} هر پارامتر را با یک گام
  \lr{Metropolis} یک‌بعدی که \lr{full conditional}ش را هدف می‌گیرد به‌روز
  می‌کند، و هر جا ممکن باشد از نمونه‌های دقیق \lr{Gibbs} استفاده می‌کند.
\end{itemize}

\begin{quote}
\textbf{پیشرفته:} \lr{JAGS} و \lr{BUGS} برای هر پارامتر خودکار یک
نمونه‌گیر انتخاب می‌کنند \lr{(Gibbs conjugate}، \lr{slice sampling}،
\lr{Metropolis} تطبیقی)، و به همین دلیل روی تقریباً هر مدلی اجرا می‌شوند.
\end{quote}

\subsubsection{3. همگرایی: منطق چند زنجیره}

یک زنجیره‌ی \lr{MCMC} وقتی همگرا شده که نمونه‌هایش نماینده‌ی پسین باشند. دو
حالت شکست: 1. \textbf{هنوز ایستا نشده:} زنجیره هنوز بازتاب مقدار شروع
است. 2. \textbf{گیرکرده یا کند:} زنجیره فقط بخشی از پسین را کاوش می‌کند
(مثلاً یکی از دو مُد، یا یک سر یک \lr{ridge} طولانی).

\(C \ge 4\) زنجیره از نقاط شروع \textbf{بیش‌پراکنده}
\lr{(over-dispersed)} اجرا کنید: اگر همه به توصیف همان توزیع برسند، این
شاهد خوبی (هرچند هرگز قطعی نیست) برای همگرایی است.

\textbf{\lr{trace plot}های} همه‌ی زنجیره‌ها روی هم باید مثل یک «کرم
پشمالوی» خوب مخلوط‌شده باشند، بدون زنجیره‌ای که آشکارا جدا افتاده یا روند
دارد.

\subsubsection{\texorpdfstring{4. ضریب کاهش مقیاس بالقوه
\(\hat R\)}{4. ضریب کاهش مقیاس بالقوه \textbackslash hat R}}

واریانس \textbf{بین} زنجیره‌ها را با واریانس \textbf{درون} زنجیره‌ها
مقایسه کنید. با \(C\) زنجیره به طول \(S\) (بعد از \lr{burn-in)}، میانگین
زنجیره‌ها \(\bar\theta_c\) و واریانس‌های درون‌زنجیره‌ای \(s_c^2\):

\[
W = \frac{1}{C}\sum_c s_c^2, \qquad
B = \frac{S}{C-1}\sum_c (\bar\theta_c - \bar\theta)^2, \qquad
\widehat{\mathrm{var}}^{+} = \frac{S-1}{S}\,W + \frac{1}{S}\,B, \qquad
\hat R = \sqrt{\frac{\widehat{\mathrm{var}}^{+}}{W}}.
\]

اگر زنجیره‌ها مخلوط نشده باشند، \(B\) نسبت به \(W\) بزرگ است و
\(\hat R > 1\). با همگرا شدن زنجیره‌ها، \(\hat R \to 1\).

\begin{itemize}
\tightlist
\item
  \textbf{\lr{Split-}\(\hat R\):} هر زنجیره را به نیمه‌ی اول و دوم تقسیم
  کنید و با آن‌ها مثل \(2C\) زنجیره رفتار کنید. این کار زنجیره‌ی تنهایی را
  هم که هنوز در حال \lr{drift} است تشخیص می‌دهد.
\item
  \textbf{آستانه:} قاعده‌ی کلاسیک \(\hat R < 1.1\) بود؛ \(\hat R < 1.01\)
\end{itemize}

\subsubsection{\texorpdfstring{5. اندازه‌ی مؤثر نمونه و خطای استاندارد
\lr{Monte Carlo}}{5. اندازه‌ی مؤثر نمونه و خطای استاندارد Monte Carlo}}

نمونه‌های خودهمبسته اطلاعات کمتری از نمونه‌های مستقل دارند.
\textbf{اندازه‌ی مؤثر نمونه} برابر است با

\[
\mathrm{ESS} = \frac{C\cdot S}{1 + 2\sum_{h=1}^{\infty}\rho_h},
\]

که در آن \(\rho_h\) خودهمبستگی با تأخیر \(h\) است که از همه‌ی زنجیره‌ها
تخمین زده و وقتی تخمین‌ها نویز می‌شوند بریده می‌شود. \lr{ESS} تعداد
نمونه‌های \emph{مستقلی} است که همان دقت را می‌دادند.

\begin{itemize}
\tightlist
\item
  \textbf{خطای استاندارد \lr{Monte Carlo}} میانگین پسین برابر
  \(\mathrm{MCSE} = \text{\lr{posterior SD}}/\sqrt{\mathrm{ESS}}\) است. باید
  نسبت به دقتی که گزارش می‌کنید کوچک باشد (مثلاً اگر \lr{MCSE} برابر 0.01
  است 3 رقم اعشار گزارش نکنید).
\item
  \textbf{قاعده‌ی سرانگشتی:} \lr{bulk-ESS} \ensuremath{\geq} 400 (با 4 زنجیره) برای
  میانگین‌ها و بازه‌های پسین قابل اعتماد؛ \lr{tail-ESS} \ensuremath{\geq} 400 برای
  \lr{quantile}های حدی قابل اعتماد.
\end{itemize}

\paragraph{خواندن ریاضی}

\begin{itemize}
\tightlist
\item
  \textbf{\(\hat R\):} \(W\) میانگین واریانس \emph{درون} هر زنجیره است؛
  \(B\) می‌سنجد \emph{میانگین‌های} زنجیره‌ها چقدر از هم دورند.
  \(\widehat{\mathrm{var}}^{+}\) تخمینی از کل واریانس \lr{posterior} است
  که اختلاف بین زنجیره‌ها را هم در بر دارد. اگر زنجیره‌ها توافق داشته
  باشند، \(B\) تقریباً چیزی اضافه نمی‌کند و
  \(\hat R = \sqrt{\widehat{\mathrm{var}}^{+}/W} \approx 1\). در مثال
  حل‌شده‌ی «همگرا نشده»، دو زنجیره حول 1.0 و 2.0 هستند، \(B\) خیلی بزرگ
  است و \(\hat R = 4.4\).
\item
  \textbf{\lr{ESS:}} \(C \cdot S\) کل تعداد نمونه‌هاست؛ مخرج
  \(1 + 2\sum\rho_h\) برای نمونه‌های همبسته بزرگ می‌شود. با
  \(\rho_h = 0.5^h\) جمع برابر \(0.5 + 0.25 + \dots = 1\) است، پس
  نمونه‌ها فقط یک‌سوم حساب می‌شوند: 4,000 نمونه ارزش 1,333 نمونه‌ی مستقل را
  دارند.
\item
  \textbf{\lr{MCSE}} \(= \text{\lr{posterior SD}}/\sqrt{\mathrm{ESS}}\) خطای
  شبیه‌سازی میانگین گزارش‌شده است؛ تعداد رقم‌های اعشار را از روی آن تصمیم
  بگیرید.
\end{itemize}

\subsubsection{\texorpdfstring{6. چک‌لیست گزارش‌دهی
\lr{MCMC}}{6. چک‌لیست گزارش‌دهی MCMC}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  نمونه‌گیر، تعداد زنجیره‌ها، تکرارها، \lr{burn-in (warm-up)} و هر
  \lr{thinning.}
\item
  مقادیر شروع (پراکنده؟).
\item
  \lr{trace plot}ها (در پیوست) و جمله‌ای که نشان دهد بررسی شده‌اند.
\item
  پیشین‌ها، و خلاصه‌های پسین با دقتی متناسب با \lr{MCSE.}
\item
  یک بررسی پیش‌بینی پسین (هفته‌ی 10).
\end{enumerate}

\paragraph{برای یک مهندس}

\begin{itemize}
\tightlist
\item
  \textbf{میانگین‌گیری وزن‌دار با دقت} در به‌روزرسانی \(\mu\) همان قاعده‌ی
  \textbf{ترکیب با نسبت بیشینه \lr{(MRC)}} یا گام به‌روزرسانی اندازه‌گیری
  در فیلتر کالمن است: تخمین‌ها را به نسبت معکوس واریانس نویزشان ترکیب کن.
\item
  \textbf{نمونه‌گیری \lr{Gibbs}} هر بار یک متغیر را به شرط بقیه به‌روز
  می‌کند؛ مثل الگوریتم‌های تکراری مختصات‌به‌مختصات (بهینه‌سازی متناوب) و
  تبادل تکراری اطلاعات شرطی در گیرنده‌های از نوع \lr{turbo.} نمونه‌گیرهای
  \lr{Gibbs} برای آشکارسازی \lr{MIMO} هم پیشنهاد شده‌اند.
\item
  \textbf{\(\hat R\)} یک \lr{ANOVA} یک‌طرفه (واریانس بین گروه‌ها در برابر
  درون گروه‌ها) روی اجراهای مستقل شبیه‌سازی است: همان بررسی‌ای که با اجرای
  یک شبیه‌سازی \lr{Monte Carlo} با چند \lr{seed} مختلف و وارسی توافق
  اجراها انجام می‌دهید.
\item
  \textbf{\lr{ESS}} \(= N/(1 + 2\sum\rho_h)\) فرمول استاندارد تعداد مؤثر
  نمونه‌های مستقل یک فرایند تصادفی همبسته است: میانگین گرفتن از \(N\)
  نمونه‌ی همبسته واریانس را فقط به اندازه‌ی میانگین گرفتن از \lr{ESS}
  نمونه‌ی مستقل کم می‌کند (مرتبط با زمان همبستگی).
\end{itemize}

\paragraph{برای مدرس}

\begin{itemize}
\tightlist
\item
  \textbf{\lr{Gibbs} را روی یک نمودار تراز رسم کنید.} دو پارامتر، یک
  \lr{posterior} بیضوی. \lr{Gibbs} فقط با گام‌های افقی و عمودی حرکت می‌کند
  (یک پلکان). بعد یک بیضی مورب باریک بکشید: گام‌های پلکان خیلی ریز
  می‌شوند. همین یک تصویر هم نحوه‌ی کار \lr{Gibbs} و هم زمان به زحمت
  افتادنش را توضیح می‌دهد.
\item
  \textbf{\(\hat R\) را اول تصویری کنید:} چهار \lr{trace plot} را روی هم
  بکشید، یک بار با زنجیره‌هایی که توافق دارند و یک بار با یک زنجیره که
  جای دیگری گیر کرده. پیش از نشان دادن هر فرمولی بپرسید «آیا این
  زنجیره‌ها یک توزیع را توصیف می‌کنند؟»
\item
  \textbf{چک‌لیست گزارش را به‌عنوان یک عادت درس دهید.} از دانشجوها بخواهید
  آن را روی پروژه‌شان و روی هر نتیجه‌ی \lr{MCMC} در \lr{lab} اعمال کنند.
\item
  \textbf{سؤال‌های رایج.}

  \begin{itemize}
  \tightlist
  \item
    \emph{«\(\hat R\) برابر 1.00 است، پس مدلم درست است؟»} نه. \(\hat R\)
    فقط بررسی می‌کند که زنجیره‌ها توافق دارند؛
  \item
    \emph{«زنجیره‌ها را \lr{thin} کنم؟»} برای درستی لازم نیست؛ فقط حافظه
    را صرفه‌جویی می‌کند.
  \end{itemize}
\end{itemize}

\subsection{مثال حل‌شده}

\subsubsection{\texorpdfstring{بخش 1: یک تکرار \lr{Gibbs} با
دست}{بخش 1: یک تکرار Gibbs با دست}}

داده: \(N = 4\) لگاریتم زمان واکنش،
\(\mathbf{y} = (6.0, 6.3, 6.1, 6.4)\)، پس \(\sum y_i = 24.8\). پیشین‌ها:
\(m_0 = 6\)، \(s_0 = 1\)، \(a_0 = 2\)، \(b_0 = 0.1\). مقادیر فعلی:
\(\mu = 6.0\)، \(\tau = 10\).

\textbf{به‌روزرسانی \(\mu\).} دقت \(= 1/1^2 + 4 \times 10 = 41\)؛ میانگین
\(= (6/1 + 10 \times 24.8)/41 = (6 + 248)/41 = 6.195\)؛ انحراف معیار
\(= 1/\sqrt{41} = 0.156\). مثلاً \(\mu = 6.22\) کشیده می‌شود.

\textbf{به‌روزرسانی \(\tau\)} با \(\mu = 6.22\). مربع باقی‌مانده‌ها:
\(0.0484 + 0.0064 + 0.0144 + 0.0324 = 0.1016\). شکل \(= 2 + 4/2 = 4\)؛
\lr{rate} \(= 0.1 + 0.1016/2 = 0.1508\). میانگین شرطی \(\tau\) برابر
\(4/0.1508 = 26.5\) است (یعنی \(\sigma \approx 0.19\)). مثلاً
\(\tau = 24.1\) کشیده می‌شود.

حالت جدید: \((\mu, \tau) = (6.22, 24.1)\). هیچ گام پذیرش/ردی لازم نبود.

\subsubsection{\texorpdfstring{بخش 2: \(\hat R\) با
دست}{بخش 2: \textbackslash hat R با دست}}

\(S = 4\)

\textbf{همگرا نشده.} زنجیره‌ی 1: (1.0، 1.2، 0.8، 1.0)؛ زنجیره‌ی 2: (2.0،
2.2، 1.8، 2.0). - میانگین زنجیره‌ها: 1.0 و 2.0. واریانس‌های درون‌زنجیره‌ای
\lr{(ddof} = 1): هر دو \(0.08/3 = 0.0267\)، پس \(W = 0.0267\). -
\(B = S \times \mathrm{var}(\text{\lr{means}}) = 4 \times 0.5 = 2.0\). -
\(\widehat{\mathrm{var}}^{+} = \tfrac34 \times 0.0267 + \tfrac14 \times 2.0 = 0.520\).
- \(\hat R = \sqrt{0.520/0.0267} = 4.4\). زنجیره‌ها به‌وضوح با هم
مخالف‌اند.

\textbf{همگرا شده (با اعداد واقع‌بینانه).} \(S = 1000\) نمونه در هر
زنجیره، \(W = 0.05\)، میانگین زنجیره‌ها 1.00 و 1.02:
\(B = 1000 \times \mathrm{var}(1.00,\allowbreak 1.02) = 1000 \times 0.0002 = 0.2\)؛
\(\widehat{\mathrm{var}}^{+} = 0.999 \times 0.05 + 0.2/1000 = 0.05015\)؛
\(\hat R = \sqrt{0.05015/0.05} = 1.0015\). کمتر از 1.01: خوب است.

\subsubsection{\texorpdfstring{بخش 3: \lr{ESS} با
دست}{بخش 3: ESS با دست}}

\(C \cdot S = 4000\) نمونه با خودهمبستگی‌های \(\rho_h = 0.5^h\). آن‌وقت
\(\sum_{h\ge1}\rho_h = 1\)، پس \(\mathrm{ESS} = 4000/(1 + 2) = 1333\).
با انحراف معیار پسین 0.2، \lr{MCSE} \(= 0.2/\sqrt{1333} = 0.0055\):
میانگین پسین را با دو رقم اعشار گزارش کنید.

\subsubsection{\texorpdfstring{بخش \lr{D:} گرفتن عدم همگرایی (30

\textbf{به این دقت کنید:} با گام‌های ریز و زنجیره‌های کوتاه، هر زنجیره
هنوز نزدیک جایی است که شروع کرده. \lr{trace}ها جدا از هم‌اند، \(\hat R\)

صاف و پایدار به نظر می‌رسد}، و دقیقاً به همین دلیل یک زنجیره هرگز کافی
نیست. نمونه‌گیر تنظیم‌شده \lr{trace}های روی هم افتاده، \(\hat R = 1.002\)
و \lr{ESS} حدود 1000 از 20 000 نمونه می‌دهد. نمونه‌گیر \lr{Gibbs} بخش
\lr{C}، با نمونه‌های تقریباً ناهمبسته، از 10 000 نمونه به \lr{ESS} حدود
9000 تا 10 000 رسید.

\subsubsection{\texorpdfstring{بخش \lr{E:} تفسیر و گزارش (15
دقیقه)}{بخش E: تفسیر و گزارش (15 دقیقه)}}

یک پاراگراف روش‌ها و نتایج \lr{MCMC} برای مدل لگاریتم زمان واکنش، طبق
چک‌لیست بخش 6 جزوه، بنویسید: نمونه‌گیر، زنجیره‌ها، تکرارها، \lr{burn-in}،
مقادیر شروع، \(\hat R\)، \lr{ESS}، میانگین‌های پسین و بازه‌های 95\%
\(\mu\) و \(\sigma\) (با \(e^\mu\) مقدار \(\mu\) را به میلی‌ثانیه
برگردانید و بگویید نماینده‌ی چیست).

\subsubsection{تمرین‌ها (به ترتیب سختی)}

\textbf{تمرین 1 \ensuremath{\star}.} به‌صورت جبری نشان دهید که \lr{full conditional}
\(\tau\) یک توزیع \lr{Gamma} است:
\(p(\tau\mid\mu,\mathbf{y}) \propto \tau^{N/2}\exp(-\tfrac{\tau}{2}\sum(y_i-\mu)^2)\cdot\tau^{a_0-1}e^{-b_0\tau}\)
را بنویسید و شکل و \lr{rate} را بخوانید.

آیا تشخیص‌ها آن را می‌گیرند؟ اعتبارسنجی با شبکه چطور؟ این درباره‌ی تفاوت
\emph{همگرایی} و \emph{درستی} چه می‌گوید؟

\textbf{تمرین 3 \ensuremath{\star}\ensuremath{\star} \lr{(Gibbs} همبسته).} یک نمونه‌گیر \lr{Gibbs} برای یک
«پسین» نرمال دومتغیره با همبستگی \(\rho\) بنویسید \lr{(full
conditional}ها:
\(\theta_1\mid\theta_2 \sim \mathcal{N}(\rho\theta_2,\allowbreak \sqrt{1-\rho^2})\)
و برعکس). \lr{ESS} را برای \(\rho = 0, 0.9, 0.99\) با 5000 تکرار مقایسه
کنید و مسیر زیگزاگ را برای \(\rho = 0.99\) رسم کنید.

\lr{ESS} در هر ثانیه را با نمونه‌گیر \lr{Metropolis} مشترک مقایسه کنید.
چرا ممکن است اینجا بدتر باشد؟ (راهنمایی: همبستگی \(A\)\lr{--}\(d\).)

\textbf{تمرین 5 \ensuremath{\star}\ensuremath{\star}\ensuremath{\star} (پیشرفته: دو شرط).} نمونه‌گیر \lr{Gibbs} را برای
لگاریتم زمان واکنش در دو شرط با میانگین‌های جداگانه‌ی \(\mu_1, \mu_2\) و
دقت مشترک \(\tau\) گسترش دهید. داده‌ای با تفاوت واقعی 0.1 شبیه‌سازی کنید،
نمونه بگیرید، و پسین \(\mu_2 - \mu_1\) و
\(P(\mu_2 > \mu_1\mid\mathbf{y})\) را با تشخیص‌ها گزارش کنید.

\subsection{\texorpdfstring{ارتباط با هوش مصنوعی و
\lr{LLM}ها}{ارتباط با هوش مصنوعی و LLMها}}

\textbf{نمونه‌گیری \lr{Gibbs} از یک مدل زبانی \lr{masked.}} مدل‌هایی مثل
\lr{BERT} آموزش می‌بینند که یک کلمه‌ی پوشانده‌شده \lr{(masked)} را با توجه
به همه‌ی کلمه‌های دیگر پر کنند: آن‌ها \textbf{\lr{full conditional}ها}
\(P(w_i \mid \text{\lr{all other words}})\) را یاد می‌گیرند. این دقیقاً چیزی
است که یک نمونه‌گیر \lr{Gibbs} لازم دارد. پوشاندن مکرر یک موقعیت و
نمونه‌گیری دوباره‌ی آن، جمله‌های کامل تولید می‌کند \lr{(Wang} \lr{\&} \lr{Cho,}
2019). مدل اسباب‌بازی پایین فقط دو \lr{full conditional} یک «زبان»
دوکلمه‌ای را می‌داند، و نمونه‌گیری \lr{Gibbs} توزیع توأم جمله‌ها را بازیابی
می‌کند. چون این زبان «چسبناک» است \lr{(dog} با \lr{barks} می‌آید، \lr{cat}
با \lr{meows)}، زنجیره کند مخلوط می‌شود، و \(\hat R\) این را در اجراهای
کوتاه تشخیص می‌دهد.

حالا برای \(\hat R < 1.01\) به چند تکرار نیاز دارید؟ - \ensuremath{\star}\ensuremath{\star} مدل‌های
\lr{masked} واقعی برای هر موقعیت جداگانه آموزش می‌بینند، پس لازم نیست
توزیع‌های شرطی‌شان از یک توزیع توأم واحد آمده باشند. این درباره‌ی اعتماد به
نمونه‌گیری که توزیع‌های شرطی‌اش یادگرفته‌شده‌اند چه می‌گوید؟

\emph{مطالعه‌ی اختیاری:} \lr{Wang, A.,} \lr{\&} \lr{Cho, K.} (2019). \lr{BERT
has a mouth, and it must speak: BERT as a Markov random field language
model.} \emph{\lr{arXiv:1902.04094}}.

\subsection{اشتباهات رایج و برداشت‌های نادرست}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{استفاده از مقادیر قدیمی داخل یک دور \lr{Gibbs.}} هر توزیع شرطی
  باید از \emph{جدیدترین} مقادیر پارامترهای دیگر استفاده کند.
\item
  \textbf{یک زنجیره‌ی بلند به‌جای چند زنجیره.} زنجیره‌ی گیرکرده می‌تواند
  کاملاً پایدار به نظر برسد. از \ensuremath{\geq} 4 زنجیره‌ی پراکنده استفاده کنید.
\item
  \textbf{گزارش نتایج با \lr{ESS} پایین.} با \lr{ESS} = 50، \lr{MCSE}
  میانگین پسین حدود 14\% انحراف معیار پسین است و \lr{quantile}های دنباله
  غیرقابل اعتمادند.
\item
  \textbf{\lr{thinning} برای «درست کردن» خودهمبستگی.} \lr{thinning}
  حافظه را کم می‌کند ولی اطلاعات را زیاد نمی‌کند. \lr{ESS} خودش خودهمبستگی
  را در نظر می‌گیرد.
\item
  \textbf{استفاده‌ی بی‌چون‌وچرا از آستانه‌ی 1.1.} راهنمایی امروزی
  \(\hat R < 1.01\) با \lr{split-}\(\hat R\) نرمال‌شده‌ی رتبه‌ای است.
\item
  \textbf{تشخیص فقط برای بعضی پارامترها.} \emph{همه‌ی} پارامترهای
  گزارش‌شده، از جمله مشتق‌شده‌ها، را بررسی کنید.
\end{enumerate}

\subsection{خودآزمایی}

\textbf{1.} چرا نمونه‌گیری \lr{Gibbs} هیچ‌وقت یک نمونه را رد نمی‌کند؟
\textbf{پاسخ} هر نمونه دقیقاً از \lr{full conditional} می‌آید. اگر آن را
\lr{Metropolis\lr{--}Hastings} با \lr{full conditional} به‌عنوان پیشنهاد
ببینیم، نسبت پذیرش همیشه 1 است.

\textbf{2.} برای اجرای یک نمونه‌گیر \lr{Gibbs} خالص به چه چیزی نیاز
دارید؟ \textbf{پاسخ} \lr{full conditional}ها برای هر پارامتر (یا بلوک)
که بتوانید مستقیماً از آن‌ها نمونه بگیرید؛ معمولاً چون پیشین‌ها به‌طور شرطی
\lr{conjugate} هستند.

\textbf{3.} یک نمونه‌گیر \lr{Gibbs} برای دو پارامتر به‌شدت همبسته \lr{ESS}
خیلی پایینی دارد. چرا، و چه می‌توانید بکنید؟ \textbf{پاسخ} \lr{Gibbs} هر
بار یک مختصه را به‌روز می‌کند، پس روی یک \lr{ridge} باریک مورب هر گام ریز
است و زنجیره آرام زیگزاگ می‌زند. راه‌حل‌ها: \lr{reparameterization} برای کم
کردن همبستگی، به‌روزرسانی هم‌زمان پارامترهای همبسته (بلوک‌بندی)، یا استفاده
از \lr{HMC/NUTS.}

\textbf{4.} \(\hat R\) را در یک جمله توضیح دهید. \textbf{پاسخ} واریانس
بین زنجیره‌ها را با واریانس درون زنجیره‌ها مقایسه می‌کند (تخمین می‌زند اگر
زنجیره‌ها تا ابد اجرا می‌شدند پراکندگی پسین چقدر می‌توانست کم شود)؛ مقادیر
نزدیک 1 (طبق راهنمایی امروزی \lr{\textless} 1.01) یعنی زنجیره‌ها توافق
دارند.

\textbf{5.} 4000 نمونه و \lr{ESS} برابر 200 برای پارامتری با انحراف
معیار پسین 0.5 دارید. \lr{MCSE} میانگین پسینش چقدر است؟ چند رقم اعشار
باید گزارش کنید؟ \textbf{پاسخ} \lr{MCSE} \(= 0.5/\sqrt{200} = 0.035\).
میانگین را با یک رقم اعشار (یا حداکثر دو) گزارش کنید، چون رقم سوم نویز
\lr{Monte Carlo} است.

\textbf{6.} چرا یک زنجیره نمی‌تواند همگرایی را نشان دهد؟ \textbf{پاسخ}
زنجیره‌ای که در یک ناحیه گیر کرده (یک مُد، یا بخشی از یک \lr{ridge)}
می‌تواند ایستا و خوب مخلوط‌شده به نظر برسد، و هیچ چیزی در خود زنجیره نواحی
کاوش‌نشده را آشکار نمی‌کند. فقط زنجیره‌هایی که از جای دیگری شروع شده و به
جای دیگری رسیده‌اند مشکل را لو می‌دهند.

\textbf{7.} چهار چیزی را که بخش روش‌ها باید درباره‌ی یک تحلیل \lr{MCMC}
گزارش کند نام ببرید. تعداد زنجیره‌ها؛ تکرارها و \lr{warm-up}؛ مقادیر
شروع؛ \(\hat R\) و \lr{ESS} برای همه‌ی پارامترها؛ اینکه \lr{trace plot}ها
بررسی شده‌اند؛ پیشین‌ها؛ یک بررسی پیش‌بینی پسین.

  مثبت بودنش را حساب کنید. \(d'\) را با \lr{MLE} هفته‌ی 4 مقایسه کنید.
\item
  \textbf{\lr{Gibbs} (2 امتیاز).} نمونه‌گیر \lr{Gibbs} دوشرطی تمرین 5 را
  صورت‌بندی کنید و پسین تفاوت میانگین‌ها را با تشخیص‌ها گزارش کنید.
\item

  حساسیت و سوگیری چه می‌کنند؟ با نتیجه‌ی آزمون نسبت درست‌نمایی هفته‌ی 4
  مقایسه کنید.

می‌شود (حداکثر 3 روز).
