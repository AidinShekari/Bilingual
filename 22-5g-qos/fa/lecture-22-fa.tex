\section{اهداف یادگیری}
پس از مطالعهٔ این درس باید بتوانید:
\begin{objectives}
  \item مدل کیفیت خدمات نسل پنجم -- جریان‌های \acro{QoS}، \acro{QFI}، نمایه‌ها و قواعد \acro{QoS} و نگاشت آن‌ها به حامل‌های رادیویی -- را
        \textbf{توضیح دهید}.
  \item ویژگی‌های استانداردشدهٔ \acro{5QI} (نوع منبع، اولویت، بودجهٔ تأخیر بسته، نرخ خطای بسته، بیشینهٔ حجم انفجار داده) را
        \textbf{تفسیر کنید}.
  \item \textbf{دنبال کنید} سیاست \acro{PCF} چگونه به جریان‌های \acro{QoS} تبدیل می‌شود که در \acro{UE}، \acro{RAN} و \acro{UPF} اعمال
        می‌شوند.
  \item بودجه‌های تأخیر، نیاز \acro{HARQ} برای یک هدف خطای بسته و اثر زمان‌بندی اولویت‌دار را \textbf{محاسبه کنید}.
  \item مدل جریان‌محور نسل پنجم را با مدل حامل‌محور \acro{LTE} \textbf{مقایسه کنید}.
\end{objectives}

\section{مقدمه و انگیزه}
کاربردهای مختلف به رفتار متفاوتی نیاز دارند: یک تماس صوتی به تأخیر کم نیاز دارد و اندکی اتلاف را تحمل می‌کند؛ انتقال یک پرونده به
اتلاف صفر نیاز دارد اما تأخیر را تحمل می‌کند؛ کنترل‌گر یک ربات هم به تأخیر کم و هم به اتلاف بسیار کم نیاز دارد؛ یک به‌روزرسانی
پس‌زمینه به هیچ‌کدام. یک شبکهٔ موبایل نمی‌تواند همهٔ این‌ها را با یک لولهٔ بهترین تلاش برآورده کند، چون منبع کمیاب -- ظرفیت رادیویی --
باید زیر بار به اشتراک گذاشته شود. سازوکارهای \acro{QoS} تعیین می‌کنند کدام بسته‌ها نخست بروند، به کدام نرخ تضمین شود و کدام ممکن است دور
ریخته شوند.

\acro{LTE} این مسئله را با \emph{حامل‌ها} حل کرد: یکی برای هر کلاس \acro{QoS}، که هر کدام به سیگنالینگ برقراری در کل مسیر نیاز داشت. نسل
پنجم حامل‌ها را در هسته با \emph{جریان‌های \acro{QoS}} سبک جایگزین می‌کند که برای هر بسته علامت‌گذاری می‌شوند و \acro{RAN} آن‌ها را
به‌طور انعطاف‌پذیر به حامل‌های رادیویی نگاشت می‌کند \cite{ts23501}. مدل حاصل مقیاس‌پذیرتر است، با ویژگی‌های جدید بحرانی از نظر تأخیر از
\acro{URLLC} پشتیبانی می‌کند و با برش‌بندی (درس ۲۱) یکپارچه است: برش شبکه را انتخاب می‌کند و جریان‌های \acro{QoS} رفتار درون آن را.

\section{مفاهیم بنیادی}

\subsection{جریان‌های \lr{QoS}}
یک نشست \acro{PDU} یک یا چند جریان \acro{QoS} دارد که هر کدام با یک \acro{QFI} شناسایی می‌شود (\cref{fig:model}). هر نشست \acro{PDU} یک
جریان \acro{QoS} پیش‌فرض (غیر\acro{GBR}) دارد که ترافیکی را که با هیچ قاعدهٔ دیگری تطابق ندارد حمل می‌کند. طبقه‌بندی در لبه‌ها انجام
می‌شود:
\begin{itemize}
  \item \textbf{فروسو}: \acro{UPF} بسته‌ها را با \acro{PDR}ها (فیلترهای بسته از قواعد \acro{PCC}) طبقه‌بندی و در سرآیند کپسوله‌سازی
        \lr{N3} با \acro{QFI} علامت‌گذاری می‌کند؛
  \item \textbf{فراسو}: \acro{UE} بسته‌ها را با \emph{قواعد \acro{QoS}} (فیلترهای بسته با تقدم) طبقه‌بندی می‌کند و لایهٔ \acro{SDAP} آن‌ها
        را علامت‌گذاری می‌کند؛
  \item \textbf{\acro{RAN}}: \acro{gNB} جریان‌های \acro{QoS} را به دلخواه خود به حامل‌های دادهٔ رادیویی (\acro{DRB}) نگاشت می‌کند -- چند
        جریان ممکن است یک \acro{DRB} را به اشتراک بگذارند.
\end{itemize}
چون علامت‌گذاری \acro{QFI} برای هر بسته است، افزودن یک جریان \acro{QoS} به تونل جدیدی نیاز ندارد: همهٔ جریان‌های یک نشست \acro{PDU} یک
تونل \lr{N3} را به اشتراک می‌گذارند.

\subsection{پارامترهای \lr{QoS}}
\begin{definitionbox}{نمایه و ویژگی‌های \lr{QoS}}
\acro{SMF} برای هر جریان \acro{QoS} یک \emph{نمایهٔ \acro{QoS}} به \acro{RAN} می‌دهد: \acro{5QI}، اولویت تخصیص و نگه‌داشت (\acro{ARP})، و
برای جریان‌های \acro{GBR} نرخ‌های بیت جریان تضمین‌شده و بیشینه (\lr{GFBR, MFBR})، و به‌طور اختیاری کنترل اعلان و بیشینهٔ نرخ اتلاف بسته.
\acro{5QI} به \emph{ویژگی‌های \acro{QoS}} اشاره می‌کند: نوع منبع (غیر\acro{GBR}، \acro{GBR}، \acro{GBR} بحرانی از نظر تأخیر)، سطح اولویت،
بودجهٔ تأخیر بسته (\acro{PDB})، نرخ خطای بسته (\acro{PER})، پنجرهٔ میانگین‌گیری و برای \acro{GBR} بحرانی از نظر تأخیر، بیشینهٔ حجم انفجار
داده (\acro{MDBV}) \cite{ts23501}.
\end{definitionbox}
محدودیت‌های تجمیعی بر ترافیک غیر\acro{GBR} اعمال می‌شوند: \emph{\lr{Session-AMBR}} برای هر نشست \acro{PDU} (که \acro{UPF} و \acro{UE}
اعمال می‌کنند) و \emph{\lr{UE-AMBR}} در همهٔ نشست‌های یک \acro{UE} (که \acro{RAN} اعمال می‌کند).

\subsection{بودجهٔ تأخیر بسته و نرخ خطای بسته}
\acro{PDB} کران بالایی برای تأخیر میان \acro{UE} و خاتمهٔ \lr{N6} در \acro{UPF} با یک اطمینان معین است؛ شامل یک بخش شبکهٔ هسته و یک بخش
شبکهٔ دسترسی است. \acro{PER} کران بالایی برای نرخ بسته‌هایی است که لایهٔ پیوند پردازش کرده اما به لایهٔ بالاتر تحویل نداده است. برای
جریان‌های \acro{GBR}، بسته‌هایی که بیش از \acro{PDB} تأخیر دارند اتلاف شمرده می‌شوند (برای \acro{GBR} بحرانی از نظر تأخیر، در \acro{PER}
شمرده می‌شوند)، پس \acro{PDB} و \acro{PER} با هم هدف قابلیت اطمینان یک جریان \acro{URLLC} را تعریف می‌کنند.

\section{توضیحات فنی تفصیلی}

\subsection{از سیاست تا جریان‌های \lr{QoS}}
\acro{QoS} با سیاست آغاز می‌شود (\cref{fig:setup}). یک تابع کاربرد (مثلاً \acro{IMS} برای یک تماس صوتی) از \acro{PCF} رفتاری برای یک جریان
رسانه می‌خواهد؛ \acro{PCF} قواعد \acro{PCC} با \acro{5QI}، \acro{ARP} و نرخ‌های بیت می‌سازد و به \acro{SMF} می‌فرستد. \acro{SMF} هر قاعدهٔ
\acro{PCC} را به یک جریان \acro{QoS} پیوند می‌دهد -- جریانی موجود با همان پارامترها یا جریانی جدید با \acro{QFI} جدید -- و نصب می‌کند:
\acro{PDR}ها و \acro{QER}ها در \acro{UPF} (\lr{N4})، نمایهٔ \acro{QoS} در \acro{gNB} (\lr{N2}) و قاعدهٔ \acro{QoS} در \acro{UE} (\acro{NAS}).
\acro{gNB} برای جریان‌های \acro{GBR} کنترل پذیرش انجام می‌دهد و \acro{DRB}ها و نگاشت \acro{SDAP} را پیکربندی می‌کند.

\subsection{\lr{QoS} بازتابی}
برای کاربردهایی با جریان‌های کوتاه بسیار (مثلاً وب‌گردی با اولویت‌های مختلف) سیگنال‌دادن یک قاعدهٔ \acro{QoS} برای هر کدام پرهزینه است. با
\emph{\acro{QoS} بازتابی}، \acro{UPF} یک نشانگر \acro{QoS} بازتابی (\acro{RQI}) را در بسته‌های فروسو تنظیم می‌کند؛ آن‌گاه \acro{UE} یک قاعدهٔ
\acro{QoS} فراسو می‌سازد که نشانی‌ها و درگاه‌های بستهٔ فروسو را با همان \acro{QFI} بازتاب می‌دهد، بدون سیگنالینگ \acro{NAS}.

\subsection{اعمال در \lr{RAN}}
زمان‌بند \acro{gNB} نمایه‌های \acro{QoS} را به تصمیم‌های زمان‌بندی ترجمه می‌کند: سطوح اولویت صف‌ها را مرتب می‌کنند، \acro{GFBR} در پنجرهٔ
میانگین‌گیری تضمین می‌شود، \acro{PDB} تعیین می‌کند یک بسته چقدر می‌تواند منتظر بماند (و کدام عددشناسی، شکاف‌های کوچک یا اعطاهای
پیکربندی‌شده لازم است) و \acro{PER} \acro{BLER} هدف و راهبرد \lr{HARQ}/تکرار را تعیین می‌کند. کنترل اعلان \acro{QoS} به \acro{gNB} اجازه
می‌دهد وقتی دیگر نمی‌تواند یک \acro{GFBR} را تضمین کند به \acro{SMF} اطلاع دهد تا کاربرد خود را تطبیق دهد (مثلاً وضوح ویدئو را کم کند).

\subsection{پایش \lr{QoS}}
از انتشار ۱۶ شبکه می‌تواند تأخیر بسته را برای هر جریان \acro{QoS} میان \acro{UE}، \acro{gNB} و \acro{UPF} اندازه بگیرد و به \acro{SMF} و از طریق
\acro{NEF} به توابع کاربرد گزارش دهد -- پایهٔ توافق‌نامه‌های سطح خدمت روی برش‌های \acro{URLLC} و کاربردهایی که با تأخیر اندازه‌گیری‌شده سازگار
می‌شوند.

\section{مدل‌های ریاضی}

\subsection{تقسیم بودجهٔ تأخیر}\label{sec:pdb}
\paragraph{مدل} \acro{PDB} میان شبکهٔ هسته (\acro{UPF} تا \acro{gNB}، $D_{\mathrm{CN}}$) و شبکهٔ دسترسی ($D_{\mathrm{AN}}$) تقسیم می‌شود:
\begin{equation}\label{eq:pdb}
  D_{\mathrm{AN}} = \mathrm{PDB} - D_{\mathrm{CN}} .
\end{equation}
\paragraph{مثال} \acro{5QI} ۱ (۱۰۰ میلی‌ثانیه) با بخش هستهٔ فرضی ۲۰ میلی‌ثانیه، ۸۰ میلی‌ثانیه برای زمان‌بندی، \acro{HARQ} و ارسال باقی
می‌گذارد -- راحت. \acro{5QI} ۸۵ (۵ میلی‌ثانیه) با یک \acro{UPF} در فاصلهٔ ۵۰ کیلومتر (۰٫۲۵ میلی‌ثانیه یک‌طرفه در فیبر به‌علاوهٔ پردازش، مثلاً ۱
میلی‌ثانیه) ۴ میلی‌ثانیه باقی می‌گذارد: \acro{UPF} باید در لبه باشد و رادیو باید از شکاف‌های کوتاه استفاده کند.

\subsection{نرخ لازم برای یک انفجار داده}\label{sec:mdbv}
\paragraph{مدل} برای یک جریان \acro{GBR} بحرانی از نظر تأخیر، انفجاری تا \acro{MDBV} بایت باید درون بودجهٔ شبکهٔ دسترسی تحویل شود. نرخ
لحظه‌ای که زمان‌بند باید در طول انفجار فراهم کند
\begin{equation}\label{eq:mdbv}
  R_{\mathrm{burst}} \ge \frac{8\,\mathrm{MDBV}}{D_{\mathrm{AN}}}
\end{equation}
است.
\paragraph{مثال} \acro{5QI} ۸۲ (\acro{MDBV} ۲۵۵ بایت، \acro{PDB} ۱۰ میلی‌ثانیه) با $D_{\mathrm{AN}} = 9$ میلی‌ثانیه: ۲۲۷ کیلوبیت بر ثانیه؛
\acro{5QI} ۸۵ (۲۵۵ بایت، \acro{PDB} ۵ میلی‌ثانیه) با $D_{\mathrm{AN}} = 4$ میلی‌ثانیه: ۵۱۰ کیلوبیت بر ثانیه. نرخ‌ها کوچک‌اند؛ چالش، تحویل
\emph{قابل اطمینان} آن‌ها در چند میلی‌ثانیه است.

\subsection{ارسال‌های \lr{HARQ} برای یک هدف \lr{PER}}\label{sec:harq}
\paragraph{مدل} اگر هر ارسال یک بلوک مستقلاً با احتمال $p$ (\acro{BLER} هدف) شکست بخورد، خطای باقی‌مانده پس از $n$ ارسال $p^n$ است. برای
رسیدن به یک \acro{PER} هدف،
\begin{equation}\label{eq:harq}
  n \ge \frac{\ln \mathrm{PER}}{\ln p},
\end{equation}
و همهٔ $n$ ارسال باید در $D_{\mathrm{AN}}$ جا شوند.
\paragraph{مثال} $\mathrm{PER} = 10^{-5}$: با $p = 0.1$، $n = 5$ ارسال (\cref{fig:harq})؛ با یک \acro{MCS} محافظه‌کارانه که $p = 0.01$ می‌دهد،
$n = 3$. اگر یک رفت‌وبرگشت \acro{HARQ} چهار شکاف ۰٫۱۲۵ میلی‌ثانیه‌ای (۱۲۰ کیلوهرتز) = ۰٫۵ میلی‌ثانیه طول بکشد، پنج ارسال به ۲ تا ۲٫۵ میلی‌ثانیه
نیاز دارند -- درون ۴ میلی‌ثانیه شدنی؛ با ۱۵ کیلوهرتز (شکاف‌های ۱ میلی‌ثانیه‌ای) حدود ۲۰ میلی‌ثانیه لازم است -- ناممکن. بنابراین
\acro{URLLC} \acro{MCS} مقاوم (کم‌کردن $p$)، شکاف‌های کوتاه و گاه تکرارهای کور را ترکیب می‌کند.
\paragraph{محدودیت} وقتی کانال در محوشدگی عمیق است خطاها مستقل نیستند؛ تنوع (بسامد، آنتن، چنداتصالی) لازم است تا فرض استقلال تقریباً برقرار
شود.

\subsection{زمان‌بندی اولویت‌دار}\label{sec:prio}
\paragraph{مدل} دو کلاس یک پیوند را که به‌صورت \lr{M/M/1} با نرخ خدمت $\mu$ مدل شده به اشتراک می‌گذارند؛ کلاس ۱ (بار $\rho_1$) نسبت به کلاس ۲
اولویت پیش‌دستانه دارد؛ بار کل $\rho$. میانگین زمان‌های اقامت عبارت‌اند از
\begin{equation}\label{eq:prio}
  T_1 = \frac{1}{\mu(1-\rho_1)}, \qquad
  T_2 = \frac{1}{\mu(1-\rho_1)(1-\rho)},
\end{equation}
در مقایسه با $T = 1/(\mu(1-\rho))$ برای \acro{FIFO}.
\paragraph{مثال} $\mu = 1000$ بسته در ثانیه و $\rho_1 = 0.2$: $T_1 = 1.25$ میلی‌ثانیه مستقل از بار کل؛ در $\rho = 0.9$، $T_2 = 12.5$ میلی‌ثانیه در
برابر ۱۰ میلی‌ثانیه برای \acro{FIFO} (\cref{fig:prio}).
\paragraph{تفسیر} اولویت کلاس پراولویت را کاملاً از بار کلاس‌های پایین‌تر محافظت می‌کند، با هزینه‌ای متوسط برای دیگران -- به شرط آنکه بار
خود کلاس پراولویت کوچک بماند. به همین دلیل جریان‌های \acro{GBR} و بحرانی از نظر تأخیر به کنترل پذیرش نیاز دارند: بدون آن، ترافیک
«پراولویت» بیش از حد، محافظت را از بین می‌برد.

\section{رویه‌های گام‌به‌گام}

\begin{procedurebox}{افزودن یک جریان \lr{QoS} از نوع \lr{GBR} برای یک تماس صوتی}\label{proc:gbr}
\begin{steps}
  \item \acro{IMS} (\acro{AF}) توصیف رسانهٔ تماس را به \acro{PCF} می‌فرستد (\lr{Npcf\_PolicyAuthorization}).
  \item \acro{PCF} یک قاعدهٔ \acro{PCC} (\acro{5QI} ۱، \lr{GFBR/MFBR} از کدک، \acro{ARP}) استخراج و به \acro{SMF} اطلاع می‌دهد.
  \item \acro{SMF} یک \acro{QFI} تخصیص می‌دهد، یک \acro{PDR} و \acro{QER} در \acro{UPF} نصب می‌کند و نمایهٔ \acro{QoS} را به \acro{gNB} و قاعدهٔ
        \acro{QoS} را از طریق \acro{AMF} به \acro{UE} می‌فرستد.
  \item \acro{gNB} جریان را می‌پذیرد، آن را با حالت \acro{RLC} و زمان‌بندی مناسب به یک \acro{DRB} نگاشت و تأیید می‌کند؛ اکنون بسته‌های صوتی رفتار
        \acro{GBR} دریافت می‌کنند.
  \item در پایان تماس \acro{PCF} قاعده را حذف می‌کند؛ \acro{SMF} جریان \acro{QoS} را در \acro{UPF}، \acro{RAN} و \acro{UE} آزاد می‌کند.
\end{steps}
\end{procedurebox}

\begin{procedurebox}{بررسی یک طراحی \lr{QoS} برای \lr{URLLC}}\label{proc:urllc}
\begin{steps}
  \item از \acro{5QI}، \acro{PDB}، \acro{PER} و \acro{MDBV} را بخوان؛ تأخیر شبکهٔ هسته را کم کن تا $D_{\mathrm{AN}}$ به دست آید (\cref{eq:pdb}).
  \item نرخ انفجار را از \acro{MDBV} (\cref{eq:mdbv}) و تعداد ارسال‌ها را از \acro{PER} و \acro{BLER} (\cref{eq:harq}) محاسبه کن.
  \item عددشناسی، شکاف‌های کوچک یا اعطاهای پیکربندی‌شده را چنان برگزین که همهٔ ارسال‌ها در $D_{\mathrm{AN}}$ جا شوند.
  \item بار پراولویت را بررسی کن: ترافیک پراولویت باید کسر کوچکی از ظرفیت بماند (\cref{eq:prio})؛ کنترل پذیرش را پیکربندی کن.
\end{steps}
\end{procedurebox}

\section{مثال‌های حل‌شده}

\begin{examplebox}{انتخاب \lr{5QI}ها}\label{ex:5qi}
\textbf{مسئله.} به این‌ها \acro{5QI} نسبت دهید: صدای \lr{VoNR}، سیگنالینگ \acro{SIP} در \acro{IMS}، یک تماس ویدئویی زنده، وب‌گردی، کنترل حرکت در
کارخانه.

\textbf{حل.} صدا ۱؛ \acro{SIP} ۵؛ ویدئوی زنده ۲ (یا ۷ اگر غیر\acro{GBR})؛ وب ۹ (پیش‌فرض) یا ۶/۸؛ کنترل حرکت یک مقدار \acro{GBR} بحرانی از نظر
تأخیر مانند ۸۲ (یا ۸۵ اگر ۵ میلی‌ثانیه لازم باشد)، با \acro{UPF} در لبه.
\end{examplebox}

\begin{examplebox}{امکان‌پذیری \lr{URLLC}}\label{ex:urllc}
\textbf{مسئله.} جریانی با \acro{5QI} ۸۵ انفجارهای ۲۰۰ بایتی حمل می‌کند. \acro{UPF} در محل است (تأخیر هسته ۰٫۵ میلی‌ثانیه). با \acro{SCS} ۳۰
کیلوهرتز (شکاف‌های ۰٫۵ میلی‌ثانیه‌ای)، رفت‌وبرگشت \acro{HARQ} چهار شکاف و $p = 0.01$، آیا \acro{PER} برآورده می‌شود؟

\textbf{حل.} $D_{\mathrm{AN}} = 4.5$ میلی‌ثانیه؛ نرخ انفجار $8\times200/0.0045 = 356$ کیلوبیت بر ثانیه؛ $n = \ln10^{-5}/\ln0.01 = 2.5 \to 3$ ارسال که
$3\times2 = 6$ میلی‌ثانیه نیاز دارند -- بیش از حد طولانی. گزینه‌ها: شکاف‌های کوچک یا \acro{SCS} ۱۲۰ کیلوهرتز (رفت‌وبرگشت \acro{HARQ} ۰٫۵
میلی‌ثانیه: در مجموع ۱٫۵ میلی‌ثانیه)، یا تکرارهای کور درون یک پنجرهٔ شکاف.
\end{examplebox}

\begin{examplebox}{محافظت اولویت}\label{ex:prio}
\textbf{مسئله.} با \cref{eq:prio}، اگر بار پراولویت از ۰٫۲ به ۰٫۶ برسد بر سر $T_1$ و در بار کل ۰٫۹ بر سر $T_2$ چه می‌آید؟

\textbf{حل.} $T_1 = 1/(1000\times0.4) = 2.5$ میلی‌ثانیه (دو برابر)؛ $T_2 = 1/(1000\times0.4\times0.1) = 25$ میلی‌ثانیه. خود ترافیک پراولویت باید محدود
شود.
\end{examplebox}

\section{نکته‌های کلیدی}
\begin{takeawaybox}
\begin{itemize}
  \item \acro{QoS} نسل پنجم جریان‌محور است: علامت‌گذاری \acro{QFI} برای هر بسته، یک تونل \lr{N3} برای هر نشست \acro{PDU}، \acro{SDAP} جریان‌ها را به
        \acro{DRB}ها نگاشت می‌کند.
  \item \acro{5QI} نوع منبع، اولویت، \acro{PDB}، \acro{PER} (و \acro{MDBV} برای \acro{GBR} بحرانی از نظر تأخیر) را می‌دهد؛ \acro{ARP} و \lr{GFBR/MFBR} نمایهٔ
        \acro{QoS} را کامل می‌کنند.
  \item \acro{PCF} $\to$ \acro{SMF} $\to$ (\acro{UPF} از طریق \lr{N4}، \acro{gNB} از طریق \lr{N2}، \acro{UE} از طریق \acro{NAS}): سیاست به قواعد اعمال‌پذیر
        تبدیل می‌شود.
  \item $D_{\mathrm{AN}} = \mathrm{PDB} - D_{\mathrm{CN}}$؛ نرخ انفجار $\ge 8\,\mathrm{MDBV}/D_{\mathrm{AN}}$؛ ارسال‌ها $n \ge \ln\mathrm{PER}/\ln p$.
  \item اولویت فقط در صورتی ترافیک پراولویت را محافظت می‌کند که بار خود آن محدود باشد: $T_1 = 1/(\mu(1-\rho_1))$.
\end{itemize}
\end{takeawaybox}

\section{تمرین‌ها}

\begin{exercisebox}{1}{جریان‌ها و حامل‌ها}\label{exr:flows}
یک \acro{UE} یک نشست \acro{PDU} اینترنت با سه جریان \acro{QoS} دارد. دست‌کم و حداکثر به چند تونل \lr{N3} و چند \acro{DRB} نیاز دارد؟
\end{exercisebox}

\begin{exercisebox}{1}{\lr{QoS} بازتابی}\label{exr:rq}
توضیح دهید چرا \acro{QoS} بازتابی سیگنالینگ را صرفه‌جویی می‌کند و یک وضعیت که نباید از آن استفاده کرد بیاورید.
\end{exercisebox}

\begin{exercisebox}{2}{تقسیم بودجه}\label{exr:pdb}
برای \acro{5QI} ۳ (\acro{PDB} ۵۰ میلی‌ثانیه) و \acro{5QI} ۸۲ (\acro{PDB} ۱۰ میلی‌ثانیه)، $D_{\mathrm{AN}}$ را وقتی \acro{UPF} در فاصلهٔ ۶۰۰ کیلومتر است
(۵ میکروثانیه بر کیلومتر به‌علاوهٔ ۲ میلی‌ثانیه پردازش) و وقتی محلی است (۰٫۵ میلی‌ثانیه) محاسبه کنید. کدام \acro{5QI} \acro{UPF} لبه را الزامی
می‌کند؟
\end{exercisebox}

\begin{exercisebox}{2}{\lr{HARQ}}\label{exr:harq}
جریانی به \acro{PER} $10^{-6}$ نیاز دارد. تعداد ارسال‌ها را برای $p = 0.1$ و $p = 0.001$ و زمان لازم را با رفت‌وبرگشت \acro{HARQ} ۱ میلی‌ثانیه
محاسبه کنید. این برای انتخاب \acro{MCS} چه معنایی دارد؟
\end{exercisebox}

\begin{exercisebox}{3}{کنترل پذیرش}\label{exr:adm}
یک سلول صدای \acro{5QI} ۱ (\acro{GBR} ۴۰ کیلوبیت بر ثانیه برای هر تماس در رادیو، با فرض هزینهٔ ۰٫۵٪ ظرفیت سلول برای هر تماس) حمل می‌کند. با
\cref{eq:prio} و صدا به‌عنوان کلاس ۱، چند تماس می‌توان پذیرفت تا ترافیک غیر\acro{GBR} در بار کل ۰٫۹ تأخیری حداکثر دو برابر مقدار \acro{FIFO} خود
ببیند؟ تفسیر کنید.
\end{exercisebox}

\begin{exercisebox}{3}{\lr{QoS} سرتاسری}\label{exr:e2e}
یک کاربرد بازی ابری به \lr{RTT} ۳۰ میلی‌ثانیه و ۵۰ مگابیت بر ثانیه فروسو نیاز دارد. \acro{5QI}(ها)، جای \acro{UPF}، اقدامات برش و انتقال را پیشنهاد
کنید و توضیح دهید کدام گره‌ها کدام پارامتر را اعمال می‌کنند.
\end{exercisebox}

\subsection*{پاسخ‌ها}
\addcontentsline{toc}{subsection}{پاسخ‌ها}

\begin{solutionbox}{\ref{exr:flows}}
در همهٔ حالت‌ها یک تونل \lr{N3} (برای هر نشست \acro{PDU}). \acro{DRB}ها: دست‌کم یکی (هر سه جریان به یک \acro{DRB} نگاشت شوند) و حداکثر سه (یکی برای
هر جریان)؛ \acro{gNB} تصمیم می‌گیرد.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:rq}}
\acro{UE} قواعد فراسو را از بسته‌های فروسوی علامت‌خورده با \acro{RQI} استخراج می‌کند، پس برای هر جریان سیگنالینگ \acro{NAS} لازم نیست -- برای
جریان‌های کوتاه‌عمر بسیار ارزشمند است. نباید جایی استفاده شود که رفتار فراسو باید با آنچه بستهٔ فروسو القا می‌کند متفاوت باشد، یا برای سیاست
حساس امنیتی که در آن شبکه باید دقیقاً کنترل کند کدام بسته‌های فراسو رفتار ممتاز بگیرند (یک قاعدهٔ بازتابی ممکن است با ترافیک فروسوی ساختگی
سوءاستفاده شود).
\end{solutionbox}

\begin{solutionbox}{\ref{exr:pdb}}
\acro{UPF} دور: $600\times5\,\mu\mathrm{s} = 3$ میلی‌ثانیه + ۲ میلی‌ثانیه = ۵ میلی‌ثانیه. \acro{5QI} ۳: $50 - 5 = 45$ میلی‌ثانیه (دور) یا ۴۹٫۵
میلی‌ثانیه (محلی)؛ \acro{5QI} ۸۲: $10 - 5 = 5$ میلی‌ثانیه (دور) یا ۹٫۵ میلی‌ثانیه (محلی). \acro{5QI} ۸۲ با \acro{UPF} دور نیمی از بودجهٔ دسترسی خود را
از دست می‌دهد -- به \acro{UPF} لبه نیاز دارد؛ \acro{5QI} ۳ نه.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:harq}}
$p = 0.1$: $n = 6$، حدود ۶ میلی‌ثانیه؛ $p = 0.001$: $n = 2$، حدود ۲ میلی‌ثانیه. یک \acro{MCS} محافظه‌کارانه (\acro{BLER} کم) بازدهی طیفی را هزینه
می‌کند اما ارسال مجدد و تأخیر را صرفه‌جویی می‌کند -- معاوضهٔ مشخصهٔ \acro{URLLC}: ظرفیت در برابر تأخیر و قابلیت اطمینان.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:adm}}
نیازمندی: $T_2 \le 2T_{\mathrm{FIFO}} \Leftrightarrow 1/(1-\rho_1) \le 2 \Leftrightarrow \rho_1 \le 0.5$: حداکثر ۱۰۰ تماس (هر کدام ۰٫۵٪). در عمل اپراتورها
بار \acro{GBR} را بسیار کمتر محدود می‌کنند (مثلاً ۲۰ تا ۳۰٪) تا تأخیر غیر\acro{GBR} نزدیک \acro{FIFO} بماند و برای تغییرات کانال که هزینهٔ هر تماس را
تغییر می‌دهند حاشیه بماند.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:e2e}}
\acro{5QI}: یک مقدار \acro{GBR} نوع ویدئو (مثلاً ۳ با ویژگی‌های بازی بی‌درنگ) برای جریان بازی با \acro{GFBR} برابر نرخ لازم، یا یک \acro{5QI} غیر\acro{GBR}
پراولویت با \lr{Session-AMBR} کافی؛ \acro{5QI} ۵/پیش‌فرض برای کنترل. \acro{UPF} در یک لبهٔ منطقه‌ای تا بخش هستهٔ \lr{RTT} چند میلی‌ثانیه باشد؛
احتمالاً یک برش \lr{eMBB}/بازی با سهم منابع \acro{RAN} در سلول‌های پرترافیک؛ نگاشت جریان به یک کلاس اولویت در انتقال. اعمال: \acro{UPF}
(طبقه‌بندی، \acro{MFBR}، \acro{AMBR})، \acro{gNB} (\acro{GFBR}، اولویت زمان‌بندی، \acro{PDB})، \acro{UE} (قواعد فراسو)؛ \acro{PCF} آن را بر اساس
درخواست \acro{AF} فراهم‌کنندهٔ بازی مجاز می‌کند.
\end{solutionbox}
