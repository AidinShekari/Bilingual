\section{اهداف یادگیری}
پس از مطالعهٔ این درس باید بتوانید:
\begin{objectives}
  \item \acro{HSS} را به‌عنوان اجتماع کارکردهای \acro{HLR}، \acro{AuC} و پایگاه دادهٔ کاربران \acro{IMS} \textbf{توصیف کنید} و
        واسط‌های آن را \textbf{فهرست کنید}.
  \item مدل دادهٔ اشتراک در \acro{EPS} و نحوهٔ استفادهٔ \acro{MME} از آن را \textbf{توضیح دهید}.
  \item تولید بردار احراز هویت در \acro{AuC} را \textbf{دنبال کنید} و مدیریت شمارهٔ ترتیب و همگام‌سازی مجدد را \textbf{توضیح دهید}.
  \item حافظهٔ \acro{HSS} و بار درخواست بردار را \textbf{اندازه‌گذاری کنید} و معاوضهٔ درخواست چند بردار در یک بار را
        \textbf{ارزیابی کنید}.
  \item دسترس‌پذیری استقرارهای \acro{HSS} با افزونگی جغرافیایی را \textbf{محاسبه کنید}.
  \item \acro{HSS} را با \lr{UDM/AUSF/UDR} در نسل پنجم و با معماری‌های لایه‌بندی‌شدهٔ داده (\acro{UDC}) \textbf{مرتبط کنید}.
\end{objectives}

\section{دانش پیش‌نیاز}
این درس بر رویهٔ \lr{EPS-AKA} و سلسله‌مراتب کلید درس ۹ و بر کاربرد \lr{S6a} درس ۱۰ بنا شده است. باید بدانید رمز
قالبی و کد احراز پیام چیستند و بتوانید احتمال رویدادهای مستقل را (برای مدل‌های دسترس‌پذیری) محاسبه کنید.

\paragraph{واژه‌های کلیدی} واژگان این درس همراه با معادل انگلیسی آن‌ها:
\begin{keyterms}
کارساز مشترکین خانگی & \lr{HSS} & پایگاه دادهٔ اصلی مشترکین در شبکهٔ \acro{3GPP} (\acro{HLR} + \acro{AuC} + دادهٔ \acro{IMS}). \\
ثبت‌کنندهٔ موقعیت خانگی & \lr{HLR} & پایگاه دادهٔ مشترکین نسل ۲ و ۳ که از طریق \acro{MAP} در دسترس است. \\
مرکز احراز هویت & \lr{AuC} & کلید محرمانهٔ $K$ را نگه می‌دارد و بردارهای احراز هویت را تولید می‌کند. \\
شناسهٔ بین‌المللی مشترک موبایل & \lr{IMSI} & شناسهٔ دائمی مشترک (\lr{MCC + MNC + MSIN}). \\
بردار احراز هویت & \lr{Authentication vector} & چهارتایی (\lr{RAND, XRES, AUTN}، $K_{\mathrm{ASME}}$) برای یک اجرای \lr{EPS-AKA}. \\
شمارهٔ ترتیب & \lr{SQN} & شمارندهٔ ۴۸ بیتی که تازگی بردار را اثبات می‌کند. \\
نشانهٔ همگام‌سازی مجدد & \lr{AUTS} & نشانه‌ای که \acro{USIM} برای همگام‌سازی دوبارهٔ \acro{SQN} برمی‌گرداند. \\
کلید پیکربندی اپراتور & \lr{OPc} & مقدار وابسته به اپراتور که از \lr{OP} و $K$ برای \lr{MILENAGE} استخراج می‌شود. \\
همگرایی دادهٔ کاربر / مخزن & \lr{UDC / UDR} & معماری‌ای که منطق کاربرد را از یک مخزن دادهٔ مشترک جدا می‌کند. \\
افزونگی جغرافیایی & \lr{Geo-redundancy} & تکثیر یک کارکرد در سایت‌های دور از هم. \\
\end{keyterms}

\section{مقدمه و انگیزه}
شبکهٔ موبایل، در کنار چیزهای دیگر، یک پایگاه دادهٔ بسیار بزرگ و بسیار دسترس‌پذیر است. هر مشترک یک هویت دائمی، یک
کلید محرمانه که فقط با سیم‌کارت مشترک است، یک نمایه از خدمات مجاز و یک موقعیت فعلی دارد؛ هر اتصال، بسیاری از
تحویل‌ها و هر تماس ورودی برخی از این‌ها را می‌خوانند یا می‌نویسند. در نسل ۲ و ۳ این داده‌ها در ثبت‌کنندهٔ موقعیت
خانگی (\acro{HLR}) و مرکز احراز هویت (\acro{AuC}) قرار داشتند؛ با \acro{IMS} (انتشار ۵)، \acro{3GPP} \emph{کارساز
مشترکین خانگی} (\acro{HSS}) را معرفی کرد که کارکردهای \acro{HLR} و \acro{AuC} را دربر دارد و دادهٔ کاربر \acro{IMS} را
به آن می‌افزاید \cite{ts23002}. در \acro{EPS}، \acro{HSS} تنها مرجع مشترکین برای \acro{MME} است و از طریق
\lr{Diameter S6a} (درس ۱۰) در دسترس است.

\acro{HSS} در کتاب‌های درسی کمتر بحث می‌شود چون بسته‌ای جابه‌جا نمی‌کند، اما قطعی \acro{HSS} هر اتصال جدید را در
یک کشور متوقف می‌کند. پرسش‌های طراحی آن -- چگونه $10^7$ تا $10^8$ نمایه را در چند سایت سازگار نگه داریم، چگونه
بردارهای احراز هویت تازه را با نرخ هزاران در ثانیه تولید کنیم، چگونه پس از خرابی شمارهٔ ترتیب را بازیابی کنیم -- نمونهٔ
نوعی سامانه‌های توزیع‌شدهٔ سطح اپراتور است.

\section{مفاهیم بنیادی}

\subsection{کارکردهای \lr{HSS}}
\begin{itemize}
  \item \textbf{شناسایی و نشانی‌دهی}: \acro{IMSI}، \acro{MSISDN}، هویت‌های عمومی و خصوصی \acro{IMS} \cite{ts23003}.
  \item \textbf{احراز هویت (\acro{AuC})}: $K$ (و \lr{OPc}) را نگه می‌دارد و بردارهای احراز هویت را تولید می‌کند
        \cite{ts33102,ts33401}.
  \item \textbf{مدیریت اشتراک}: نمایه -- \acro{APN}های مجاز، کیفیت خدمات، محدودیت‌های رومینگ و دسترسی -- را نگه
        می‌دارد و تغییرات را به گرهٔ خدمت‌دهنده می‌فرستد.
  \item \textbf{مدیریت تحرک}: \lr{MME/SGSN/MSC} خدمت‌دهنده را ثبت می‌کند تا شبکه بتواند به مشترک برسد، و هنگام
        تغییر گرهٔ خدمت‌دهنده ثبت قبلی را لغو می‌کند.
  \item \textbf{پشتیبانی از \acro{IMS}}: \lr{S-CSCF} خدمت‌دهنده، نمایه‌های خدمت \acro{IMS} و معیارهای فیلتر اولیه را نگه
        می‌دارد و به کارسازهای کاربرد پاسخ می‌دهد \cite{ts23228}.
\end{itemize}

\subsection{دادهٔ اشتراک}
\begin{definitionbox}{نمایهٔ اشتراک \lr{EPS}}
نمایه‌ای که \acro{HSS} در \lr{ULA} (و بعدها در \lr{IDR}) می‌فرستد، از جمله شامل وضعیت مشترک، \acro{MSISDN}، حالت
دسترسی به شبکه، ممنوعیت‌های تعیین‌شده از سوی اپراتور، \lr{UE-AMBR} مشترک‌شده، دادهٔ محدودیت دسترسی (مثلاً «بدون
\lr{E-UTRAN}»)، زمان‌سنج \acro{TAU} دوره‌ای و یک \emph{نمایهٔ پیکربندی \acro{APN}} است: فهرستی از پیکربندی‌های
\acro{APN} که هر یک شناسهٔ زمینه، نام \acro{APN} (\lr{Service-Selection})، نوع \acro{PDN}، \acro{QCI} و \acro{ARP}
حامل پیش‌فرض و \lr{APN-AMBR} را دارد \cite{ts29272}.
\end{definitionbox}
\acro{MME} تا زمانی که مشترک نزد آن ثبت است نمایه را در حافظهٔ نهان نگه می‌دارد. \acro{HSS} نسخهٔ اصلی باقی می‌ماند: نسخهٔ
نهان را با \lr{Insert-Subscriber-Data} (\lr{IDR}) یا \lr{Delete-Subscriber-Data} (\lr{DSR}) تغییر می‌دهد و با
\lr{Cancel-Location} (\lr{CLR}) پس می‌گیرد.

\section{توضیحات فنی تفصیلی}

\subsection{تولید بردار احراز هویت}
برای هر بردار، \acro{AuC} یک \acro{RAND} تصادفی برمی‌گزیند، شمارهٔ ترتیب بعدی \acro{SQN} را برمی‌دارد و پنج تابع از
$K$، \acro{RAND}، \acro{SQN} و \acro{AMF} را محاسبه می‌کند (\cref{fig:av}): $f_1$ کد احراز شبکه \acro{MAC}، $f_2$ پاسخ
مورد انتظار \acro{XRES}، $f_3$ و $f_4$ کلیدهای رمز و یکپارچگی \acro{CK} و \acro{IK} و $f_5$ کلید گمنامی \acro{AK} را می‌دهد که
\acro{SQN} را پنهان می‌کند \cite{ts33102}. مجموعهٔ الگوریتم توصیه‌شده، \lr{MILENAGE}، هر پنج تابع را از یک رمز قالبی
۱۲۸ بیتی (\acro{AES}) و ثابت اپراتور \lr{OPc} می‌سازد \cite{ts35206}. برای \lr{E-UTRAN}، \acro{HSS} \emph{بیت جداسازی
\acro{AMF}} را برابر ۱ قرار می‌دهد و $K_{\mathrm{ASME}}$ را از \acro{CK} و \acro{IK} استخراج می‌کند که بردار را به شبکهٔ
خدمت‌دهنده گره می‌زند \cite{ts33401}. بردار عبارت است از
\begin{equation}\label{eq:autn}
  \mathrm{AV} = \bigl(\mathrm{RAND},\ \mathrm{XRES},\ \mathrm{AUTN},\ K_{\mathrm{ASME}}\bigr),\qquad
  \mathrm{AUTN} = (\mathrm{SQN}\xor\mathrm{AK})\concat\mathrm{AMF}\concat\mathrm{MAC}.
\end{equation}
کلید محرمانهٔ $K$ هرگز از \acro{AuC} خارج نمی‌شود؛ \acro{MME} فقط بردارها را دریافت می‌کند.

\subsection{شماره‌های ترتیب و تازگی}
\acro{USIM} فقط در صورتی \acro{AUTN} را می‌پذیرد که \acro{MAC} درست \emph{و} \acro{SQN} تازه باشد. \lr{TS 33.102} در یک
پیوست آگاهی‌بخش طرحی را توصیف می‌کند که در آن \acro{SQN} به یک شمارندهٔ ۴۳ بیتی \acro{SEQ} و یک شاخص ۵ بیتی
\acro{IND} تقسیم می‌شود ($\mathrm{SQN} = \mathrm{SEQ}\concat\mathrm{IND}$) \cite{ts33102}. \acro{USIM} آرایه‌ای از
بزرگ‌ترین \acro{SEQ} پذیرفته‌شده برای هر یک از $2^5 = 32$ مقدار \acro{IND} نگه می‌دارد و \acro{SQN} جدید را در صورتی
می‌پذیرد که \acro{SEQ} آن از مقدار ذخیره‌شده در خانهٔ \acro{IND} بزرگ‌تر باشد. این آرایه اجازه می‌دهد بردارهایی که با
یک ترتیب تولید شده‌اند با ترتیب دیگری استفاده شوند -- مثلاً وقتی دو \acro{MME} دسته‌هایی از بردارهای یک مشترک را نگه
می‌دارند -- بدون شکست‌های کاذب. یک بررسی سن اختیاری نیز مقادیر \acro{SQN} را که بیش از حد از بزرگ‌ترین مقدار
پذیرفته‌شده عقب‌ترند رد می‌کند.

اگر \acro{SQN} تازه نباشد، \acro{USIM} با یک \emph{شکست همگامی} پاسخ می‌دهد که \acro{AUTS} را حمل می‌کند؛ \acro{AUTS}
شامل \acro{SQN} خود \acro{USIM} است که با کلیدی مشابه \acro{AK} پنهان و با یک \acro{MAC} (\lr{MAC-S}) محافظت شده است.
\acro{MME} مقادیر \acro{RAND} و \acro{AUTS} را در یک \lr{AIR} جدید (\lr{Re-Synchronization-Info}) به \acro{HSS}
می‌فرستد؛ \acro{AuC} \lr{MAC-S} را بررسی می‌کند، شمارندهٔ خود را برابر مقدار \acro{USIM} قرار می‌دهد و بردارهای تازه
برمی‌گرداند (\cref{fig:resync}).

\subsection{مدیریت تحرک در \lr{HSS}}
در هر \lr{ULR}، \acro{HSS} \acro{MME} خدمت‌دهنده (\lr{Origin-Host} و قلمرو) را ثبت می‌کند. اگر \acro{MME} دیگری ثبت
شده بود، \acro{HSS} به آن \lr{CLR} می‌فرستد تا زمینهٔ خود را حذف کند؛ این تضمین می‌کند که هر مشترک یک گرهٔ
خدمت‌دهنده دارد. \acro{MME}ای که یک مشترک مدت‌ها غیرفعال را حذف می‌کند \lr{Purge-UE} (\lr{PUR}) می‌فرستد و پس از آن
\acro{HSS} مشترک را پاک‌شده علامت می‌زند. پس از راه‌اندازی مجدد، \acro{HSS}ای که ممکن است ثبت‌ها را از دست داده باشد
می‌تواند \lr{Reset} (\lr{RSR}) را به \acro{MME}ها بفرستد و آن‌ها مشترکین متأثر را در تماس بعدی تازه‌سازی می‌کنند
\cite{ts29272}.

\subsection{پیاده‌سازی: بخش‌های جلویی، مخزن داده و \lr{UDC}}
\acro{HSS}های تجاری \emph{بخش‌های جلویی} بی‌حالت را که \acro{MAP} و \lr{Diameter} را خاتمه می‌دهند و منطق کاربرد را
اجرا می‌کنند، از یک \emph{مخزن داده}ٔ تکثیرشده و عمدتاً درون‌حافظه‌ای جدا می‌کنند. \acro{3GPP} این را به‌صورت
\emph{همگرایی دادهٔ کاربر} (\acro{UDC}) تعمیم داد: \lr{HLR/HSS}، \acro{AuC} و دیگر بخش‌های جلویی یک مخزن دادهٔ کاربر
(\acro{UDR}) را از طریق واسط \lr{Ud} به اشتراک می‌گذارند، به‌طوری که هر مشترک به‌جای یک نسخه در هر گره فقط یک رکورد
دارد (\cref{fig:arch}). هستهٔ نسل پنجم همین تقسیم را حفظ می‌کند: \acro{UDM} و \acro{AUSF} بخش‌های جلویی‌اند و
\acro{UDR} مخزن است (درس‌های ۱۸ و ۱۹). چون \acro{HSS} یک نقطهٔ کنترل واحد است، مخزن داده در سایت‌های جغرافیایی
جدا از هم تکثیر می‌شود.

\section{مدل‌های ریاضی}

\subsection{طول عمر شمارهٔ ترتیب}\label{sec:sqn}
\paragraph{مدل} با یک \acro{SEQ} با $b$ بیت و یک احراز هویت در هر $\tau$ ثانیه، شمارنده به مدت زیر دوام می‌آورد:
\begin{equation}\label{eq:sqnlife}
  T_{\mathrm{life}} = 2^{b}\,\tau .
\end{equation}
\paragraph{مثال} $b = 43$ و یک احراز هویت در دقیقه: $T_{\mathrm{life}} = 8.8\times10^{12}$ دقیقه $\approx 1.7\times10^{7}$
سال. تمام‌شدن شمارنده نگرانی نیست؛ شمارنده بزرگ است تا طرح‌هایی که مقادیری را جا می‌اندازند (مثلاً \acro{SEQ}
مبتنی بر ساعت) ممکن بمانند.

\subsection{حافظه}\label{sec:storage}
\paragraph{مدل} $N$ مشترک با اندازهٔ میانگین نمایهٔ $s$، با $R$ بار تکثیر و ضریب پرشدگی $\eta$ (کسری از حافظه که
می‌توان استفاده کرد):
\begin{equation}\label{eq:storage}
  M = \frac{N\,s\,R}{\eta}.
\end{equation}
\paragraph{مثال} $N = 10^7$، $s = 4$ کیلوبایت، $R = 3$ و $\eta = 1$: $M = 120$ گیگابایت -- آن‌قدر کوچک که روی چند
کارساز در حافظه نگه داشته شود. بنابراین حافظه مسئلهٔ دشوار نیست؛ سازگاری و دسترس‌پذیری دشوارند.

\subsection{درخواست‌های بردار}\label{sec:air}
\paragraph{مدل} اگر احراز هویت‌ها با نرخ $\lambda_{\mathrm{auth}}$ رخ دهند و \acro{MME} در هر \lr{AIR} تعداد $n$ بردار
درخواست کند و همه را مصرف کند، نرخ \lr{AIR} برابر است با
\begin{equation}\label{eq:air}
  \lambda_{\mathrm{AIR}} = \frac{\lambda_{\mathrm{auth}}}{n}.
\end{equation}
\paragraph{مثال} $10^7$ مشترک که روزانه ۱۰ بار احراز هویت می‌شوند: $\lambda_{\mathrm{auth}} = 1157$\,s$^{-1}$؛ با $n = 5$،
$\lambda_{\mathrm{AIR}} = 231$\,s$^{-1}$. \paragraph{محدودیت} بردارهایی که هنگام رفتن مشترک به \acro{MME} دیگر استفاده‌نشده
می‌مانند هدر می‌روند و بردارهای ذخیره‌شده حافظهٔ \acro{MME} را مصرف می‌کنند؛ بنابراین $n$ معمولاً کوچک است
(\cref{exr:batch}).

\subsection{دسترس‌پذیری سایت‌های افزونه}\label{sec:avail}
\paragraph{مدل} هر سایت با احتمال $a$ (عدم دسترس‌پذیری $u = 1-a$) و مستقل از دیگران در دسترس است. اگر خدمت دست‌کم
به $k$ سایت از $N$ سایت نیاز داشته باشد،
\begin{equation}\label{eq:avail}
  A_{k\text{-of-}N} = \sum_{i=k}^{N}\binom{N}{i}a^{i}(1-a)^{N-i}.
\end{equation}
حالت‌های خاص: $A_{1\text{-of-}2} = 1-u^2$؛ $A_{2\text{-of-}3} = 1-(3u^2-2u^3)$؛ $A_{2\text{-of-}2} = a^2$.
\paragraph{مثال} $a = 0.999$ (۸٫۷۶ ساعت قطعی در سال): دو سایت که هر کدام می‌توانند کل بار را حمل کنند $u^2 = 10^{-6}$،
یعنی ۳۱٫۵ ثانیه در سال، می‌دهند؛ دو سایت که هر دو لازم‌اند ۱۷٫۵ ساعت در سال می‌دهند (\cref{fig:avail}).
\paragraph{فرض‌ها و تفسیر} استقلال فرض حیاتی است: اشکال‌های نرم‌افزاری، خطاهای پیکربندی و خطاهای تکثیر همهٔ سایت‌ها را
هم‌زمان متأثر می‌کنند و در عمل بر قطعی‌های \acro{HSS} غالب‌اند. مدل یک کران بالا برای آنچه افزونگی می‌تواند به دست
دهد ارائه می‌کند، نه یک پیش‌بینی.

\section{رویه‌های گام‌به‌گام}

\begin{procedurebox}{تولید یک بردار \lr{E-UTRAN} در \lr{AuC}}\label{proc:av}
\begin{steps}
  \item \lr{AIR} را با \acro{IMSI} و هویت شبکهٔ خدمت‌دهنده (\lr{Visited-PLMN-Id}) دریافت کن.
  \item $K$ و \lr{OPc} را پیدا کن؛ یک \acro{RAND} تازهٔ ۱۲۸ بیتی بساز؛ \acro{SQN} بعدی (\acro{SEQ} و \acro{IND}) را بردار.
  \item \acro{MAC} $= f_1$، \acro{XRES} $= f_2$، \acro{CK} $= f_3$، \acro{IK} $= f_4$ و \acro{AK} $= f_5$ را با بیت جداسازی
        \acro{AMF} = ۱ محاسبه کن.
  \item \acro{AUTN} را با \cref{eq:autn} بساز؛ $K_{\mathrm{ASME}}$ را از \acro{CK}، \acro{IK}، هویت شبکهٔ خدمت‌دهنده و
        $\mathrm{SQN}\xor\mathrm{AK}$ استخراج کن.
  \item بردارها را در \lr{AIA} برگردان؛ \acro{HSS} \acro{CK} و \acro{IK} را دور می‌اندازد.
\end{steps}
\end{procedurebox}

\begin{procedurebox}{همگام‌سازی مجدد شمارهٔ ترتیب}\label{proc:resync}
\begin{steps}
  \item \acro{USIM} \acro{MAC} را بررسی می‌کند، \acro{SQN} را بازیابی می‌کند و آن را تازه نمی‌یابد؛ \acro{AUTS} را محاسبه
        می‌کند و \acro{UE} پیام \lr{Authentication Failure} (شکست همگامی) را می‌فرستد.
  \item \acro{MME} بردارهای باقی‌ماندهٔ این مشترک را دور می‌اندازد و \lr{AIR} را با
        $\text{Re-Synchronization-Info} = \mathrm{RAND}\concat\mathrm{AUTS}$ می‌فرستد.
  \item \acro{AuC} \lr{MAC-S} را بررسی می‌کند، \acro{SQN} خود را برابر مقدار \acro{USIM} قرار می‌دهد و بردارهای جدید
        تولید می‌کند.
  \item \acro{MME} احراز هویت را با یک بردار جدید از نو آغاز می‌کند؛ این بار موفق می‌شود.
\end{steps}
\end{procedurebox}

\section{مثال‌های حل‌شده}

\begin{examplebox}{اندازهٔ یک \lr{AIA}}\label{ex:aia}
\textbf{مسئله.} اندازهٔ \acro{AVP} از نوع \lr{Authentication-Info} را که پنج بردار \lr{E-UTRAN} را حمل می‌کند محاسبه
کنید (همهٔ \acro{AVP}ها ویژهٔ فروشنده، بدون \lr{Item-Number} اختیاری).

\textbf{حل.} با قاعدهٔ طول \acro{AVP} درس ۱۰ و سرآیند ۱۲ بایتی: \acro{RAND} $12+16 = 28$، \acro{XRES} (۶۴ بیت)
$12+8 = 20$، \acro{AUTN} $12+16 = 28$ و $K_{\mathrm{ASME}}$ $12+32 = 44$. یک \lr{E-UTRAN-Vector}: $12 + 120 = 132$ بایت؛
پنج بردار در \lr{Authentication-Info} گروهی: $12 + 5\times132 = 672$ بایت.
\end{examplebox}

\begin{examplebox}{بازیابی از نسخهٔ پشتیبان}\label{ex:resync}
\textbf{مسئله.} یک \acro{HSS} با $10^7$ مشترک از یک نسخهٔ پشتیبان یک‌روزه بازیابی می‌شود، بنابراین شمارنده‌های
\acro{SQN} آن از شمارنده‌های \acro{USIM}ها عقب‌اند. اگر همهٔ مشترکین در ساعت بعد احراز هویت شوند، چه بار اضافه‌ای روی
\lr{S6a} ایجاد می‌شود؟

\textbf{حل.} نخستین احراز هویت هر مشترک با شکست همگامی روبه‌رو می‌شود و یک \lr{AIR} اضافه لازم دارد:
$10^7/3600 = 2778$ \lr{AIR} اضافه در ثانیه به مدت یک ساعت -- افزون بر ترافیک عادی و هم‌اندازهٔ آن. اپراتورها با تکثیر
پیوستهٔ به‌روزرسانی‌های \acro{SQN} به‌جای تکیه بر نسخهٔ پشتیبان، یا با جلو بردن شمارنده‌های بازیابی‌شده به اندازهٔ یک
حاشیهٔ امن، از این وضعیت پرهیز می‌کنند.
\end{examplebox}

\begin{examplebox}{انتخاب طرح افزونگی}\label{ex:geo}
\textbf{مسئله.} سایت‌ها $a = 0.999$ دارند. یک سایت، دو سایت که هر کدام کل بار را حمل می‌کنند، و سه سایت که هر دوتای
آن‌ها می‌توانند بار را حمل کنند را مقایسه کنید.

\textbf{حل.} از \cref{eq:avail}: یک سایت $u = 10^{-3}$ (۸٫۷۶ ساعت در سال)؛ ۱ از ۲: $u^2 = 10^{-6}$ (۳۱٫۵ ثانیه در سال)؛
۲ از ۳: $3u^2-2u^3 \approx 3.0\times10^{-6}$ (۹۴٫۵ ثانیه در سال). طرح ۲ از ۳ سه برابر کم‌دسترس‌پذیرتر از ۱ از ۲ است، اما به
جای ۱۰۰٪ فقط ۵۰٪ ظرفیت در هر سایت لازم دارد، یعنی در مجموع ۱۵۰٪ به‌جای ۲۰۰٪.
\end{examplebox}

\section{نکته‌های کلیدی}
\begin{takeawaybox}
\begin{itemize}
  \item \acro{HSS} = \acro{HLR} + \acro{AuC} + دادهٔ کاربر \acro{IMS}؛ تنها مرجع مشترکین که از طریق \lr{S6a}، \lr{Cx}،
        \lr{Sh}، \lr{SWx} و \acro{MAP} در دسترس است.
  \item \acro{AuC} کلید $K$ را محرمانه نگه می‌دارد و بردارهای (\lr{RAND, XRES, AUTN}، $K_{\mathrm{ASME}}$) را با $f_1$ تا
        $f_5$ (\lr{MILENAGE}) صادر می‌کند.
  \item $\mathrm{SQN} = \mathrm{SEQ}\concat\mathrm{IND}$ تازگی را اثبات می‌کند؛ شکست‌ها با همگام‌سازی مجدد \acro{AUTS}
        ترمیم می‌شوند.
  \item \acro{HSS} \acro{MME} خدمت‌دهنده را ثبت می‌کند، ثبت‌های قدیمی را لغو می‌کند و تغییرات نمایه را می‌فرستد.
  \item حافظه کوچک است (حدود ۱۰۰ گیگابایت برای $10^7$ کاربر)؛ دسترس‌پذیری و سازگاری بخش‌های دشوارند که با افزونگی
        $k$ از $N$ و فرض کلیدی استقلال مدل می‌شوند.
  \item تقسیم بخش جلویی/مخزن (\acro{UDC}) مستقیماً به \lr{UDM/AUSF/UDR} در نسل پنجم می‌رسد.
\end{itemize}
\end{takeawaybox}

\begin{exercisebox}{1}{نسخهٔ اصلی و نهان}\label{exr:split}
کدام داده‌های مشترک به‌طور دائمی در \acro{HSS} ذخیره می‌شوند و کدام در \acro{MME} نهان می‌شوند؟ کدام پیام‌ها نسخهٔ نهان

\end{exercisebox}

\begin{exercisebox}{1}{قلمرو از روی \lr{IMSI}}\label{exr:realm}
برای \acro{IMSI} با مقدار \lr{432101234567890} و \acro{MNC} دو رقمی، قلمرو شبکهٔ خانگی \acro{EPC} را که یک \acro{MME}
میزبان به‌عنوان \lr{Destination-Realm} به کار می‌برد بنویسید.
\end{exercisebox}

\begin{exercisebox}{2}{اندازه‌گذاری حافظه}\label{exr:storage}
حافظهٔ لازم برای $5\times10^7$ مشترک با نمایهٔ ۶ کیلوبایتی، سه نسخهٔ تکثیر و ضریب پرشدگی ۰٫۷ را محاسبه کنید.
\end{exercisebox}

\begin{exercisebox}{2}{چند بردار در هر درخواست؟}\label{exr:batch}
با $\lambda_{\mathrm{auth}} = 1157$\,s$^{-1}$، نرخ \lr{AIR} را برای $n = 1, 3, 5$ محاسبه کنید. اگر مشترک به‌طور میانگین در هر
حضور نزد یک \acro{MME} پیش از رفتن به \acro{MME} دیگر دو بردار مصرف کند، برای $n = 5$ چه کسری از بردارهای تولیدشده
هدر می‌رود؟
\end{exercisebox}

\begin{exercisebox}{3}{طراحی افزونگی}\label{exr:kofn}
عبارت‌های $A_{2\text{-of-}3}$ و $A_{1\text{-of-}2}$ را از \cref{eq:avail} استخراج کنید. سپس فرض کنید افزون بر خرابی‌های
مستقل، یک خرابی هم‌شکل (مثلاً یک انتشار نرم‌افزاری معیوب) با احتمال $p_c = 10^{-5}$ همهٔ سایت‌ها را از کار می‌اندازد.
زمان قطعی ۱ از ۲ را دوباره محاسبه و تفسیر کنید.
\end{exercisebox}

\begin{exercisebox}{3}{آرایهٔ \lr{IND}}\label{exr:ind}
\lr{MME A} یک بردار استفاده‌نشده با $\mathrm{IND} = 7$ دارد. \acro{HSS} به صدور بردار برای \lr{MME B} ادامه می‌دهد و
\acro{IND} را به‌صورت چرخشی افزایش می‌دهد. پس از صدور چند بردار جدید ممکن است بردار نزد \lr{MME A} غیرقابل پذیرش شود و
تحت چه شرطی واقعاً شکست می‌خورد؟
\end{exercisebox}

\subsection*{پاسخ‌ها}
\addcontentsline{toc}{subsection}{پاسخ‌ها}

\begin{solutionbox}{\ref{exr:split}}
دائمی در \acro{HSS}: \acro{IMSI}، \acro{MSISDN}، $K$/\lr{OPc} و \acro{SQN} (\acro{AuC})، نمایهٔ اشتراک و هویت \acro{MME}
خدمت‌دهنده. نهان در \acro{MME}: نمایه (از \lr{ULA})، بردارهای استفاده‌نشده و زمینهٔ امنیتی. سازگاری: \lr{IDR/DSR}
(تغییرات نمایه)، \lr{CLR} (پس‌گیری)، \lr{PUR} (حذف مشترک توسط \acro{MME})، \lr{RSR} (راه‌اندازی مجدد \acro{HSS}).
\end{solutionbox}

\begin{solutionbox}{\ref{exr:realm}}
$\mathrm{MCC} = 432$ و $\mathrm{MNC} = 10$ (که به سه رقم ۰۱۰ تکمیل می‌شود)، پس قلمرو عبارت است از
\[\texttt{epc.mnc010.mcc432.3gppnetwork.org}\]
\cite{ts23003}. طول \acro{MNC} در \acro{IMSI} کدگذاری نمی‌شود؛ \acro{MME} آن را از پیکربندی می‌گیرد.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:storage}}
$M = 5\times10^7\times6\times10^3\times3/0.7 = 1.29\times10^{12}$ بایت $\approx 1.3$ ترابایت -- توزیع‌شده روی یک خوشه،
همچنان در حافظه امکان‌پذیر است.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:batch}}
$\lambda_{\mathrm{AIR}} = 1157,\ 386,\ 231$\,s$^{-1}$. با $n=5$ هر حضور یک \lr{AIR} ایجاد می‌کند و دو بردار از پنج بردار را
مصرف می‌کند: سه بردار از پنج (۶۰٪) هدر می‌رود، پس \acro{AuC} $5/2 = 2.5$ برابر بردار لازم محاسبه می‌کند. با $n = 3$
یک \lr{AIR} در هر حضور و ۳۳٪ هدررفت. بهترین $n$ با تعداد معمول احراز هویت‌ها در هر حضور تطبیق دارد.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:kofn}}
$A_{1\text{-of-}2} = 2a(1-a) + a^2 = 1-(1-a)^2$. $A_{2\text{-of-}3} = 3a^2(1-a) + a^3$، پس $1-A = 1 - 3a^2 + 2a^3
= 3u^2 - 2u^3$ با $u = 1-a$. با خرابی هم‌شکل: $1-A \approx p_c + u^2 = 10^{-5} + 10^{-6} = 1.1\times10^{-5}$، یعنی ۳۴۷
ثانیه در سال به‌جای ۳۱٫۵ ثانیه. جملهٔ هم‌شکل با ضریب ده غالب است: فراتر از دو سایت، انتشار مرحله‌ای نرم‌افزار و انضباط
پیکربندی دسترس‌پذیری را بیش از سخت‌افزار اضافی بهبود می‌دهند.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:ind}}
\acro{IND} ۳۲ مقدار دارد، پس بردار بعدی با $\mathrm{IND} = 7$ (و \acro{SEQ} بزرگ‌تر) پس از ۳۲ بردار جدید صادر می‌شود.
بردار نزد \lr{MME A} فقط در صورتی شکست می‌خورد که آن بردار جدیدتر پیش از بردار \lr{MME A} توسط \acro{USIM}
\emph{استفاده} شود: در این صورت \acro{USIM} \acro{SEQ} بزرگ‌تر را در خانهٔ ۷ ذخیره و بردار قدیمی‌تر را رد می‌کند. بنابراین
تا ۳۱ بردار جدیدتر می‌توانند با هر ترتیبی استفاده شوند بی‌آنکه بر آن اثر بگذارند.
\end{solutionbox}
