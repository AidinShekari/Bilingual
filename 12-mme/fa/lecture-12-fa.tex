\section{اهداف یادگیری}
پس از مطالعهٔ این درس باید بتوانید:
\begin{objectives}
  \item کارکردها و نقاط مرجع \acro{MME} را \textbf{فهرست کنید} و \textbf{توضیح دهید} چرا یک گرهٔ صرفاً صفحهٔ کنترل است.
  \item یک \acro{GUTI} را \textbf{رمزگشایی کنید} و \textbf{توضیح دهید} هویت‌های موقت چگونه از حریم خصوصی محافظت و
        سیگنالینگ را درون یک مخزن \acro{MME} مسیریابی می‌کنند.
  \item رویهٔ اتصال اولیه را از ابتدا تا انتها \textbf{دنبال کنید}.
  \item راهبردهای فراخوانی را بر اساس هزینهٔ سیگنالینگ و تأخیر مورد انتظار \textbf{ارزیابی کنید}.
  \item یک مخزن \acro{MME} را با ضرایب وزنی و حاشیهٔ خرابی \textbf{اندازه‌گذاری کنید} و اضافه‌بار \acro{MME} را با یک مدل صف
        \textbf{تحلیل کنید}.
  \item \acro{MME} را با \lr{SGSN/VLR} نسل ۲ و ۳ و \acro{AMF} نسل ۵ \textbf{مقایسه کنید}.
\end{objectives}

\section{دانش پیش‌نیاز}
به معماری \acro{EPC} و حالت‌های \lr{EMM/ECM} درس ۹، رویه‌های \lr{S1AP} و \acro{RRC} درس ۶، طراحی ناحیهٔ مکانی درس ۴

پواسون با نرخ $\lambda$، خدمت نمایی با نرخ $\mu$ و یک خدمت‌دهنده.

\section{مقدمه و انگیزه}
در \acro{GPRS} و \acro{UMTS}، \acro{SGSN} همه‌کار مشترکین بسته‌ای را انجام می‌داد: احراز هویت، ردیابی، نگه‌داشتن
نشست‌ها و هدایت بسته‌ها. \acro{EPS} این گره را دو بخش کرد. \emph{موجودیت مدیریت تحرک} (\acro{MME}) کارکردهای کنترلی را
نگه می‌دارد و دروازهٔ خدمت‌دهنده بسته‌ها را \cite{ts23401}. این تقسیم به اپراتورها اجازه می‌دهد ظرفیت سیگنالینگ (که با
تعداد دستگاه‌ها و تحرک آن‌ها رشد می‌کند) را مستقل از ظرفیت صفحهٔ کاربر (که با حجم ترافیک رشد می‌کند) مقیاس دهند، و
نخستین گام جداسازی صفحهٔ کنترل و کاربر است که نسل پنجم آن را کامل می‌کند.

\acro{MME} همتای \acro{UE} در هسته است: هر پیام \acro{NAS} -- اتصال، به‌روزرسانی ناحیهٔ ردیابی، درخواست خدمت، درخواست
حامل -- در آن خاتمه می‌یابد. این درس بررسی می‌کند \acro{MME} چه چیزی ذخیره می‌کند، دستگاه‌های بیکار را چگونه می‌یابد،
\acro{MME}ها چگونه بار را در یک مخزن تقسیم می‌کنند، و وقتی سیگنالینگ از ظرفیت فراتر رود چه رخ می‌دهد.

\subsection{کارکردهای \lr{MME}}
\begin{itemize}
  \item \textbf{سیگنالینگ \acro{NAS} و امنیت}: رویه‌های \acro{EMM} و \acro{ESM}، احراز هویت با بردارهای \acro{HSS}، رمزنگاری و

  \item \textbf{دسترس‌پذیری و تحرک}: فهرست‌های ناحیهٔ ردیابی، فراخوانی \acro{UE}های بیکار، تحویل میان \acro{eNB}ها با جابه‌جایی

  \item \textbf{مدیریت نشست}: برقراری حامل پیش‌فرض و اختصاصی از طریق \lr{S-GW} (\lr{S11})، انتخاب \lr{S-GW} و \lr{P-GW}.

\subsection{هویت‌ها}
\acro{IMSI} دائمی تا حد امکان به‌ندرت روی هوا فرستاده می‌شود. پس از نخستین اتصال، \acro{MME} یک \emph{شناسهٔ موقت یکتای
سراسری} (\acro{GUTI}) تخصیص می‌دهد \cite{ts23003} (\cref{fig:guti}). بخش نخست آن، \acro{GUMMEI}، \acro{MME} تخصیص‌دهنده
را نام می‌برد (\acro{PLMN}، شناسهٔ ۱۶ بیتی گروه \acro{MME}، کد ۸ بیتی \acro{MME})؛ بخش پایانی آن، \lr{M-TMSI} ۳۲ بیتی،

فراخوانی و برقراری اتصال \acro{RRC} به کار می‌رود. تخصیص مجدد منظم \acro{GUTI} مانع از آن می‌شود که یک ناظر

\subsection{مخزن‌های \lr{MME}}
یک \acro{eNB} می‌تواند به چند \acro{MME} وصل شود (\lr{S1-flex}). \acro{MME}هایی که به یک مجموعه ناحیهٔ ردیابی خدمت
می‌دهند یک \emph{مخزن} می‌سازند. وقتی \acro{UE} نخستین بار متصل می‌شود، تابع انتخاب گرهٔ \acro{NAS} در \acro{eNB} یک
\acro{MME} را با وزن \emph{ظرفیت نسبی} که هر \acro{MME} در \lr{S1 Setup} اعلام کرده است برمی‌گزیند؛ پس از آن کد \acro{MME}
در \lr{S-TMSI} هر اتصال جدید آن \acro{UE} را به همان \acro{MME} هدایت می‌کند. مخزن‌ها تقسیم بار، تاب‌آوری (\acro{UE}های یک
\acro{MME} خراب نزد دیگران دوباره ثبت‌نام می‌کنند) و امکان خارج‌کردن آرام یک \acro{MME} از سرویس را فراهم می‌کنند.

\section{توضیحات فنی تفصیلی}

\subsection{زمینهٔ \lr{UE}}
برای هر \acro{UE} ثبت‌شده، \acro{MME} این‌ها را نگه می‌دارد: \acro{IMSI} و \acro{GUTI}، زمینهٔ امنیتی (کلیدها، شمارنده‌های
\acro{NAS}، الگوریتم‌ها)، نمایهٔ اشتراک از \acro{HSS}، فهرست \acro{TAI} و آخرین \acro{TAI} و \acro{eNB} شناخته‌شده، حالت
\lr{EMM/ECM}، \lr{S-GW} انتخاب‌شده و زمینه‌های حامل (\acro{EBI}، کیفیت خدمات، \acro{TEID}های \lr{S-GW}). در
\lr{ECM-CONNECTED} زمینه شناسه‌های \lr{S1AP} اتصال \acro{eNB} آن \acro{UE} را نیز دارد. همین حالت است که \acro{MME} را
\emph{لنگر} سیگنالینگ \acro{UE} می‌کند: از دست رفتن آن (مثلاً پس از خرابی \acro{MME}) \acro{UE} را به اتصال دوباره وادار
می‌کند.

\subsection{نواحی ردیابی و فراخوانی}
\acro{MME} به هر \acro{UE} فهرستی از نواحی ردیابی (فهرست \acro{TAI}) می‌دهد؛ \acro{UE} فقط وقتی وارد ناحیه‌ای خارج از
فهرست شود یا زمان‌سنج دوره‌ای \lr{T3412} (پیش‌فرض ۵۴ دقیقه) منقضی شود \acro{TAU} انجام می‌دهد \cite{ts24301}. وقتی دادهٔ
فروسو برای یک \acro{UE} بیکار می‌رسد، \lr{S-GW} پیام \lr{Downlink Data Notification} را می‌فرستد و \acro{MME} \acro{UE} را در
سلول‌های فهرست \acro{TAI} آن با \lr{S-TMSI} فراخوانی می‌کند. اپراتورها بار فراخوانی را با راهبردهای \emph{مرحله‌ای} --
نخست آخرین \acro{eNB} دیده‌شده، سپس آخرین \acro{TA}، سپس کل فهرست -- به بهای تأخیر بیشتر کاهش می‌دهند (\cref{sec:paging}).

اگر \acro{TAU} دوره‌ای نرسد، \acro{MME} \emph{زمان‌سنج دسترس‌پذیری موبایل} (کمی طولانی‌تر از \lr{T3412}) را آغاز می‌کند؛ با
انقضای آن، \acro{MME} فراخوانی \acro{UE} را متوقف می‌کند و پس از یک زمان‌سنج جداسازی ضمنی دیگر آن را جدا می‌کند.

\subsection{انتخاب دروازه}
هنگام اتصال، \acro{MME} یک \lr{S-GW} را که به ناحیهٔ ردیابی \acro{UE} خدمت می‌دهد و یک \lr{P-GW} را که به \acro{APN}
درخواستی خدمت می‌دهد برمی‌گزیند؛ معمولاً با پرس‌وجوهای \acro{DNS} روی نام‌هایی که از \acro{TAI} و \acro{APN} ساخته می‌شوند
و دروازه‌های نامزد را با اولویت و وزن برمی‌گردانند. \lr{P-GW} باید در طول عمر اتصال \acro{PDN} ثابت بماند (مالک نشانی
\acro{IP} است)؛ \lr{S-GW} ممکن است با تحرک تغییر کند (درس ۱۳).

\subsection{متعادل‌سازی بار، توزیع مجدد و کنترل اضافه‌بار}
در شرایط عادی \acro{eNB}ها \acro{UE}های جدید را به نسبت ظرفیت‌های نسبی توزیع می‌کنند. برای خالی‌کردن یک \acro{MME} (برای
نگهداری) یا توزیع مجدد، \acro{MME} \acro{UE}ها را با علت \emph{\acro{TAU} برای متعادل‌سازی بار لازم است} آزاد می‌کند:
\acro{UE} یک \acro{TAU} انجام می‌دهد و چون \lr{S-TMSI} آن ارسال نمی‌شود، \acro{eNB} یک \acro{MME} جدید برمی‌گزیند. وقتی یک
\acro{MME} دچار اضافه‌بار است، پیام \msg{Overload Start} از \lr{S1AP} را به \acro{eNB}های خود می‌فرستد و آن‌ها اتصال‌های
جدید \acro{RRC} از انواع منتخب (مثلاً همه به‌جز اضطراری و پایان‌یافته در موبایل) را تا \msg{Overload Stop} رد می‌کنند
\cite{ts36413}. زمان‌سنج‌های عقب‌نشینی \acro{NAS} و ممنوعیت کلاس دسترسی (درس ۸) این کار را در \acro{UE} تکمیل می‌کنند.

\section{مدل‌های ریاضی}

\subsection{فراخوانی مرحله‌ای}\label{sec:paging}
\paragraph{مدل} \acro{MME} در مراحل $s = 1,2,\dots$ فراخوانی می‌کند؛ مرحلهٔ $s$ تعداد $n_s$ سلول را پوشش می‌دهد و \acro{UE} را
با احتمال (غیرشرطی) $p_s$ می‌یابد، با $\sum_s p_s = 1$. هر مرحله یک زمان پاسخ فراخوانی $T_\mathrm{p}$ طول می‌کشد. تعداد
مورد انتظار سلول‌های فراخوانده‌شده و تأخیر مورد انتظار عبارت‌اند از
\begin{equation}\label{eq:paging}
  \E[M] = \sum_{s} n_s\Bigl(1-\sum_{j<s}p_j\Bigr), \qquad
  \E[D] = T_\mathrm{p}\sum_{s}\Bigl(1-\sum_{j<s}p_j\Bigr).
\end{equation}
\paragraph{مثال} سه مرحله: آخرین \acro{eNB} ($n_1 = 1$، $p_1 = 0.6$)، آخرین \acro{TA} ($n_2 = 20$، $p_2 = 0.3$)، فهرست
\acro{TAI} ($n_3 = 60$، $p_3 = 0.1$): $\E[M] = 1 + 0.4\times20 + 0.1\times60 = 15$ سلول، در مقایسه با ۶۰ برای فراخوانی
یکجا؛ $\E[D] = 1.5\,T_\mathrm{p}$ به‌جای $T_\mathrm{p}$.
\paragraph{تفسیر} مرحله‌بندی وقتی می‌ارزد که احتمال‌ها در مراحل نخست متمرکز باشند (کاربران ساکن، سلول‌های کوچک)، و برای
خدمات حساس به تأخیر مانند تماس صوتی ورودی جذاب نیست؛ برای آن‌ها اپراتورها اغلب کل فهرست را یکجا فراخوانی می‌کنند.

\subsection{وزن‌های مخزن و حاشیهٔ خرابی}\label{sec:pool}
\paragraph{مدل} سهم \acro{UE}های جدیدی که \acro{MME} شمارهٔ $i$ دریافت می‌کند
\begin{equation}\label{eq:weight}
  s_i = \frac{w_i}{\sum_j w_j}
\end{equation}
است، که $w_i$ ظرفیت نسبی آن است. اگر مخزنی از $m$ \acro{MME} یکسان باید از دست‌رفتن یکی را بدون اضافه‌بار تاب آورد،
هر \acro{MME} حداکثر تا مقدار زیر می‌تواند بارگذاری شود:
\begin{equation}\label{eq:pool}
  \rho_{\max} = \frac{m-1}{m}.
\end{equation}
\paragraph{مثال} وزن‌های ۱۰۰، ۱۰۰ و ۵۰ سهم‌های ۰٫۴، ۰٫۴ و ۰٫۲ را می‌دهند. مخزنی از چهار \acro{MME} یکسان می‌تواند در کار
عادی با ۷۵٪ بار کار کند؛ مخزنی از دو \acro{MME} فقط با ۵۰٪.

\subsection{تأخیر پردازش و اضافه‌بار}\label{sec:mm1}
\paragraph{مدل} پردازندهٔ سیگنالینگ \acro{MME} را یک صف \lr{M/M/1} با ظرفیت $\mu$ رویه بر ثانیه و بار ارائه‌شدهٔ $\lambda$
($\rho = \lambda/\mu < 1$) در نظر بگیرید. میانگین زمان پاسخ
\begin{equation}\label{eq:mm1}
  W = \frac{1}{\mu-\lambda} = \frac{1}{\mu(1-\rho)}
\end{equation}
است.
\paragraph{مثال} $\mu = 5000$\,s$^{-1}$: $W = 1$ میلی‌ثانیه در $\lambda = 4000$، ۱۰ میلی‌ثانیه در ۴۹۰۰ و ۱۰۰ میلی‌ثانیه در
۴۹۹۰ (\cref{fig:overload}).
\paragraph{تفسیر} تأخیر با $\rho\to1$ بی‌کران رشد می‌کند. وقتی زمان‌های پاسخ به زمان‌سنج‌های \acro{NAS} نزدیک شوند
(\lr{T3410} = ۱۵ ثانیه برای اتصال)، \acro{UE}ها دوباره ارسال می‌کنند، بار ارائه‌شده بیشتر بالا می‌رود و سامانه ممکن است
فرو بپاشد؛ بنابراین کنترل اضافه‌بار باید بسیار پیش از $\rho = 1$ وارد عمل شود. مدل \lr{M/M/1} برای ورودهای انفجاری یک
طوفان سیگنالینگ خوش‌بینانه است.

\section{رویه‌های گام‌به‌گام}

\begin{procedurebox}{اتصال اولیه (ساده‌شده)}\label{proc:attach}
\begin{steps}
  \item \acro{UE} پیام \lr{Attach Request} را همراه با \lr{PDN Connectivity Request} درون برقراری \acro{RRC} می‌فرستد؛
        \acro{eNB} یک \acro{MME} را برمی‌گزیند و پیام را در \lr{S1AP Initial UE Message} هدایت می‌کند.
  \item \acro{MME} \acro{UE} را شناسایی می‌کند (\acro{GUTI} یا \acro{IMSI})، \lr{EPS-AKA} را با یک بردار از \acro{HSS} اجرا
        و امنیت \acro{NAS} را فعال می‌کند.
  \item \acro{MME} نزد \acro{HSS} ثبت‌نام می‌کند (\lr{ULR/ULA}) و نمایهٔ اشتراک را دریافت می‌کند.
  \item \acro{MME} \lr{S-GW} و \lr{P-GW} را برمی‌گزیند و \lr{Create Session Request} را می‌فرستد؛ \lr{P-GW} نشانی
        \acro{IP} را تخصیص می‌دهد و با \acro{PCRF} سیاست را به دست می‌آورد؛ پاسخ‌ها نقطهٔ پایانی تونل فراسوی \lr{S-GW}
        را برمی‌گردانند.
  \item \acro{MME} پیام \lr{Initial Context Setup Request} را با \lr{Attach Accept} و حامل پیش‌فرض به \acro{eNB} می‌فرستد و
        \acro{eNB} حامل رادیویی را با \lr{RRC Connection Reconfiguration} پیکربندی می‌کند.
  \item \acro{UE} با \lr{Attach Complete} پاسخ می‌دهد؛ \acro{MME} پیام \lr{Modify Bearer Request} را با نقطهٔ پایانی تونل
        فروسوی \acro{eNB} به \lr{S-GW} می‌فرستد. اکنون حامل پیش‌فرض ترافیک را در هر دو جهت حمل می‌کند.
\end{steps}
\end{procedurebox}

\begin{procedurebox}{خالی‌کردن یک \lr{MME} برای نگهداری}\label{proc:rebalance}
\begin{steps}
  \item ظرفیت نسبی \acro{MME} را صفر کن (\lr{S1AP MME Configuration Update}) تا \acro{eNB}ها دیگر آن را برای \acro{UE}های
        جدید انتخاب نکنند.
  \item \acro{UE}های متصل را با علت \emph{\acro{TAU} برای متعادل‌سازی بار لازم است} آزاد کن؛ \acro{UE}های بیکار را فراخوانی
        کن یا تا تماس بعدی آن‌ها صبر کن و به همان روش آزادشان کن.
  \item هر \acro{UE} یک \acro{TAU} بدون مسیریابی \lr{S-TMSI} انجام می‌دهد؛ \acro{eNB} \acro{MME} دیگری را برمی‌گزیند که زمینه
        را از طریق \lr{S10} می‌گیرد و نزد \acro{HSS} ثبت‌نام می‌کند.
  \item وقتی هیچ زمینه‌ای باقی نماند، \acro{MME} را می‌توان از سرویس خارج کرد.
\end{steps}
\end{procedurebox}

\section{مثال‌های حل‌شده}

\begin{examplebox}{رمزگشایی یک \lr{GUTI}}\label{ex:guti}
\textbf{مسئله.} یک \acro{GUTI} دارای \acro{MCC} ۴۳۲، \acro{MNC} ۳۵، \acro{MMEGI} \lr{\texttt{0x8001}}، \acro{MMEC}
\lr{\texttt{0x02}} و \lr{M-TMSI} \lr{\texttt{0xC0A80001}} است. \acro{eNB} از کدام هویت‌ها استفاده می‌کند و کدام \acro{MME} به
\acro{UE} خدمت می‌دهد؟

\textbf{حل.} $\mathrm{GUMMEI} = (432, 35, 32769, 2)$: کد \acro{MME} برابر ۲ در گروه ۳۲۷۶۹ از \acro{PLMN} ۴۳۲-۳۵. \lr{S-TMSI}
برابر \lr{\texttt{0x02C0A80001}} (۴۰ بیت) است؛ فراخوانی و برقراری \acro{RRC} فقط همین را حمل می‌کنند. تابع انتخاب گرهٔ
\acro{NAS} در \acro{eNB}، \acro{MMEC} ۲ را به اتصال \lr{S1} متناظر در مخزن خود نگاشت می‌کند.
\end{examplebox}

\begin{examplebox}{هزینهٔ فراخوانی}\label{ex:paging}
\textbf{مسئله.} یک شهر در ساعت اوج $5\times10^5$ رویداد پایان‌یافته در موبایل دارد. پیام‌های فراخوانی در ساعت اوج را برای
فراخوانی یکجا روی ۶۰ سلول و راهبرد سه‌مرحله‌ای \cref{sec:paging} مقایسه کنید.

\textbf{حل.} یکجا: $5\times10^5\times60 = 3.0\times10^7$ فراخوانی سلولی در ساعت. مرحله‌ای: $5\times10^5\times15 = 7.5\times10^6$،
یعنی چهار برابر کمتر، با تأخیر میانگین $1.5\,T_\mathrm{p}$.
\end{examplebox}

\begin{examplebox}{اندازه‌گذاری مخزن}\label{ex:pool}
\textbf{مسئله.} یک منطقه $6\times10^6$ \acro{UE} ثبت‌شده دارد. هر \acro{MME} از $2\times10^6$ زمینه پشتیبانی می‌کند. اگر مخزن باید
یک خرابی را تاب آورد، چند \acro{MME} لازم است؟

\textbf{حل.} بدون افزونگی $\lceil 6/2\rceil = 3$. با تحمل یک خرابی، $m-1$ \acro{MME} باقی‌مانده باید همهٔ زمینه‌ها را نگه دارند:
$(m-1)\times2\times10^6 \ge 6\times10^6 \Rightarrow m = 4$، که هر یک در کار عادی با $6/8 = 75\,\%$ بارگذاری می‌شود، سازگار
با \cref{eq:pool}.
\end{examplebox}

\section{نکته‌های کلیدی}
\begin{takeawaybox}
\begin{itemize}
  \item \acro{MME} لنگر صفحهٔ کنترل \acro{EPC} است: \acro{NAS}، امنیت، تحرک، کنترل حامل، انتخاب دروازه؛ بدون دادهٔ کاربر.
  \item \lr{GUTI = GUMMEI + M-TMSI}؛ \lr{S-TMSI} سیگنالینگ را به \acro{MME} درست در مخزن هدایت می‌کند و از \acro{IMSI}
        محافظت می‌کند.
  \item اتصال، احراز هویت، ثبت نزد \acro{HSS}، ایجاد نشست و برقراری حامل رادیویی را ترکیب می‌کند.
  \item فراخوانی مرحله‌ای تأخیر را با سیگنالینگ معاوضه می‌کند: $\E[M] = \sum n_s(1-\sum_{j<s}p_j)$.
  \item مخزن‌ها بار را بر اساس وزن تقسیم می‌کنند؛ افزونگی $N+1$ بار عادی را به $(m-1)/m$ محدود می‌کند.
  \item زمان پاسخ با $\rho\to1$ منفجر می‌شود؛ کنترل اضافه‌بار باید زود وارد عمل شود.
\end{itemize}
\end{takeawaybox}

\section{تمرین‌ها}

\begin{exercisebox}{1}{صفحهٔ کنترل یا کاربر؟}\label{exr:split}
برای هر کارکرد مشخص کنید \acro{MME}، \lr{S-GW} یا \lr{P-GW} آن را انجام می‌دهد: (الف) فراخوانی؛ (ب) بافرکردن بسته‌های فروسو
برای یک \acro{UE} بیکار؛ (ج) تخصیص نشانی \acro{IP}؛ (د) رمزنگاری \acro{NAS}؛ (هـ) انتخاب \lr{S-GW}.
\end{exercisebox}

\begin{exercisebox}{1}{چرا هویت‌های موقت؟}\label{exr:guti}
دو دلیل بیاورید که چرا شبکه در فراخوانی و برقراری \acro{RRC} به‌جای \acro{IMSI} از \lr{GUTI/S-TMSI} استفاده می‌کند.
\end{exercisebox}

\begin{exercisebox}{2}{راهبرد فراخوانی}\label{exr:paging}
\cref{eq:paging} را برای $p = (0.4, 0.4, 0.2)$ و $n = (1, 30, 90)$ ارزیابی کنید. با فراخوانی یکجا روی ۹۰ سلول مقایسه کنید.
برای تماس‌های صوتی ورودی کدام راهبرد را به کار می‌برید؟
\end{exercisebox}

\begin{exercisebox}{2}{وزن‌ها پس از خرابی}\label{exr:weights}
مخزنی پنج \acro{MME} با ظرفیت‌های نسبی $2,2,1,1,1$ دارد (ظرفیت متناسب با وزن). سهم \acro{UE}های جدید را پیش و پس از خرابی
یک \acro{MME} با وزن ۲ و بیشینهٔ بهره‌وری در کار عادی را که هنوز اجازهٔ جذب این خرابی را می‌دهد محاسبه کنید.
\end{exercisebox}

\begin{exercisebox}{3}{اضافه‌بار}\label{exr:overload}
یک \acro{MME} با $\mu = 5000$ رویه بر ثانیه داریم. (الف) چه باری در مدل \lr{M/M/1} مقدار $W \le 10$ میلی‌ثانیه را حفظ
می‌کند؟ (ب) پس از یک قطعی برق منطقه‌ای، $2\times10^6$ \acro{UE} در ۶۰ ثانیه برای اتصال تلاش می‌کنند. بار ارائه‌شده را با
$\mu$ مقایسه کنید و توضیح دهید کدام سازوکارها سامانه را پایدار نگه می‌دارند.
\end{exercisebox}

\begin{exercisebox}{3}{انتخاب \lr{T3412}}\label{exr:t3412}
یک \acro{MME} $5\times10^6$ \acro{UE} بیکار دارد. نرخ \acro{TAU} دوره‌ای را برای \lr{T3412} برابر ۵۴ و ۱۲ دقیقه محاسبه کنید و
معاوضهٔ میان سیگنالینگ، باتری \acro{UE} و زمان لازم برای تشخیص \acro{UE}های دست‌نیافتنی را بررسی کنید. چرا دستگاه‌های
\acro{IoT} مقادیر بسیار طولانی‌تری به کار می‌برند؟
\end{exercisebox}

\subsection*{پاسخ‌ها}
\addcontentsline{toc}{subsection}{پاسخ‌ها}

\begin{solutionbox}{\ref{exr:split}}
(الف) \acro{MME} (با تحریک \lr{Downlink Data Notification} از \lr{S-GW})؛ (ب) \lr{S-GW}؛ (ج) \lr{P-GW}؛ (د) \acro{MME}؛
(هـ) \acro{MME}.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:guti}}
حریم خصوصی: \acro{IMSI} دائمی است، پس ارسال آشکار آن به شنودگر اجازهٔ ردیابی مشترک را می‌دهد؛ \acro{GUTI} دوباره تخصیص
می‌یابد. کارایی و مسیریابی: \lr{S-TMSI} ۴۰ بیتی از \acro{IMSI} کوتاه‌تر است و کد \acro{MME} آن به \acro{eNB} می‌گوید کدام عضو مخزن
زمینه را نگه می‌دارد.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:paging}}
$\E[M] = 1 + 0.6\times30 + 0.2\times90 = 37$ سلول در برابر ۹۰؛ $\E[D] = (1 + 0.6 + 0.2)\,T_\mathrm{p} = 1.8\,T_\mathrm{p}$. برای
صدا، تأخیر برقراری تماس حیاتی است و \acro{UE} باید فوراً یافته شود: کل فهرست \acro{TAI} را فراخوانی کنید (یا دست‌کم مرحلهٔ
تک‌سلولی را حذف کنید).
\end{solutionbox}

\begin{solutionbox}{\ref{exr:weights}}
پیش از خرابی: $2/7 = 28.6\,\%$ برای هر \acro{MME} با وزن ۲ و $1/7 = 14.3\,\%$ برای هر \acro{MME} با وزن ۱. پس از خرابی: وزن کل ۵،
پس $40\,\%$ برای \acro{MME} باقی‌مانده با وزن ۲ و $20\,\%$ برای هر \acro{MME} با وزن ۱. ظرفیت از ۷ به ۵ واحد کاهش می‌یابد،
پس بهره‌وری عادی نباید از $5/7 = 71\,\%$ فراتر رود.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:overload}}
(الف) $1/(\mu-\lambda) \le 0.01 \Rightarrow \lambda \le 4900$\,s$^{-1}$ ($\rho \le 0.98$)؛ با ترافیک انفجاری حد واقعی کمتر است.
(ب) $2\times10^6/60 = 3.3\times10^4$ اتصال در ثانیه، تقریباً هفت برابر $\mu$ (و هر اتصال چند پیام دارد). پایداری به کنترل
اضافه‌بار \lr{S1AP} (\acro{eNB}ها اتصال‌ها را رد یا به تأخیر می‌اندازند)، ممنوعیت کلاس دسترسی و عقب‌نشینی \acro{RACH} در
\acro{UE}، زمان‌سنج‌های تصادفی تلاش مجدد \acro{NAS} و یک مخزن که چند \acro{MME} بار را تقسیم کنند نیاز دارد.
\end{solutionbox}

\begin{solutionbox}{\ref{exr:t3412}}
$5\times10^6/3240 = 1543$ \acro{TAU} بر ثانیه برای ۵۴ دقیقه؛ $5\times10^6/720 = 6944$ \acro{TAU} بر ثانیه برای ۱۲ دقیقه.
زمان‌سنج کوتاه \acro{UE}های گم‌شده را زودتر تشخیص می‌دهد (تشخیص حدود \lr{T3412} به‌علاوهٔ حاشیهٔ دسترس‌پذیری طول
می‌کشد) اما سیگنالینگ را چند برابر و باتری را تخلیه می‌کند. دستگاه‌های \acro{IoT} به‌ندرت ارسال می‌کنند و باید سال‌ها با
یک باتری کار کنند؛ زمان‌سنج‌های گسترش‌یافتهٔ چندساعته همراه با حالت صرفه‌جویی توان، سیگنالینگ و مصرف انرژی آن‌ها را
کاهش می‌دهند و شبکه می‌پذیرد که بیشتر وقت دست‌نیافتنی باشند.
\end{solutionbox}
